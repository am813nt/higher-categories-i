\chapter{Simplicially Enriched Categories}
    The goal of this section is to introduce the homotopy coherent nerve of Cordier and Porter which will be our main source of examples of quasi-categories. First we will have to define the input objects, and this will take us back to ordinary category theory for a while.

    \section{Definition}\label{chIV-1}
        A \emph{monoidal structure} on a category $\Cc$ consists of
        \begin{enumerate}[label=\roman*)]
            \item
                a functor $\mu \colon \Cc^2 \to \Cc$, usually written as
                \begin{equation*}
                    (X, Y) \mapsto X \boxtimes Y \,,
                \end{equation*}
            \item
                an object $\1 \in \Cc$, regarded as a functor
                \begin{equation*}
                    \1 \,\colon * \to \Cc
                \end{equation*}
            \item
                a natural isomorphism
                \begin{equation*}
                    \alpha \,\colon \mu \circ \left( \mu \times \id_{\Cc} \right) \Rightarrow \mu \circ \left( \id_{\Cc} \times \mu \right)
                \end{equation*}
                of functors $\Cc^3 \to \Cc$, and
            \item
                natural isomorphisms
                \begin{align*}
                    \eta_L \,\colon \id_{\Cc} &\Rightarrow \mu \circ \left( \1 \times \id_{\Cc} \right) \\
                    \eta_R \,\colon \id_{\Cc} &\Rightarrow \mu \circ \left( \id_{\Cc} \times \1 \right)
                \end{align*}
        \end{enumerate}
        such that the following two diagrams commute:
        \begin{equation*}
            \begin{tikzcd}
                \mu \circ (\id \times \mu) \circ (\mu \times \id \times \id) \arrow[r, equal]                                   & \mu \circ (\mu \times \id) \circ (\id \times \id \times \mu) \arrow[d, "\alpha", Rightarrow] \\
                \mu \circ (\mu \times \id) \circ (\mu \times \id \times \id) \arrow[u, "\alpha"', Rightarrow] \arrow[d, "\alpha", Rightarrow] & \mu \circ (\id \times \mu) \circ (\id \times \id \times \mu)                                 \\
                \mu \circ (\mu \times \id) \circ (\id \times \mu \times \id) \arrow[r, "\alpha"', Rightarrow]                                 & \mu \circ (\id \times \mu) \circ (\id \times \mu \times \id) \arrow[u, "\alpha", Rightarrow]
            \end{tikzcd}
        \end{equation*}
        \smallskip
        
        \begin{equation*}
            \begin{tikzcd}[row sep=tiny, column sep=normal]
                                                                     & \mu \circ (\mu \times \id) \circ (\id \times \1 \times \id) \arrow[dd, "\alpha", Rightarrow] \\
\mu = \mu \circ (\id \times \id) \arrow[ru, "\eta_R", Rightarrow, start anchor={[yshift=-0.5ex]north east}, end anchor={[yshift=-0.5ex]real west}] \arrow[rd, "\eta_L"', Rightarrow, start anchor={[yshift=0.5ex]south east}, end anchor={[yshift=0.5ex]real west}] &                                    \\
                                                                     & \mu \circ (\id \times \mu) \circ (\id \times \1 \times \id)                                 
            \end{tikzcd}
        \end{equation*}
        These diagrams encode that
        \begin{equation*}
            \begin{tikzpicture}[
                scale=1.2,   % Adjust scale to fit the long equations
                implies/.style={
                    double, 
                    double equal sign distance, 
                    -Implies
                },    
            ]
            
                \node (Top) at (90:2.5cm) {$((X \boxtimes Y) \boxtimes W) \boxtimes Z$};

                \node (TopRight) at (18:2.5cm) {$(X \boxtimes (Y \boxtimes W)) \boxtimes Z$};

                \node (BotRight) at (-54:2.5cm) {$X \boxtimes ((Y \boxtimes W) \boxtimes Z)$};

                \node (BotLeft) at (-126:2.5cm) {$X \boxtimes (Y \boxtimes (W \boxtimes Z))$};

                \node (TopLeft) at (162:2.5cm) {$(X \boxtimes Y) \boxtimes (Y \boxtimes Z)$};
            
                \draw[implies] (Top) -- node[sloped, above] {} (TopRight);

                \draw[implies] (BotRight) -- node[right, xshift=2pt] {} (TopRight);

                \draw[implies] (BotLeft) -- node[sloped, below] {} (BotRight);

                \draw[implies] (TopLeft) -- node[sloped, above] {} (Top);

                \draw[implies] (TopLeft) -- node[left, xshift=-2pt] {} (BotLeft);

            \end{tikzpicture}
        \end{equation*}
        and
        \begin{equation*}
            \begin{tikzcd}[row sep=tiny, column sep=normal]
                                                                     & (X \boxtimes \1) \boxtimes Y \arrow[dd, Rightarrow] \\
X \boxtimes Y \arrow[ru, Rightarrow, start anchor={[yshift=-0.5ex]north east}, end anchor={[yshift=-0.5ex]real west}] \arrow[rd, Rightarrow, start anchor={[yshift=0.5ex]south east}, end anchor={[yshift=0.5ex]real west}] &                                    \\
                                                                     & X \boxtimes (\1 \boxtimes Y)                                 
            \end{tikzcd}
        \end{equation*}
        commute. In Mac Lane's book there is a third axiom, namely $\eta_L = \eta_R \colon \1 \to \1 \boxtimes \1$, which we will need later, so let me prove it.

    \section{Lemma (Kelly)}
        The diagrams
        \begin{equation*}
            \begin{tikzcd}[row sep=tiny, column sep=large]
                                                                     & (X \boxtimes Y) \boxtimes \1 \arrow[dd, "\alpha", Rightarrow] \\
(X \boxtimes Y) \arrow[ru, "\eta_R", Rightarrow, start anchor={[yshift=-0.5ex]north east}, end anchor={[yshift=-0.5ex]real west}] \arrow[rd, "\id_X \boxtimes \, \eta_R"' sloped, Rightarrow, start anchor={[yshift=0.5ex]south east}, end anchor={[yshift=0.5ex]real west}] &                                    \\
                                                                     & X \boxtimes (Y \boxtimes \1)                                 
            \end{tikzcd}
            \quad
            \begin{tikzcd}[row sep=tiny, column sep=large]
                                                                     & \1 \boxtimes (X \boxtimes Y) \arrow[dd, "\alpha", Rightarrow] \\
(X \boxtimes Y) \arrow[ru, "\eta_L", Rightarrow, start anchor={[yshift=-0.5ex]north east}, end anchor={[yshift=-0.5ex]real west}] \arrow[rd, "\eta_L \boxtimes \, \id_Y"' sloped, Rightarrow, start anchor={[yshift=0.5ex]south east}, end anchor={[yshift=0.5ex]real west}] &                                    \\
                                                                     & (\1 \boxtimes X) \boxtimes Y                                 
            \end{tikzcd}            
        \end{equation*}
        commute for all $X, Y \in \Cc$, and consequently
        \begin{equation*}
            \eta_R = \eta_L \,\colon \1 \to \1 \boxtimes \1 \,.
        \end{equation*}
        \smallskip

        \emph{Proof}. Consider the diagram
        \begin{equation*}
            \begin{tikzcd}[sep=huge]
                ((\1 \boxtimes \1) \boxtimes Y) \arrow[r, Rightarrow, "\alpha"] \arrow[rrdd, phantom, "?" very near end, end anchor={north}] & (\1 \boxtimes (\1 \boxtimes X)) \boxtimes Y \arrow[rd, Rightarrow, "\alpha"] & \\
                (\1 \boxtimes (\1 \boxtimes X)) \boxtimes Y \arrow[u, Rightarrow, "(\eta_R \, \boxtimes \, \id_X) \, \boxtimes \, \id_Y"] \arrow[rd, Rightarrow, "\alpha"] \arrow[ru, Rightarrow, "(\id_{\1} \, \boxtimes \, \eta_L) \, \boxtimes \, \id_Y" sloped] & & \1 \boxtimes ((\1 \boxtimes X) \boxtimes Y) \arrow[d, Rightarrow, "\alpha"] \\ 
                & \1 \boxtimes (X \boxtimes Y) \arrow[r, Rightarrow, "\id_{\1} \, \boxtimes \, \eta_L"'] \arrow[ru, Rightarrow, "\id_{\1} \, \boxtimes \, (\eta_L \, \boxtimes \, \id_Y)" sloped] & \1 \boxtimes (\1 \boxtimes (X \boxtimes Y))                      
            \end{tikzcd}            
        \end{equation*}
        Then commutativity of the right triangle in the claim is then equivalent to that of the triangle labeled with a question mark, as the latter is $\1 \boxtimes -$ applied to the former (and $\1 \boxtimes - \overset{\eta_L}{\cong} \id_{\Cc}$ is an equivalence of categories). But because of the left and middle diagrams commuting and consisting of isomorphisms, this equivalent to the commutativity of the outer diagram
        \begin{equation*}
            \begin{tikzcd}[sep=huge]
                ((\1 \boxtimes \1) \boxtimes Y) \arrow[r, Rightarrow, "\alpha"] \arrow[rd, Rightarrow, magenta, "\alpha"] & (\1 \boxtimes (\1 \boxtimes X)) \boxtimes Y \arrow[rd, Rightarrow, "\alpha"] & \\
                (\1 \boxtimes (\1 \boxtimes X)) \boxtimes Y \arrow[u, Rightarrow, "(\eta_R \, \boxtimes \, \id_X) \, \boxtimes \, \id_Y"] \arrow[rd, Rightarrow, "\alpha"] & |[text=magenta]|(\1 \boxtimes \1) \boxtimes (X \boxtimes Y) \arrow[rd, Rightarrow, magenta, "\alpha"] & \1 \boxtimes ((\1 \boxtimes X) \boxtimes Y) \arrow[d, Rightarrow, "\alpha"] \\ 
                & \1 \boxtimes (X \boxtimes Y) \arrow[r, Rightarrow, "\id_{\1} \, \boxtimes \, \eta_L"'] \arrow[u, Rightarrow, magenta, "\eta_R \, \boxtimes \, (\id_X \boxtimes \id_Y)"' near start] & \1 \boxtimes (\1 \boxtimes (X \boxtimes Y))                      
            \end{tikzcd}            
        \end{equation*}
        But this follows because of the pink stuff (left parallelogram - naturality of $\alpha$, upper right pentagon - pentagon axiom, lower triangle - unit axiom). The claim for $\eta_R$ follows by the dual argument. To obtain $\eta_L = \eta_R \colon \1 \to \1 \boxtimes \1$ observe that it again suffices to prove $\eta_L \boxtimes \1 = \eta_R \boxtimes \1 \colon \1 \boxtimes \1 \to (\1 \boxtimes \1) \boxtimes \1$. But from naturality of $\eta_L$ we find the commutative diagram
        \begin{equation*}
            \begin{tikzcd}
                \1 \arrow[r, "\eta_L"] \arrow[d, "\eta_L"'] & \1 \boxtimes \1 \arrow[d, "\1 \boxtimes \eta_L"] \\
                \1 \boxtimes \1 \arrow[r, "\eta_L"] & \1 \boxtimes (\1 \boxtimes \1)
            \end{tikzcd}
        \end{equation*}
        which shows
        \begin{equation*}
            \1 \boxtimes \eta_L = \eta_L \,\colon \1 \boxtimes \1 \to \1 \boxtimes (\1 \boxtimes \1) \,.
        \end{equation*}
        But the first part of the proposition then gives $\1 \boxtimes \eta_L = (\eta_L \boxtimes \1) \circ \alpha$. But from the unit axiom we also find $\1 \boxtimes \eta_L = (\eta_R \boxtimes \1) \circ \alpha$, so
        \begin{equation*}
            (\eta_R \boxtimes \1) \circ \alpha = (\eta_L \boxtimes \1) \circ \alpha
        \end{equation*}
        which gives the claim since $\alpha$ is an isomorphism.\hfill$\Box$
        \medskip

        Now Mac Lane proved that given the commutativity of these diagrams, \emph{every} sensible diagram in $\Cc$ built from objects of $\Cc$ by tensoring and units commutes; more formally, we showed that there is a canonical isomorphism between any two meaningful ways of positioning brackets and units in the expression
        \begin{equation*}
            X_1 \boxtimes X_2 \boxtimes \ldots \boxtimes X_n \,,
        \end{equation*}
        e.g.
        \begin{equation*}
            ((X_1 \boxtimes \1) \boxtimes X_2) \boxtimes X_3 \quad \text{and} \quad (\1 \boxtimes (X_1 \boxtimes X_2)) \boxtimes (\1 \boxtimes X_3)
        \end{equation*}
        but not
        \begin{equation*}
            (X_1 \boxtimes \1 \boxtimes X_2) \boxtimes X_3 \,,
        \end{equation*}
        for any $X_1, \ldots, X_n \in \Cc$ that is any two ``formal'' ways of moving between such expressions give the same result. Of course such an isomorphism is generally not unique: On the one hand for the stupid reason that there can for example be isomorphisms $(X \boxtimes Y) \boxtimes Z \to X \boxtimes (Y \boxtimes Z)$ that have nothing to do with the associator, but also for the more subtle reason that it may happen that two differently bracketed expressions are "accidentally" equal in $\Cc$, which allows one to build non-commuting diagrams made solely from associators, by skipping some rebracketings. See Mac Lane's book for a precise statement of this \emph{coherence theorem}.

    \section{Remark}
        Of course the conclusion of the coherence theorem should really be the \emph{definition} of a monoidal category, and the theorem should be that it suffices to check the pentagon and unit axioms to establish a monoidal structure! The same already goes for associativity in algebra: its definition should be that any bracketing of $a_1 \cdot \ldots \cdot a_n$ gives the same result and it is a proposition that $(a_1 a_2) a_3 = a_1 (a_2 a_3)$ already implies this. This becomes visible when associativity is replaced by weaker conditions such as \emph{alternativity} (meaning that bracketing of $a_1 \cdot \ldots \cdot a_n$ only need to agree if $| \left\{ a_1, \ldots, a_n \right\}| \leq 2$) or \emph{power-associativity} (rebracketing requirement only for $| \left\{ a_1, \ldots, a_n \right\}| = 1$, i.e. $a_1 = a_2 = \ldots = a_n$). That rebracketing for expressions of length $3$ suffices is a non-trivial theorem of Artin in the former case, and simply false in the latter!

        In the case of quasi-categories there can also not be a coherence theorem, albeit for a different reason: The commutativity of a rebracketing diagram is then a \emph{datum} and not a condition and larger and larger diagrams require higher and higher simplices as witnesses for their commutativity. The definition of a monoidal quasi-category is therefore necessarily quite different from \ref{chIV-1}; in fact, it is much more elegant as it avoids the choices we shall have to make in the examples below.

    \section{Examples}
        \begin{enumerate}[label=\roman*)]
            \item 
                Using your favorite model of the product of two sets and your favorite one element set $*$ ($\left\{ \varnothing \right\}$, no?) we obtain a functor
                \begin{equation*}
                    - \times - \,\colon \Set \to \Set
                \end{equation*}
                and as part of the package come bijections
                \begin{equation*}
                    (X \times Y) \times Z \longleftrightarrow X \times (Y \times Z)
                \end{equation*}
                and
                \begin{equation*}
                    * \times Z \longleftrightarrow Z \longleftrightarrow Z \times *
                \end{equation*}
                which give the associator and unit isomorphisms of a monoidal structure on $\Set$.
            \item 
                Let $\Cc$ be a category which admits finite products. Then pick a functor
                \begin{equation*}
                    \mu \,\colon \Cc^2 \to \Cc \,,
                \end{equation*}
                which upon postcomposition with Yoneda's embedding $\Cc \to \mathrm{P}(\Cc)$ becomes naturally isomorphic to
                \begin{equation*}
                    \Cc^2 \overset{Y_{\Cc^2}}{\longrightarrow} \mathrm{P}(\Cc^2) \overset{\Delta^*}{\longrightarrow} \mathrm{P}(\Cc)
                \end{equation*}
                say via an isomorphism $\varphi$ and let furthermore $* \in \Cc$ be a terminal object; the assumptions exactly guarantee the existence of such objects! \\
                Unwinding definitions the functor $\mu$ sends $(X, Y)$ to a choice of product $X \times Y$. Now $\varphi$ induces isomorphisms of both composites of
                \begin{equation*}
                    \mu \circ (\id_{\Cc} \circ \mu) \quad \text{and} \quad \mu \circ (\mu \circ \id_{\Cc})
                \end{equation*}
                and the Yoneda embedding of $\Cc$ with
                \begin{equation*}
                    \Cc^3 \overset{Y_{\Cc^3}}{\longrightarrow} \mathrm{P}(\Cc^3) \overset{\Delta}{\longrightarrow} \mathrm{P}(\Cc)
                \end{equation*}
                and the resulting isomorphism between the two composites is a candidate associator. \\
                Similarly,
                \begin{equation*}
                    \mu \circ (\id_{\Cc} \times *) \quad \text{and} \quad \mu \circ (* \times \id_{\Cc})
                \end{equation*}
                composed with $Y_{\Cc}$ both represent the Yoneda embedding of $\Cc$ itself, so are canonically equivalent to the identity. Checking the axioms is tedious but easy.\\
                what we have just constructed is a \emph{cartesian} monoidal structure on $\Cc$; by definition this is a monoidal structure with terminal unit such that the resulting maps
                \begin{gather*}
                    X \boxtimes Y \longrightarrow X \boxtimes \1 \overset{Y^{-1}_R}{\longrightarrow} X \\
                    X \boxtimes Y \longrightarrow \1 \boxtimes Y \overset{Y^{-1}_L}{\longrightarrow} Y
                \end{gather*}
                exhibit $X \boxtimes Y$ as a product of $X$ and $Y$. Such a structure is not unique but once we have introduced monoidal functors, it is easy to see that they form the objects of a contractible groupoid, so are essentially unique, i.e. the choices occurring in the construction above do not really matter.\\
                A dual discussion applies to \emph{cocartesian} monoidal structures on categories with finite coproducts.
            \item 
                The most prominent example of a monoidal structure that is neither cartesian nor cocartesian is the tensor product $\otimes_R$ on the category $\Mod{R}$, for $R$ a commutative ring, the unit is the ring $R$ itself. Similarly, for a not necessarily commutative ring $A$ the category of $A$-$A$-bimodules, which is just $\Mod{A \otimes A^{\op}}$, also carries a monoidal structure given by $\otimes_A$ with unit $A$.
            \item 
                For a category $\Cc$ the category $\Fun(\Cc, \Cc)$ carries a monoidal structure given by composition with unit $\id_{\Cc}$. This is an example of a \emph{strict} monoidal category, that is one where the associator and unitors $\alpha, \eta_L, \eta_R$ are in fact identities.
            \item 
                For $M$ a monoid the category $\mathbb{D}M$ is also strictly monoidal. Similarly a commutative monoid $N$ gives rise to a monoidal structure on $\mathbb{B}N$.
            \item 
                A monoidal structure on $\Cc$ induces one on $\Cc^{\op}$, by sending $(\mu, \1, \alpha, \eta_L, \eta_R)$ to \\
                $(\mu^{\op}, \1, \alpha^{-1}, \eta_L^{-1}, \eta_R^{-1})$, and also a reverse monoidal structure on $\Cc$ via $(\mu \circ \tau, \1, \alpha^{-1} \circ \sigma, \eta_R, \eta_L)$, where $\sigma \colon \Cc^3 \to \Cc^3,\; \tau \colon \Cc^2 \to \Cc^2$ are the order reversals. For example,
                \begin{equation*}
                    \mathbb{D}(M^{\op}) \simeq (\mathbb{D}M)^{\text{rev}}, \qquad \Fun(\Cc^{\op}, \Cc^{\op}) \simeq \Fun(\Cc, \Cc)^{\op} 
                \end{equation*}
                as monoidal categories.
        \end{enumerate}

    \section{Definition}
        A \emph{lax monoidal functor} between monoidal categories $\Cc$ and $\Dd$ consists of
        \begin{enumerate}[label=\roman*)]
            \item 
                a functor $\Cc \to \Dd$
            \item 
                a natural transformation
                \begin{equation*}
                    \lambda \,\colon \mu_{\Cc} \circ (F \times F) \Rightarrow F \circ \mu_{\Dd}
                \end{equation*}
                of functors $\Cc^2 \to D$
            \item 
                a map
                \begin{equation*}
                    u \,\colon \1_{\Dd} \to F(\1_{\Cc})
                \end{equation*}
        \end{enumerate}
        such that
        \begin{equation*}
            \begin{tikzcd}
                (FX \underset{\Dd}{\boxtimes} FY) \underset{\Dd}{\boxtimes} FZ  \arrow[r, "\lambda"] \arrow[d, "\alpha_{\Dd}"'] & F(X \underset{\Cc}{\boxtimes} Y) \underset{\Dd}{\boxtimes} FZ \arrow[d, "\lambda"] \\
                FX \underset{\Dd}{\boxtimes} (FY \underset{\Dd}{\boxtimes} FZ) \arrow[d, "\lambda"'] & F((X \underset{\Cc}{\boxtimes} Y) \underset{\Cc}{\boxtimes} Z) \arrow[d, "\alpha_{\Cc}"] \\
                FX \underset{\Dd}{\boxtimes} F(Y \underset{\Cc}{\boxtimes} Z) \arrow[r, "\lambda"] & F(X \underset{\Cc}{\boxtimes} (Y \underset{\Cc} {\boxtimes} Z))
            \end{tikzcd}
        \end{equation*}
        and
        \begin{equation*}
            \begin{tikzcd}
                F(X) \arrow[r, "F \eta^{\Cc}_R"] \arrow[d, "\eta^{\Dd}_R"'] & F(X \boxtimes \1_{\Cc}) \\
                F(X) \underset{\Dd}{\boxtimes} \1_{\Dd} \arrow[r, "u"] & F(X) \boxtimes F(\1_{\Cc}) \arrow[u, "\lambda"']
            \end{tikzcd}
            \quad
            \begin{tikzcd}
                F(X) \arrow[r, "F \eta^{\Cc}_L"] \arrow[d, "\eta^{\Dd}_L"'] & F(\1_{\Cc} \boxtimes X) \\
                \1_{\Dd} \underset{\Dd}{\boxtimes} F(X) \arrow[r, "u"] & F(\1_{\Cc}) \underset{\Dd}{\boxtimes} F(X) \arrow[u, "\lambda"']
            \end{tikzcd}
        \end{equation*}
        commute. The triple $(F, \lambda, u)$ is called (strong) \emph{monoidal} if both $\lambda$ and $u$ are isomorphisms. Dually, an \emph{oplax monoidal functor} has as data
        \begin{enumerate}[label=\roman*)]
            \item 
                a functor $F \colon \Cc \to \Dd$
            \item 
                a natural transformation
                \begin{equation*}
                    \omega \,\colon F \circ \mu_{\Dd} \Rightarrow \mu_{\Cc} \circ (F \times F) 
                \end{equation*}
            \item 
                a map
                \begin{equation*}
                    c \,\colon F(\1_{\Cc}) \to \1_{\Dd}
                \end{equation*}
        \end{enumerate}
        satisfying dual axioms.
        \medskip

        Note that a monoidal functor in particular also gives rise to an oplax monoidal one (by inverting $\lambda$ and $u$), so we find categories
        \begin{equation*}
            \MonCat^{\lax} \supseteq \MonCat \subseteq \MonCat^{\oplax} \,.
        \end{equation*}

    \section{Examples}
        \begin{enumerate}[label=\roman*)]
            \item 
                If $\Cc$ and $\Dd$ carry cartesian monoidal structures then any functor $F \colon \Cc \to \Dd$ admits an unique oplax monoidal structure: We have to provide maps
                \begin{equation*}
                    F(X \times Y) \to FX \times FY \quad \text{and} \quad F(*_{Cc}) \to *_{\Dd} \,. 
                \end{equation*}
                But for the latter there is an unique choice since $*_{\Dd}$ is terminal, and for the former we have to take the map induced by
                \begin{gather*}
                    F(X \times Y) \to F(X \times *_{\Cc}) \overset{F_{\eta^{-1}_R}}{\rightarrow} F(X) \\
                    F(X \times Y) \to F(*_{\Cc} \times Y) \overset{F_{\eta^{-1}_L}}{\rightarrow} F(Y)
                \end{gather*}
                Similarly, if $\Cc$ and $\Dd$ carry cocartesian structures then all functors $\Cc \to \Dd$ have unique lax monoidal structures. \\
                These structures are strong monoidal if and only if $F$ preserves finite products or coproducts, respectively.\label{chIV-6-i}
            \item 
                For an adjoint pair $L \colon \Cc \rightleftarrows \Dd \colon R$ between monoidal structures, there is a bijection between
                \begin{equation*}
                    \left\{ \text{oplax monoidal structures on L} \right\} \quad \text{and} \quad \left\{ \text{lax monoidal structures on R} \right\}
                \end{equation*}\label{chIV-6-ii}
            \item 
                Here is a concrete example: Consider the adjoint functors
                \begin{equation*}
                    \text{free} \colon \Set \rightleftarrows \Ab \colon \text{forget}
                \end{equation*}
                Then we have 4 monoidal structures to play with: $(\Set, +)$, $(\Set, \times)$, $(\Ab, \oplus)$ and $\Ab, \otimes$. For the cocartesian structures we find the lax structures given by the evident maps
                \begin{align*}
                    \mathbb{Z}[S] \oplus \mathbb{Z}[T] &\to \mathbb{Z}[S + T],  &&0 \to 0 = \mathbb{Z}[\varnothing] &&\in \Ab \\
                    A + B &\to A \oplus B, &&\varnothing \to 0 &&\in \Set
                \end{align*}
                of which the former is clearly strong monoidal (this proves the linear algebra fact, that bases of products are unions of bases). As there are no maps $0 \to \varnothing$ the forgetful functor does not carry an oplax structure. \\
                For the cartesian structures we find the oplax structures given by
                \begin{align*}
                    \mathbb{Z}[S \times T] &\to \mathbb{Z}[S] \oplus \mathbb{Z}[T], &&\mathbb{Z}[*] = \mathbb{Z} \to 0 &&\in \Ab \\
                    A \oplus B &\overset{\id}{\rightarrow} A \times B, &&0 \to + &&\in \Set
                \end{align*}
                where clearly the latter functor is strong monoidal as part \ref{chIV-6-i} implies. Conversely, the free functor this time also carries a lax structure: The zero map
                \begin{equation*}
                    \mathbb{Z}[S] \oplus \mathbb{Z}[T] \overset{\circ}{\rightarrow}, \quad 0 \to \mathbb{Z} = \mathbb{Z}[*] \in \Ab
                \end{equation*}
                Finally, the free functor is also strong monoidal
                \begin{equation*}
                    \text{free} \,\colon (\Set, \times) \to (\Ab, \otimes)
                \end{equation*}
                since
                \begin{equation*}
                    \mathbb{Z}[S \times T] \cong \mathbb{Z}[S] \otimes \mathbb{Z}[T], \quad \mathbb{Z}[*] \cong \mathbb{Z} \,. 
                \end{equation*}
                This also admits a categorical proof, but let us just observe that it gives a lax monoidal structure to the forgetful functor, and once you're done the exercise proving \ref{chIV-6-ii} it will unwind to give
                \begin{equation*}
                    A \times B \to A \otimes B, \quad * \overset{1}{\rightarrow} \mathbb{Z}
                \end{equation*}
                where the former map is the canonical bilinear form, which can conversely be used to construct the map
                \begin{equation*}
                    \mathbb{Z}[S \times T] \to \mathbb{Z}[S] \otimes \mathbb{Z}[T]
                \end{equation*}
                via \ref{chIV-6-ii}. I don't think there are any refinements in the final case $(\Set, +)$ and $(\Ab, \otimes)$.
            \item 
                the functors
                \begin{equation*}
                    \Hom_{\Cc}(\1, -) \colon \Cc \to \Set, \quad \Hom_{\Cc}(-, \1) \colon \Cc^{\op} \to \Set
                \end{equation*}
                are both lax monoidal for the cartesian structure on $\Set$: We require a map
                \begin{equation*}
                    * \to \Hom_{\Cc}(\1, \1)
                \end{equation*}
                in both cases, for which we take $\id_{\1}$, and then maps
                \begin{gather*}
                    \Hom_{\Cc}(\1, X) \times \Hom_{\Cc}(\1, Y) \to \Hom_{\Cc}(\1, X \boxtimes Y) \\
                    \Hom_{\Cc}(X, \1) \times \Hom_{\Cc}(Y, \1) \to \Hom_{\Cc}(X \boxtimes Y, \1)
                \end{gather*}
                for which we take
                \begin{equation*}
                    (f \colon \1 \to X, g \colon \1 \to Y) \mapsto (\1 \overset{\eta}{\longrightarrow} \1 \boxtimes \1 \overset{f \otimes g}{\longrightarrow} X \boxtimes Y)
                \end{equation*}
                and the dual construction. Checking the axioms is a good exercise.
        \end{enumerate}

    \section{Definition}
        Let $F, G \colon \Cc \to \Dd$ be lax monoidal functors. Then a transformation $\tau \colon F \to G$ is called \emph{monoidal} if the diagrams
        \begin{equation*}
            \begin{tikzcd}
                FX \underset{\Dd}{\boxtimes} FY \arrow[r, "\lambda_F"] \arrow[d, "\tau"] & F(X \underset{\Cc}{\boxtimes} Y) \arrow[d, "\tau"] \\
                GX \underset{\Dd}{\boxtimes} GY \arrow[r, "\lambda_G"] & G(X \underset{\Cc}{\boxtimes} Y)
            \end{tikzcd}
            \qquad
            \begin{tikzcd}[sep=small]
                & F(\1_{\Cc}) \arrow[dd, "\tau"] \\
                \1_{\Dd} \arrow[ur, "u_F"] \arrow[dr, "u_G"'] & \\
                & G(\1_{\Cc})
            \end{tikzcd}
        \end{equation*}
        commute for all $X, Y \in \Cc$.\\
        The analogous definition applies to transformations of oplax monoidal functors.

        Note that for a strong monoidal functor the two arising notions of monoidal transformation agree, We obtain full inclusions
        \begin{equation*}
            \Fun^{\lax}(\Cc, \Dd) \supseteq \Fun^{\text{mon}}(\Cc, \Dd) \subseteq \Fun^{\oplax}(\Cc, \Dd) \,.
        \end{equation*}
        Note furthermore, that these definitions provide us with the notions monoidal natural isomorphism of functor (which can again be tested pointwise) and monoidal equivalence of categories. Here a priori, there are three notions which might be called lax, oplax and strog monoidal equivalence. These turn out to agree: Two lax monoidal functors
        \begin{equation*}
            F \,\colon \Cc \leftrightarrows \Dd \colon G
        \end{equation*}
        with $F \circ G$ and $G \circ F$ monoidally isomorphic to the respective identities, are automatically strong monoidal (as one easily checks by writing out the definitions) Let me immediately warn you that the same does not apply if $F \colon \Cc \to \Dd$ is lax monoidal and an equivalence of categories, see the example \ref{chIV-8-ii} below.
    
        A word on terminology: Many authors call such transformation lax monoidal and oplax monoidal. There is, however, nothing lax about the transformations or their monoidality, but rather only about the functors they run between. Clearly, that's no good!

    \section{Examples}
        \begin{enumerate}[label=\roman*)]
            \item
                If $\Cc$ and $\Dd$ are cartesian monoidal then any natural transformation between functors $F, G \colon \Cc \to \Dd$ is monoidal for the canonical oplax structure of $F$ and $G$. In particular, the forgetful functor
                \begin{equation*}
                    \Fun^{\oplax}(\Cc, \Dd) \to \Fun(\Cc, \Dd)
                \end{equation*}
                is an isomorphism of categories in this case, and  similarly
                \begin{equation*}
                    \Fun^{\lax}(\Cc, \Dd) \to \Fun(\Cc, \Dd)
                \end{equation*}
                is an isomorphism for cocartesian $\Cc$ and $\Dd$.\label{chIV-8-i}
            \item
                If $\Cc$ admits a \emph{zero-object}, i.e. one that is simultaneously initial and terminal, and both finite products and coproducts, then the identity functor $\id_{\Cc}$ admits a canonical oplax refinement
                \begin{equation*}
                    (\Cc, +) \to (\Cc, \times)
                \end{equation*}
                or equivalently, lax refinement
                \begin{equation*}
                    (\Cc, \times) \to (\Cc, +)
                \end{equation*}
                given by the unique map on units, and
                \begin{equation*}
                    X + Y \to X \times Y
                \end{equation*}
                induced by
                \begin{gather*}
                    X + Y \longrightarrow X + 0 \overset{\eta^{-1}_R}{\longrightarrow} X \\
                    X + Y \longrightarrow 0 + X \overset{\eta^{-1}_L}{\longrightarrow} Y        
                \end{gather*}
                or equivalently
                \begin{gather*}
                    X  \overset{\eta_R}{\longrightarrow} X \times 0 \longrightarrow X \times Y \\
                    X  \overset{\eta_L}{\longrightarrow} 0 \times X \longrightarrow X \times Y \,;
                \end{gather*}
                concisely, by the matrix
                \begin{equation*}
                    \begin{pmatrix} \id_X & 0 \\ 0 & \id_Y \end{pmatrix} \,\colon X + Y \to X \times Y \,,
                \end{equation*}
                where $0$ denotes the unique map of any object factoring through the zero-object. This refinement is not strong for example fo $\nicefrac{*}{\Set}$.\label{chIV-8-ii}
            \item
                If $R \to S \to T$ are homomorphisms of commutative rings then the functors
                \begin{align*}
                    &\Mod{R} \to \Mod{S}, &&M \mapsto S \otimes_R M \\
                    &\Mod{R} \to \Mod{T} \overset{\text{forget}}{\longrightarrow} \Mod{S}, &&M \mapsto T \otimes_R M
                \end{align*}
                are lax monoidal for the respective tensor product and the upper even strongly so. The map $S \to T$ then induces a monoidal transformation
                \begin{equation*}
                    S \otimes_R - \implies T \otimes_R - \,.
                \end{equation*}
        \end{enumerate}

    \section{Definition}\label{chIV-9}
        Let $\Vv$ be a monoidal category. Then a \emph{$\Vv$-enriched category} $\Cc$ consists of
        \begin{enumerate}[label=\roman*)]
            \item 
                a class $O$
            \item 
                a map $F \colon O \times O \to \obj(\Vv)$
            \item 
                a map $i \colon O \to \mor(\Vv)$
            \item 
                a map
                \begin{equation*}
                    \comp \colon O \times  O \times O \to \mor(\Vv)
                \end{equation*}
        \end{enumerate}
        such that
        \begin{enumerate}[label=\alph*)]
            \item
                \begin{equation*}
                    s (\comp(x, y, z)) = F(y, z) \boxtimes F(x, y)
                \end{equation*}
                and
                \begin{equation*}
                    t(\comp(x, y, z)) = F(x, z)
                \end{equation*}
            \item
                \begin{equation*}
                    s(i(x)) = \1, \quad t(i(x)) = F(x, x)
                \end{equation*}
            \item
                the diagrams
                    \begin{equation*}
                        \begin{tikzcd}
                            (F(y, z) \boxtimes F(x, y)) \boxtimes F(w, x) \arrow[r, "\comp"] \arrow[d, "\alpha"] & F(x, z) \boxtimes F(w, x) \arrow[dd, "\comp"] \\
                            F(y, z) \boxtimes (F(x, y) \boxtimes F(w, x)) \arrow[d, "\comp"] & \\
                            F(y, z) \boxtimes F(w, y) \arrow[r, "\comp"] & F(w, z)
                        \end{tikzcd}
                    \end{equation*}
                    and
                    \begin{equation*}
                        \begin{tikzcd}
                            F(x, y) \arrow[r, "\id"] \arrow[d, "\eta_R"] & F(x, y) \\
                            F(x, y) \boxtimes \1 \arrow[r, "i"] & F(x, y) \boxtimes F(x, x) \arrow[u, "\comp"]
                        \end{tikzcd}
                        \quad
                        \begin{tikzcd}
                            F(x, y) \arrow[r, "\id"] \arrow[d, "\eta_L"] & F(x, y) \\
                            \1 \boxtimes F(x, y) \arrow[r, "i"] & F(y, y) \boxtimes F(x, y) \arrow[u, "\comp"]
                        \end{tikzcd}        
                    \end{equation*}
                    commute.      
        \end{enumerate}

    \section{Examples}
        \begin{enumerate}[label=\roman*)]
            \item
                A $(\Set, \times)$-enriched category is just an ordinary category.
            \item
                A $(\Top, \times)$-enriched category is an ordinary category $\Cc$ together with a topology on each of the sets $\Hom_{\Cc}(x, y)$ such that composition
                \begin{equation*}
                    \Hom_{\Cc}(x, y) \times \Hom_{\Cc}(w, x) \to \Hom_{\Cc}(w, y)
                \end{equation*}
                is continuous. Let me warn you that the compact-open topology does not unfortunately provide such an enrichment on $\Top$ itself. Some restrictions need to be imposed, making topologically enriched categories somewhat annoying to work with.
            \item 
                An $(\Ab, \otimes)$-enriched category translates to an ordinary category together with an abelian group structure on each $\Hom_{\Cc}(x, y)$ such that composition
                \begin{equation*}
                    \Hom_{\Cc}(x, y) \times \Hom_{\Cc}(w, x) \to \Hom_{\Cc}(w, y)
                \end{equation*}
                is biadditive. Since the question arose in the lecture let me give a brief discussion of he existence and uniqueness of such enrichments. First of all, we did not need inverses for any of the discussions above, and the same decoding works for $(\CMon, \otimes)$-enriched categories. Now a category $\Cc$ which admits both finite products and finite coproducts, which agree (in the sense that the initial element is terminal, and the arising lax transformation
                \begin{equation*}
                    (\Cc, +) \to (\Cc, \times)
                \end{equation*}
                is actually strong), admits an \emph{unique} $(\CMon, \otimes)$-enrichment. \\ 
                Conversely, if a $(\CMon, \otimes)$-enriched category admits one of finite products or finite coproducts, then it necessarily also admits the other, and they agree. Without the existence of products or coproducts, however, even $(\Ab, \otimes)$-enrichments need not be unique! For example an $(\Ab, \otimes)$-enriched category with a single object is the same thing as a ring, and the underlying category precisely encodes the multiplicative monoid.
                \begin{equation*}
                    \mathbb{Z}_4 \quad \text{and} \quad \mathbb{Z}_2[t]/{t^2} 
                \end{equation*}
                are two rings with isomorphic multiplicative monoids, but different additive groups.
        \end{enumerate}

        \section{Construction/Lemma}\label{chIV-15}
            Let $\Cc$ be simplicially enriched. Then for $i \in \mathbb{N}$ consider the category $\Cc_i \in \Cat$ with
            \begin{gather*}
                \obj(\Cc_i) = \obj(\Cc) \\
                \mor(\Cc_i) = \sum_{x, y \in \Cc}F(x, y) \,;
            \end{gather*}
            i.e. $\Hom_{\Cc_i}(x, y)$ consists of the $i$-simplices of $F(x, y)$. It is (hopefully) clear that $\Cc_i$ really is a category and furthermore one readily checks that the $\Cc_i$ assemble into a simplicial category
            \begin{equation*}
                S(\Cc) \,\colon \Delta^{\op} \to \Cat, \quad [i] \mapsto \Cc_i
            \end{equation*}
            with induced functors the identity on objects and the structure maps in $F(x, y)$ on isomorphisms. Conversely, given a simplicial category $T \colon \Delta^{\op} \to \Cat$ such that the functor
            \begin{equation*}
                \Delta^{\op} \overset{T}{\longrightarrow} \Cat \overset{\obj}{\longrightarrow} \Set
            \end{equation*}
            is constant with value $O$, then $T$ yields a simplicially enriched category $\Cc$ with objects $O$ and morphisms $F(x, y) \in \sSet$ given by
            \begin{equation*}
                F_{\Cc}(x, y)_i = \Hom_{T(i)}(x, y)
            \end{equation*}
            with the evident simplicial structure induced from $T$. These processes are evidently inverse to one another.

    \section{Construction/Lemma}\label{chIV-18}

        \begin{equation*}
            \Phi_{*} \,\colon \Cat^{\Vv}_2 \to \Cat^{\Ww}_2
        \end{equation*}
        and this is part of a $2$-functor
        \begin{align*}
            \MonCat^{\lax} &\to \hCat{2}\footnotemark \\
            \Vv &\mapsto \Cat^{\Vv}_2
        \end{align*}
        \footnotetext{$\Cat^{\Cat}_2$, $2$-category of $2$-categories, $2$-functors and $2$-natural transformations}
        \smallskip
        
        \emph{Proof}. To define $\Phi_{*}$ on objects
        \begin{gather*}
            \obj(\Phi_{*} \Cc) \coloneq \obj(\Cc), \quad F_{\Phi_{*} \Cc}(x, y) \coloneq \Phi F_{\Cc}(x, y) \\
            i_{\Phi_{*} \Cc}(x) \coloneq \Phi(i_{\Cc} x) \circ u \,\colon \1_{\Ww} \to \Phi(\1_{\Vv}) \to \Phi F_{\Cc}(x, x)
        \end{gather*}
        with composition
        \begin{equation*}
            \Phi F_{\Cc}(y, z) \underset{\Ww}{\boxtimes} \Phi F_{\Cc}(x, y) \xrightarrow{\lambda} \Phi (F_{\Cc}(y, z) \underset{\Vv}{\boxtimes} F_{\Cc}(x, y)) \xrightarrow{\Phi(- \circ -)} \Phi F_{\Cc}(x, z) \,.
        \end{equation*}
        To define $\Phi_{*}$ on morphisms send a $\Vv$-enriched functor $G \colon \Cc \to \Dd$ to
        \begin{gather*}
            (\Phi_{*} G)_o \coloneq G \,\colon \obj(\Phi_{*} \Cc) = \obj(\Cc) \to \obj(\Dd) = \obj(\Phi_{*} \Dd) \\

        \end{gather*}
        Finally, to define $\Phi_{*}$ on $2$-morphisms send $\tau \colon G \Rightarrow G'$, a $\Vv$-enriched transformation, to
        \begin{equation*}
            (\Phi_{*} \tau)(x) \coloneq \Phi(\tau(x)) \circ u \,\colon \1_{\Ww} \to \Phi(\1_{\Vv}) \to \Phi F_{\Cc}(G x, G' x) = F_{\Phi_{*} \Cc}((\Phi_{*} G) x, (\Phi_{*} G') x) \,.
        \end{equation*}
        One readily checks that these data really give a $2$-functor $\Phi_{*} \colon \Cat^{\Vv}_2 \to \Cat^{\Ww}_2$ as desired.\\

            \MonCat^{\lax}_2 \to \hCat{2} \,.

            F_{\Phi_{*} \Cc}(x, y) = \Phi F_{\Cc}(x, y) \xrightarrow{\tau} \Psi F_{\Cc}(x, y) = F_{\Psi_{*} \Cc}(x, y) \,.

                \end{equation*}
                Then $\pi_0 X = 0$ since $\gr{X} \cong \mathbb{R}           _{\geqslant 0}$, and so also $\prod\limits_{\mathbb{N}} \pi_0 X = 0$. But $\pi_0 \prod\limits_{i \in \mathbb{N}} X \neq 0$, as the $0$-simplices $(0, \ldots, 0, \ldots)$ and $(0,1,2,3, \ldots)$ of $\prod\limits_{i \in \mathbb{N}}X$ cannot be connected by a finite sequence of $1$-simplices in $\prod\limits_{i \in \mathbb{N}}X$. In fact, $\pi_0 \prod\limits_{i \in \mathbb{N}}X$ consists of all sequences in $\mathbb{N}$ modulo the relation of bounded difference. Anyway, we obtain

                    (\pi_0)_{*} \,\colon \Cat^{\text{(k)}\Top} \to \Cat \quad \text{and} \quad (\pi_0)_{*} \,\colon \Cat^{\sSet} \to \Cat
                \end{equation*}
                sending $\Cc$ to its \emph{homotopy category}.

                \begin{equation*}
                    \Cat \to \hCat{2}, \quad \Cat \to \Cat^{\sSet}, \quad \Cat \to \Cat^{\Top}
                \end{equation*}
            \item 
                Lemma \ref{chIV-18} in particular implies that adjoint functors
                \begin{equation*}
                    L \colon \Vv \leftrightarrows \Ww \colon R
                \end{equation*}
                between monoidal categories, such that both $L$ and $R$ are lax monoidal and the unit and counit are monoidal (a lax monoidal adjunction\footnotemark) give rise to adjoint functors
                \begin{equation*}
                    L_{*} \colon \Cat^{\Vv} \leftrightarrows \Cat^{\Ww} \colon R_{*} \,.
                \end{equation*}
                Essentially all functors above come in such adjoint pairs.
            \item 
                The adjunction
                \begin{equation*}
                    h \colon \sSet \leftrightarrows \Cat \colon \Nn
                \end{equation*}
                is also monoidal for the cartesian structures (by \hyperref[chIV-8-i]{IV.8 i)} and exercise 3, sheet 5). We obtain functors
                \begin{equation*}
                    h_{*} \colon \Cat^{\sSet} \leftrightarrows \hCat{2} \colon \Nn_{*}
                \end{equation*}
                embedding $\hCat{2}$ into $\Cat^{\sSet}$ and conversely sending a simplicially enriched category to its \emph{homotopy $2$-category}. Let me immediately warn you that a quasi-category does \emph{not} have a strict homotopy $2$-category. Rather there is a notion of a weak $2$-category, traditionally known as a \emph{bicategory} introduced by Benabou, and every quasi-category has a homotopy bicategory (though we shall not even define bicategories rigorously in this course). A typical example of a bicategory is
                \begin{equation*}
                    \text{BiMod} =
                    \begin{cases*}
                        \text{objects}     & \text{rings} \\
                        \text{morphisms}   & \text{bimodules} \\
                        \text{2-morphisms} & \text{bimodule homomorphisms}
                    \end{cases*}
                \end{equation*}
                with the composition of two morphisms
                \begin{equation*}
                    M \in R \otimes \Mod{S^{\op}} \quad \text{and} \quad N \in S \otimes \Mod{T^{\op}}
                \end{equation*}
                being $M \otimes_S N \in R \otimes \Mod{T^{\op}}$, which is not a strictly associative operation.\\

                    h_{*} \colon \Cat^{\Kan} \leftrightarrows \Cat^{\Grpd} \colon \Nn_{*}
                \end{equation*}
        \end{enumerate}
        \footnotetext{In fact, there's a notion of adjointness in any $2$-category; so the same works for $\Cat^{\Vv}_2 \leftrightarrows \Cat^{\Ww}_2$.}

    \section{Construction}
        Let $\Cc$ be a monoidal category and $\Dd \colon \bbDelta \to \Cc$ a cosimplicial object in $\Cc$, equipped with
        \begin{enumerate}[label=\roman*)]
            \item 
                a \emph{coproduct} $d \colon \Dd \Rightarrow \Dd \boxtimes \Dd$, where the target is
                \begin{equation*}
                    \bbDelta \xrightarrow{\Delta} \bbDelta \times \bbDelta \xrightarrow{\Dd \times \Dd} \Cc \times \Cc \xrightarrow{- \, \boxtimes \, -} \Cc 
                \end{equation*}
            \item 
                a \emph{counit} map
                \begin{equation*}
                    c \,\colon \Dd \Rightarrow \const_{\bbDelta} \1_{\Cc} \quad \color{magenta}{\text{, i.e. a map } \Dd[0] \Rightarrow \1_{\Cc}}
                \end{equation*}
        \end{enumerate}
        such that
        \begin{equation*}
            \begin{tikzcd}[row sep=tiny]
                & \Dd \boxtimes \Dd \arrow[r, Rightarrow, "\id \boxtimes \, d"] & \Dd \boxtimes (\Dd \boxtimes \Dd) \\
                \Dd \arrow[ur, Rightarrow, "d", start anchor={[yshift=-0.5ex]north east}, end anchor={[yshift=-0.5ex]real west}] \arrow[dr, Rightarrow, "d"', start anchor={[yshift=0.5ex]south east}, end anchor={[yshift=0.5ex]real west}] & & \\
                & \Dd \boxtimes \Dd \arrow[r, Rightarrow, "d  \, \boxtimes \, \id"] & (\Dd \boxtimes \Dd) \boxtimes \Dd \arrow[uu, Rightarrow, "\alpha_{\Cc}"']
            \end{tikzcd}
        \end{equation*}
        and
        \begin{equation*}
            \begin{tikzcd}[column sep=tiny]
                \Dd \boxtimes \Dd \arrow[rr, Rightarrow, "\id \boxtimes \, c"] & & \Dd \boxtimes \const_{\bbDelta} \1_{\Cc} \\
                & \Dd \arrow[ul, Rightarrow, "d"] \arrow[ur, Rightarrow, "\eta_R"'] &
            \end{tikzcd}
            \quad
            \begin{tikzcd}[column sep=tiny]
                \Dd \boxtimes \Dd \arrow[rr, Rightarrow, "c \, \boxtimes \, \id"] & & \const_{\bbDelta} \1_{\Cc} \boxtimes \Dd \\
                & \Dd \arrow[ul, Rightarrow, "d"] \arrow[ur, Rightarrow, "\eta_R"'] &
            \end{tikzcd}        
        \end{equation*}
        commute.\\
        Then we can define a simplicially enriched category $\Cc_{\Dd} \in \Cat^{\sSet}$ with
        \begin{align*}
            \obj \Cc_{\Dd} &= \obj \Cc \\
            F_{\Cc_{\Dd}}(X, Y)_n &= \Hom_{\Cc}(X \boxtimes \Dd[n], Y) \\
            \id_{\Cc_{\Dd}} \colon X \boxtimes \Dd[0] &\xrightarrow{\id \times c} X \boxtimes \1_{\Cc} \xrightarrow{\eta^{-1}_R} X
        \end{align*}
        and composition of $f \colon X \boxtimes \Dd[n] \to Y$ and $g \colon Y \boxtimes \Dd[n] \to Z$ given by
        \begin{equation*}
            X \boxtimes \Dd[n] \xrightarrow{\id \boxtimes \, d} X \boxtimes (\Dd[n] \boxtimes \Dd[n]) \xrightarrow{\alpha^{-1}_{\Cc}} (X \boxtimes \Dd[n]) \boxtimes \Dd[n] \xrightarrow{f \, \boxtimes \id} Y \boxtimes \Dd[n] \xrightarrow{g} Z
        \end{equation*}
        which is readily checked ti be unital and associative. The counit also provides a canonical functor
        \begin{equation*}
            c_{*} \,\colon \Cc \to u \Cc_{\Dd}
        \end{equation*}
        which is the identity on objects and send $f \colon X \to Y$ to
        \begin{equation*}
            X \boxtimes \Dd[0] \xrightarrow{\id \boxtimes \, c} X \boxtimes \1_{\Cc} \xrightarrow{\eta^{-1}_R} X \xrightarrow{f} Y \,.
        \end{equation*}
        The functor $c_{*}$ is an isomorphism if $c$ itself is one.

    \section{Construction}
        For a totally ordered set $\Jj$ and $i \leq j \in \Jj$ set
        \begin{equation*}
            \Jj_{i,j} = \left\{ \II \subseteq [i, j] \mid i, j \in \II \right\}
        \end{equation*}
        the poset of paths from $i$ to $j$, ordered by refinement, i.e. inclusion.
        \smallskip

        For $i \leq j \leq k$ there is a map
        \begin{align*}
            \Jj_{j,h} \times \Jj_{i,j} &\to \Jj_{i,k} \\
              (S, T) &\mapsto S \cup T
        \end{align*}
        which is clearly order preserving. We obtain a poset enriched category $\mathfrak{P}\Jj$ with
        \begin{equation*}
            \obj \mathfrak{P}\Jj = \Jj \quad F_{\mathfrak{P}\Jj}(i, j) = \Jj_{i,j}
        \end{equation*}
        with composition given by the union map above and identities the singletons.

But since $F_{\Cc}(G(0), G(n))$ is assumed Kan a filler exists as the inclusion on the left is anodyne.\hfill$\Box$
