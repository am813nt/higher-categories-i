    \section{Definition}
        Two maps $f, g \colon X \to Y$ of simplicial sets are called \emph{homotopic} if $[f] = [g] \in \pi_0 F(X,Y)$. A \emph{simplicial homotopy} from $f$ to $g$ is an element
        \begin{equation*}
            H \in F(X, Y)_1 = \Hom_{\sSet} (X \times \Delta^1, Y)
        \end{equation*}
            with $d_1H = f$ and $d_0H = g$.

        \begin{equation*}
            \tau \,\colon \Hom_{\Cat}(hX, \Cc) \longleftrightarrow \Hom_{\sSet}(X, \Nn\Cc) \;\colon \sigma
        \end{equation*}
        as follows: Given $F \colon hX \to \Cc$ we define
        \begin{equation*}
            \tau(F)_n \,\colon X_n \to \Nn_n\Cc
        \end{equation*}
        by
        \begin{equation*}
            \begin{tikzcd}[column sep=scriptsize]
                X_n \arrow[r, "\cong"] \arrow[d] & \Hom_{\sSet}(\Delta^n, X) \arrow[d, "\text{restriction}"] & \\
                \longpb{X_1}{X_0} \arrow[r, "\cong"] \arrow[d] & \Hom_{\sSet}(\II_n, X) & \\
                \longpb{\mor(hX)}{\obj(hX)} \arrow[r, "F"] & \longpb{\mor(\Cc)}{\obj(\Cc)} \arrow[r, phantom, "\cong"] & \Nn_n(\Cc)
            \end{tikzcd}
        \end{equation*}
        Conversely, given $g \colon X \to \Nn\Cc$, we define
        \begin{align*}
            \sigma(g) \,\colon \obj(hX) &\to \obj(\Cc) \\
            X_0 &\mapsto \Nn_0 \Cc
        \end{align*}
        and $\sigma(g) \mor(hX) \to \mor(\Cc)$ on generators by

            \sigma(g)(f_1, \ldots, f_n) = g_1(f_1) \circ g_1(f_2) \circ \ldots \circ g_1(f_n) \,.

        The maps $\Seg_i \colon X_n \to X_1$ induced by $[1] \to [n], 0 \mapsto i, 1 \mapsto i + 1$ are called the \emph{Segal maps} and always combine into the diagram
        \begin{equation*}
            \begin{tikzcd}[row sep=small, column sep=huge]
                X_n \arrow[r, "{(\Seg_0, \ldots, \Seg_{n - 1})}"] \arrow[d, phantom, "\cong" rotate=90] & \longpb{X_1}{X_0} \arrow[d, phantom, "\cong" rotate=90] \\
                \Hom_{\sSet}(\Delta^n, X) \arrow[r] & \Hom_{\sSet}(\II_n, X)
            \end{tikzcd}
        \end{equation*}
        occurring in the proof, but make sense for any simplicial object. A simplicial object for which they exhibit $X_n$ as the pullback displayed on the right is called a \emph{Segal object}.

        The unit and counit
        \begin{equation*}
            X \to \Nn hX \quad \text{and} \quad h\Nn\Cc \to \Cc
        \end{equation*}
        of the adjunction constructed in the proof are given by
        \begin{equation*}
            X_n \xrightarrow{\Seg} \longpb{X_1}{X_0} \rightarrow \longpb{\mor(hX)}{\obj(hX)} \cong \Nn_n hX
        \end{equation*}
        \begin{equation*}
            \begin{tikzcd}
                \longpb{\Nn_1 \Cc}{\Nn_0 \Cc} = \longpb{\mor{\Cc}}{\obj{\Cc}} \arrow[d, start anchor={[xshift=-9ex]south}, end anchor={[xshift=-3ex]north}] \arrow[r, "\circ"] & \mor{\Cc} \\
                \mor{h \Nn \Cc} \arrow[ur, dashed]
            \end{tikzcd}           
        \end{equation*}

        We next aim to prove:
        \section{Theorem}\label{chII-25}
            The functor $\Nn \colon \Cat \to \sSet$ is fully faithful and its essential image is given equivalently by all simplicial sets satisfying one of two equivalent conditions:
            \begin{enumerate}[label=\alph*)]
                \item 
                    the map $\Hom(\Delta^n, X) \xrightarrow{\text{rest.}} \Hom(\II_n, X)$ is bijective for all $n \geq 0$, in other words $X$ is a Segal set, or\label{chII-25-a}
                \item 
                    the map $\Hom(\Delta^n, X) \rightarrow \Hom(\horn{n}{i}, X)$ is bijective  for all $0 < i < n$.\label{chII-25-b}
            \end{enumerate}

            One says that a simplicial set satisfying \ref{chII-25-b} has "unique inner horn fillers"; said diagrammatically:
            \begin{equation*}
                \begin{tikzcd}
                    \horn{n}{i} \arrow[r] \arrow[d, hook]          & X \\
                    \Delta^n \arrow[ru, "{\exists \, !}"', dashed] &  
                \end{tikzcd}
            \end{equation*}
            for all $0 < i < n$. The proof will consist of several steps: We already observed that nerves satisfy \ref{chII-25-a} and the first step will be to show that Segal sets satisfy \ref{chII-25-b}. For simplicial sets $X$ satisfying a weak version of \ref{chII-25-b} there is a drastically simpler description of $hX$, which we will then use to show that the unit of a simplicial set satisfying \ref{chII-25-b} and counit of an arbitrary category are isomorphisms. Before we dive into that let us, however, observe:

    \section{Lemma}
        For a topological space $X$ the simplicial set $\Sing X$ has (not-necessarily unique) horn fillers for all horns.
        \smallskip
        
        \emph{Proof}. By adjunction the extension problems
        \begin{equation*}
            \begin{tikzcd}[column sep=scriptsize]
                \horn{n}{i} \arrow[r] \arrow[d, phantom, "\subseteq" rotate=-90] & \Sing X \\
                \Delta^n \arrow[ru, dashed]           &        
            \end{tikzcd}
            \quad \text{and} \quad
            \begin{tikzcd}
                \gr{\horn{n}{i}} \arrow[r] \arrow[d, phantom, "\subseteq" rotate=-90] & X \\
                \gr{\Delta^n} \arrow[ru, dashed]           &        
            \end{tikzcd}
        \end{equation*}
        are equivalent. As $\gr{\horn{n}{i}} \subseteq \gr{\Delta^n}$ is the inclusion of a (deformation) retract, the right hand  one clearly admits a solution\footnote{In fact, a contractible space of such solutions.}.\hfill$\Box$

        We can thus give the central definition of this course:
    \section{Definition}
        A simplicial set $X$ is called a \emph{Kan complex} if every horn $\horn{n}{i} \to X$ admits a filler, i.e.
        \begin{equation*}
              \begin{tikzcd}
                \horn{n}{i} \arrow[r] \arrow[d, phantom, "\subseteq" rotate=-90] & X \\
                \Delta^n \arrow[ru, dashed, "\exists"']           &        
            \end{tikzcd}
        \end{equation*}
        $X$ is called a \emph{quasi-category} if $X$ admits fillers for all inner horns, i.e. those $\horn{n}{i}$ with $0 < i < n$.

        Thus a quasi-category is a hybrid of the nerve of a category and the singular complex of a space.

        Before we look more closely at these definitions, let us prove \ref{chII-25}.

    \section{Lemma}
        Every Segal set $X$ has unique inner horn fillers and in fact unique fillers for all horns $\horn{n}{i}$ with $n \geq 4$.
        \smallskip

        \emph{Proof}. Let $f \colon \horn{n}{j} \to X$ with $0 < j < n$ be given. In particular, $n \geq 2$ and since $n = 2$ forces $j = 1$ and $\horn{2}{1} = \II_2$ this case follows from an assumption. For $n \geq 3$ consider the inclusion $\II_n \subseteq \horn{n}{j}$ and let $g \colon \Delta^n \to X$ be the unique extension of ${f|}_{\II_n} \colon \II_n \to X$. We have to show that ${g|}_{\horn{n}{j}} = f$. As
        \begin{equation*}
            \horn{n}{j} = \bigcup_{\substack{0 \leq i \leq n \\ i \neq j}} \im(\Delta^{n-1} \to \Delta^n)
        \end{equation*}
        it suffices to show that
        \begin{equation*}
            f d_i = g d_i \,\colon \Delta^{n-1} \to X \quad \text{for} \enspace i \neq j \,.
        \end{equation*}
        This is easy for $i = 0, n$ since in these cases
        \begin{equation*}
            d_i(\II_{n-1}) \subseteq \II_n
        \end{equation*}
        so, since ${f|}_{\II_n} = {g|}_{\II_n}$ by definition both $f d_i$ ad $g d_i$ are extensions of $f{d_i|}_{\II^{n-1}} = g{d_i|}_{\II^{n-1}}$. But by assumption there is only one such extension so $fd_i = gd_i$. We thus obtain that $f$ equals $g$ on all of
        \begin{equation*}
            \II_n \cup d_0 \Delta^{n-1} \cup d_n \Delta^{n-1} \subseteq \horn{N}{j} \,.
        \end{equation*}
        But now for $0 < i < n$ we still have
        \begin{equation*}
            d_i \II_{n-1} \subseteq \II_n \cup d_0 \Delta^{n-1} \cup d_n \Delta^{n-1} \,\colon
        \end{equation*}
        a $k$-simplex on the left is a map $\varphi \colon [k] \to [n]$ that is monotone and whose image consists of at most two consecutive numbers in $\left\{ 0, \dots, i - 1, i + 1, \ldots, n \right\}$. Thus $\varphi \in \II_n$ unless
        \begin{equation*}
            \im(\varphi) = \left\{ i - 1, i + 1 \right\} \,.
        \end{equation*}
        But then $\varphi \in d_0 \Delta^n$ unless $i - 1 = 0$, in which case $i + 1 = 2 < n$, so $\varphi \in d_n \Delta^n$. Thus we can run the same argument again for all $0 < i < n, \enspace i \neq j$: The restriction of $f d_i$ and $g d_i$ to $\II_{n-1}$ agree by the above, so by uniqueness of extensions $f d_i = g d_i$.
        For the outer horns for $n \geq 4$ observe first that the previous argument shows that the restriction map
        \begin{equation*}
            \Hom(\partial \Delta^n, X) \to \Hom(\II_n, X)
        \end{equation*}
        is bijective for any Segal set (we only had to exclude $i = j$ from our reasoning previously, because $f$ was not defined on $d_j \Delta^n$). But since
        \begin{equation*}
            \begin{tikzcd}[row sep=small, column sep=tiny]
                {\Hom(\Delta^n, X)} \arrow[rd, "\cong"'] \arrow[rr] &                  & {\Hom(\partial \Delta^n, X)} \arrow[ld, "\cong"] \\
                                                    & {\Hom(\II_n, X)} &                                                 
            \end{tikzcd}
        \end{equation*}
        commutes and has two isomorphisms it follows that every Segal set is $2$-coskeletal. But then
        \begin{align*}
            \Hom(\horn{n}{j}, X) &\cong \Hom(\horn{n}{j}, \cosk_2 X) \\                                     &\cong \Hom(\sk_2 \horn{n}{j}, X) \\
                                 &\cong \Hom(\sk_2 \Delta^n, X) \\
                                 &\cong \Hom(\Delta^n, X)
        \end{align*}
        and tracing through shows the map given by restriction.\hfill$\Box$
        \medskip
