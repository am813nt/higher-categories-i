\chapter{Simplicial Homotopy Theory}
\section{Motivation and basic definitions}
    We proved in \emph{Topology II} that the canonical map
    \begin{equation*}
        \gr{\Sing(X)} \to X
    \end{equation*}
    is a weak equivalence for every topological space $X$, so we ``know'' that simplicial sets are sufficient to model (weak) homotopy types. We would like to try to expand on this idea by showing that it is possible to ``do homotopy theory'' in the category of simplicial sets just as one usually does in topological spaces. One of the key outcomes of this discussion will be the observation that there is essentially no difference between simplicial sets and topological spaces as far as homotopy theory is concerned. Of course, the translation between these two categories will be performed through the familiar adjunction between geometric realisation and the singular set functor. However, our goal will be to develop a homotopy theory for simplicial sets independently of topological spaces and only make the comparison at the very end.

    To get things off the ground, let us attempt to find simplicial analogues of the key notions that allow us to do homotopy theory in topological spaces: these are fibrations, cofibrations and weak equivalences.

    Let us first find an appropriate notion of fibration. The following terminology is not only useful for this purpose, but will reappear throughout the entire lecture course.

    \subsection{Definition}
        Let $\Cc$ be a category, let $f \colon X \to Y$ be a morphism in $\Cc$ and let $\Sigma$ be a class of morphisms in $\Cc$.
        \begin{enumerate}[label=\roman*)]
            \item 
                The map $f$ has the \emph{right lifting property} (or RLP for short), with respect to $\Sigma$ if for every commutative square
                \begin{equation*}
                    \begin{tikzcd}
                        S \arrow[r, "x"] \arrow[d, "s"] & X \arrow[d, "f"] \\
                        T \arrow[r, "y"] & Y
                    \end{tikzcd}
                \end{equation*}
                with $s \in \Sigma$ there exists a commutative diagram of the following form:
                \begin{equation*}
                    \begin{tikzcd}
                        S \arrow[r, "x"] \arrow[d, "s"] & X \arrow[d, "f"] \\
                        T \arrow[r, "y"] \arrow[ur, "\overline{y}"]& Y
                    \end{tikzcd}
                \end{equation*}
            \item 
                The map $f$ has the \emph{left lifting property} (or LLP) with respect to $\Sigma$ if for every commutative square 
                \begin{equation*}
                    \begin{tikzcd}
                        X \arrow[r, "x"] \arrow[d, "f"] & S \arrow[d, "s"] \\
                        Y \arrow[r, "y"] & T
                    \end{tikzcd}
                \end{equation*}
                with $s \in \Sigma$ there exists a commutative diagram of the following form:
                \begin{equation*}
                    \begin{tikzcd}
                        X \arrow[r, "x"] \arrow[d, "f"] & S \arrow[d, "s"] \\
                        Y \arrow[r, "y"] \arrow[ur, "\overline{y}"]& T
                    \end{tikzcd}
                \end{equation*}
        \end{enumerate}

    \subsection{Notation}
        Let $\Cc$ be a category with finite products and pushouts. For $i \colon A \to B$ and $f \colon X \to Y$ morphisms in $\Cc$, let
        \begin{equation*}
            i \boxtimes f \,\colon (A \times Y) \cup_{A \times X} (B \times X) \to B \times Y
        \end{equation*}
        be the canonical arrow induced by the universal property of the pushout.
        \smallskip

        Recall that a map of topological spaces $p \colon E \to B$ is a (Serre) fibration if it has the RLP for the set of inclusions
        \begin{equation*}
            \left\{ (S^{n-1} \hookrightarrow D^n) \boxtimes ({0} \hookrightarrow [0, 1]) \mid n \geq 1 \right\} \,.
        \end{equation*}
        There is an evident analogue of these maps in simplicial sets, namely
        \begin{equation*}
            (\partial \Delta^n \hookrightarrow \Delta^n) \boxtimes ({0} \hookrightarrow \Delta^1) \,.
        \end{equation*}
        However, note that there is a homeomorphism
        \begin{equation*}
            (S^{n-1} \hookrightarrow D^n) \boxtimes ({0} \hookrightarrow [0, 1]) \cong (S^{n-1} \hookrightarrow D^n) \boxtimes ({1} \hookrightarrow [0, 1])
        \end{equation*}
        which has no simplicial analogue. This is the reason why it will also be important to consider the maps
        \begin{equation*}
            (\partial \Delta^n \hookrightarrow \Delta^n) \boxtimes ({1} \hookrightarrow \Delta^1) \,.
        \end{equation*}

    \subsection{Definition}
        \begin{enumerate}[label=\roman*)]
            \item 
                A map of simplicial sets is a \emph{left fibration} if it has the RLP with respect to the set
                \begin{equation*}
                    LC \coloneq \left\{ (\partial \Delta^n \hookrightarrow \Delta^n) \boxtimes ({0} \hookrightarrow \Delta^1) \mid n \geq 1 \right\} \,.
                \end{equation*}
            \item 
                A map of simplicial sets is a \emph{right fibration} if it has the RLP with respect to the set
                \begin{equation*}
                    RC \coloneq \left\{ (\partial \Delta^n \hookrightarrow \Delta^n) \boxtimes ({1} \hookrightarrow \Delta^1) \mid n \geq 1 \right\} \,.
                \end{equation*}
            \item 
                A map of simplicial sets is a Kan fibration if it is both a left and a right fibration, i.e. if it has the RLP with respect to the set
                \begin{equation*}
                    C \coloneq LC \cup RC \,.
                \end{equation*}
        \end{enumerate}    

    \subsection{Remark}
        With this definition, it is not clear that a simplicial set $X$ is Kan if and only if $X \to \Delta^0$ is a Kan fibration. This will be shown in \ref{chV-2-15}.
        \smallskip

        Similarly, cofibrations in topological spaces were defined in terms of the homotopy extension property. We could try to mimic this definition directly but this turns out to be a bad idea. For example, we would certainly expect the inclusion $(\partial \Delta^n \hookrightarrow \Delta^n) \boxtimes (\{0\} \hookrightarrow \Delta^1)$ to be a cofibration. However, if we took ``cofibration'' to mean ``has the homotopy extension property with respect to all simplicial sets'', then this map would have a left inverse. It is easy to check in the case $n = 1$ that no such retraction exists (no does it for any other $n$).

        On the other hand, we can think of every (left) homotopy extension problem as a commutative diagram
        \begin{equation*}
            \begin{tikzcd}
                A \times \Delta^1 \cup_{A \times \{0\}} B \times \{0\} \arrow[r] \arrow[d] & X \arrow[d] \\
                B \times \Delta^1 \arrow[ur, dashed] \arrow[r] & \Delta^0
            \end{tikzcd}
        \end{equation*}
        In particular, for $A = \partial \Delta^n$ and $B = \Delta^n$ we have already \emph{defined} this problem to have a solution if $X \to \Delta^0$ is a left fibration.

    \subsection{Definition}
        A morphism $i \colon A \to B$ is a \emph{left cofibration} if $i \boxtimes (\{0\} \hookrightarrow \Delta^1)$ has the LLP with respect to all Kan fibrations $X \to \Delta^0$.

    \subsection{Remark}\label{chV-1-6}
        \begin{enumerate}[label=\roman*)]
            \item 
                It would actually be more natural to require $X \to \Delta^0$ to be a left Kan fibration in the definition of a left cofibration. We will see later that every left fibration over $\Delta^0$ is a Kan fibration.
            \item 
                One could equally well consider \emph{right cofibrations} and \emph{fibrations}, and develop a parallel theory. We will show soon that this is not a parallel theory, but the \emph{same} theory.
        \end{enumerate}

    \subsection{Examples}
        \begin{enumerate}[label=\roman*)]
            \item 
                By definition, the inclusion $\partial \Delta^n \to \Delta^n$ is a left cofibration for all $n \geq 0$ if we set $\partial \Delta^0 \coloneq \varnothing$.
            \item 
                The map $\varnothing \to B$ is a left cofibration for every simplicial set $B$.
        \end{enumerate}
        Finally, recall that the following statements are equivalent for a map $f \colon X \to Y$ of CW-complexes:
        \begin{enumerate}[label=\roman*)]
            \item 
                $f$ is a homotopy equivalence;\label{chV-1-7-1}
            \item 
                $f$ induces a bijection on $\pi_0$ and an isomorphism on all homotopy groups with respect to every choice of base point;\label{chV-1-7-2}
            \item 
                for every space $Z$, the map $f$ induces a weak homotopy equivalnce
                \begin{equation*}
                    f^{*} \,\colon \Hom_{\Top}(Y, Z) \to \Hom_{\Top}(X, Z) \,;
                    \end{equation*}\label{chV-1-7-3}
            \item 
                for every space $Z$, the map $f$ induces a bijection
                \begin{equation*}
                    f^* \,\colon \pi_0 \Hom_{\Top}(Y, Z) \to \pi_0 \Hom_{\Top}(X, Z) \,.
                \end{equation*}\label{chV-1-7-4}
        \end{enumerate}
        When defining cofibrations, we already observed that homotopies in the category of simplicial sets are more rigid than homotopies in topological spaces, so it is unlikely that copying condition \ref{chV-1-7-1} will give a sensible notion. For example, our geometric intuition tells us that $(\partial \Delta^n \hookrightarrow \Delta^n) \boxtimes ({0} \hookrightarrow \Delta^1)$ should be an equivalence, but it is certainly not a homotopy equivalence. Conditions \ref{chV-1-7-2} and \ref{chV-1-7-3} would require us to first develop a sensible notion of homotopy groups for simplicial sets. So we will try to adopt \ref{chV-1-7-4} as our definition. Note that $\pi_0\Map(Y, Z)$ will correspond to homotopy classes of maps $Y \to Z$. If we were to require this condition for every simplicial set $Z$, we would run into the same problems we faced when defining cofibrations. Hence, we only require condition \ref{chV-1-7-4} to hold when $Z$ is a Kan complex (see again \ref{chV-1-6}).

    \subsection{Definition}
        A map of simplicial sets $f \colon X \to Y$ is a \emph{weak equivalence} if it induces a bijection
        \begin{equation*}
            f^{*} \,\colon \pi_0F(Y, K) \to \pi_0F(X,K)
        \end{equation*}
        for every left Kan complex $K$.

    \subsection{Definition}
        A morphism $f \colon X \to Y$ is a  \emph{retract} of a morphism $f' \colon X' \to Y'$ if it is a retract in $\Fun([1],\Cc)$, i.e. there exists a commutative diagram
        \begin{equation*}
            \begin{tikzcd}
                X \arrow[r, "i"] \arrow[d, "f"] & X' \arrow[r, "p"] \arrow[d, "f'"] & X \arrow[d, "f"] \\
                Y \arrow[r, "j"] & Y' \arrow[r, "q"] & Y
            \end{tikzcd}
        \end{equation*}
        such that $pi=\id_X$ and $qj = \id_Y$.

    \subsection{Definition}
        Let $\Cc$ be a category and let $\Sigma$ be a class of morphisms in $\Cc$.
        \begin{enumerate}[label=\roman*)]
            \item 
                The class $\Sigma$ satisfies the \emph{two-out-of-six property} if the following holds: Given three composable morphisms
                \begin{equation*}
                    X_0 \xrightarrow{f_1} X_1 \xrightarrow{f_2} X_2 \xrightarrow{f_3} X_3
                \end{equation*}
                such that $f_2f_1$ and $f_3f_2$ belong to $\Sigma$, then $f_1$, $f_2$, $f_3$ and $f_3f_2f_1$ also belong to $\Sigma$.
            \item 
                The class $\Sigma$ is \emph{closed under retracts}, i.e. every morphism which is a retract of a member of $\Sigma$ also belongs to $\Sigma$.
        \end{enumerate}

    \subsection{Proposition}
        The class of weak equivalences satisfies the two-out-of-six property and is closed under retracts.\hfill$\Box$
        \smallskip

        \emph{Proof}. This follows directly since bijections satisfy two-out-of-six and are closed under retracts.

\section{Fibrations via Horn-Filling}
    As already promised, there is an easier criterion to check whether a map of simplicial sets is a left or right fibration:

    \subsection{Theorem}\label{chV-2-1}
        Let $f \colon X \to Y$ be a map of simplicial sets.
        \begin{enumerate}[label=\roman*)]
            \item 
                $f$ is a left fibration if and only if it has the RLP with respect to the set of left horn inclusions
                \begin{equation*}
                    LH \coloneq \left\{ \horn{n}{i} \to \Delta^n \mid n \geq 1, 0 \leq i < n \right\} \,. 
                \end{equation*}
            \item 
                $f$ is a right fibration if and only if it has the RLP with respect to the set of right horn inclusions
                \begin{equation*}
                    RH \coloneq \left\{ \horn{n}{i} \to \Delta^n \mid n \geq 1, 0 < i \leq n \right\} \,.
                \end{equation*}
            \item 
                $f$ is a Kan fibration if and only if it has the RLP with respect to the set of horn inclusions $H \coloneq LH \cup RH$.\footnote{These are the standard definitions one will find in the literature.}            
            \end{enumerate}
            The proof of the theorem relies on the observation that we may enlarge the class of morphisms against which we test the RLP without changing the class of morphisms which possess the RLP.

    \subsection{Lemma}\label{chV-2-2}
        Let $\Cc$ be a category with countable colimits and let $f \colon X \to Y$ be a morphism in $\Cc$.
        \begin{enumerate}[label=\roman*)]
            \item 
                Suppose that $f$ has the RLP with respect to $s \colon S \to T$ and let $g \colon S \to Z$ be arbitrary. Then $f$ has the RLP with respect to the induced map $Z \to Z \cup_S T$.
            \item 
                Suppose that $f$ has the RLP with respect to $s'$ and that $s$ is a retract of $s'$. Then $f$ has the RLP with respect to $s$.
            \item 
                If $\left\{ s_i \colon S_i \to T_i\right\}_{i \in I}$ is an arbitrary family of morphisms and $f$ has the RLP with respect to every $s_i$, then $f$ has the RLP with respect to the coproduct $\coprod_{i \in I}s_i \colon \coprod_{i \in I} S_i \to \coprod_{i \in I} T_i$.
            \item 
                Let $S_0 \xrightarrow{s_1} S_1 \xrightarrow{s_2} \ldots$ be a sequence of composable arrows. If $f$ has the RLP with respect to every $s_n$, then $f$ has the RLP with respect to the canonical map $S_0 \to \colim_n S_n$.
        \end{enumerate}
        \smallskip

        \emph{Proof}. Draw the appropriate diagrams, see Marcus Land's ``Introduction to infinity-categories'', Lemma 3.8.\hfill$\Box$

    \subsection{Remark}\label{chV-2-3}
        Note that $f$ has the RLP with respect to $s$ if and only if $s$ has the LLP with respect to $f$. Lemma \ref{chV-2-2} can therefore also be read as a statement about preservation of the LLP.

    \subsection{Definition}
        Let $\Cc$ be a category and let $\Sigma$ be a class of morphisms in $\Cc$. Then $\Sigma$ is \emph{saturated} if it is closed under the four constructions appearing in \ref{chV-2-2}:
        \begin{enumerate}[label=\roman*)]
            \item 
                $S$ is \emph{closed under pushouts}: If $s \colon S \to T$ is in $\Sigma$ and $g \colon S \to Z$ is an arbitrary morphism, then the induced morphism $Z \to Z \cup_S T$ is also in $\Sigma$.
            \item 
                $S$ is \emph{closed under retracts}: If $s'$ is a retract of a morphism $s \in \Sigma$, then $s' \in \Sigma$.
            \item 
                $S$ is \emph{closed under coproducts}: If $\left\{ s_i \colon S_i \to T_i\right\}_{i \in I}$ is a family of morphisms in $\Sigma$, then $\coprod_{i \in I}s_i \colon \coprod_{i \in I} S_i \to \coprod_{i \in I} T_i$ is in $\Sigma$.
            \item 
                $S$ is \emph{closed under countable compositions}: If $S_0 \xrightarrow{s_1} S_1 \xrightarrow{s_2} \ldots$ is a sequence of composable arrows in $S$, then $S_0 \to \colim_n S_n$ is in $\Sigma$.
        \end{enumerate}

        Every class of morphisms $\Sigma$ in a category $\Cc$ is contained in a smallest saturated class of morphisms $\sat(\Sigma)$, which we call the \emph{saturation} of $\Sigma$. Since any intersection of saturated classes is saturated,
        \begin{equation*}
            \sat(\Sigma) = \bigcap_{\substack{S \subseteq \mor(\Cc) \text{ saturated } \\ \Sigma \subseteq S}} S \,.
        \end{equation*}
        Note that the indexing set of the intersection is non-empty since $\mor(\Cc)$ certainly is saturated.

        With this terminology, lemma \ref{chV-2-2} can be rephrased as saying that a morphism has the RLP with respect to a class $\Sigma$ if and only if it has the RLP with respect to $\sat(\Sigma)$.

    \subsection{Remark}
        Note that the saturation of a class of injective maps in $\sSet$ contains only injections.
        \medskip

        There are numerous classes of morphisms we have considered already and whose saturation we might be interested in.

    \subsection{Definition}
        \begin{enumerate}[label=\roman*)]
            \item 
                The class $B$ of \emph{boundary inclusions} is given by
                \begin{equation*}
                    B \coloneq \left\{ \partial \Delta^n \hookrightarrow \Delta^n \mid n \geq 1 \right\} \,.
                \end{equation*}
            \item 
                The class $I$ of injections is given by
                \begin{equation*}
                    I \coloneq \left\{ i \colon X \to Y \mid i \text{ injective} \right\} \,.
                \end{equation*}
        \end{enumerate}
        Observe that $I$ is a saturated class. As a little warm-up, we consider the classes $B$ and $I$.

    \subsection{Proposition}\label{chV-2-7}
        The following classes of morphisms coincide:
        \begin{enumerate}[label=\roman*)]
            \item 
                $\sat(B)$;
            \item 
                $I$.
        \end{enumerate}
        \smallskip

        \emph{Proof}. Exercise.\hfill$\Box$

    \subsection{Corollary}\label{chV-2-8}
        If a map of simplicial sets has the RLP with respect to $B$, then it is a Kan fibration and a homotopy equivalence.
        \smallskip

        \emph{Proof}. Let $f \colon X \to Y$ have the RLP with respect to $B$. By \ref{chV-2-7}, it has the RLP with respect to all inclusions, so it is a Kan fibration. Since $\varnothing \to Y$ is an injection, there exists a section $s \colon Y \to X$. Since $fsf = f$, we have a commutative diagram
        \begin{equation*}
            \begin{tikzcd}[column sep=large]
                X \times \{0, 1\} \arrow[r, "sf \,\sqcup\, \id_X"] \arrow[d, hook, "(\varnothing \to X) \,\boxtimes\, (\{0{,}1\} \to \Delta^1)"'] & X \arrow[d, "f"] \\
                X \times \Delta^1 \arrow[r] & Y
            \end{tikzcd}
        \end{equation*}
        where the map $X \times \Delta^1 \to Y$ is the constant homotopy of $f$. Since $f$ has the RLP with respect to all injections, there exists a homotopy $X \times \Delta^1 \to X$ witnessing $sf \simeq \id_X$.\hfill$\Box$

        Obviously, $LC \subseteq C$, $RC \subseteq C$, $LH \subseteq H$ and $RH \subseteq H$. The analogous inclusions hold for the saturations of these classes.\hfill$\Box$

    \subsection{Proposition}\label{chV-2-9}
        Let $\widetilde{LC}$ denote the class of morphisms
        \begin{equation*}
            \left\{ (X \to Y) \boxtimes (\{0\} \hookrightarrow \Delta^1) \mid X \to Y \text{ injective} \right\}\,.
        \end{equation*}
        Then the following holds:
        \begin{enumerate}[label=\roman*)]
            \item 
                $\widetilde{LC} \subseteq \sat(LC)$;\label{chV-2-9-1}
            \item 
                $LC \subseteq \sat(LH)$;\label{chV-2-9-2}
            \item 
                $LH \subseteq \sat(LC)$;\label{chV-2-9-3}
        \end{enumerate}

        \emph{Proof}. For \ref{chV-2-9-1}, we use again that for every injection $j \colon X \to Y$ there is a filtration
        \begin{equation*}
            X = Y^{(-1)} \subseteq Y^{(0)} \subseteq Y^{(1)} \subseteq Y^{(2)} \subseteq \ldots \subseteq Y
            \end{equation*}
         such that
        \begin{equation*}
            Y^{(n)} \cong Y^{(n-1)} \underset{\coprod_{i \in I_n} \partial \Delta^n}{\cup} \coprod_{i \in I_n} \Delta^n \,,
        \end{equation*}
        $\colim_n Y^{(n)} \cong Y$ and $j$ becomes identified with the canonical map $X \to \colim_n Y^{(n)}$. Since pushouts commute with each other and $- \times \Delta^1$ commutes with pushouts, there exists for every $n \geq 0$ a pushout
        \begin{equation*}
            \begin{tikzcd}
                \coprod\limits_{i \in I_n} ((\Delta^n \times \{0\}) \underset{\partial \Delta^n \times \{0\}}{\cup} (\partial \Delta^n \times \Delta^1)) \arrow[r] \arrow[d] & (Y \times \{0\}) \underset{Y^{(n-1)} \times \{0\}}{\cup} (Y^{(n-1)} \times \Delta^1) \arrow[d] \\
                \coprod\limits_{i \in I_n} (\Delta^n \times \Delta^1) \arrow[r] & (Y \times \{0\}) \underset{Y^{(n)} \times \{0\}}{\cup} (Y^{(n)} \times \Delta^1)
            \end{tikzcd}
        \end{equation*}
        Consequently, the sequence of maps
        \begin{equation*}
            \ldots \rightarrow (Y \times \{0\}) \underset{Y^{(n-1)} \times \{0\}}{\cup} (Y^{(n-1)} \times \Delta^1) \rightarrow (Y \times \{0\}) \underset{Y^{(n)} \times \{0\}}{\cup} (Y^{(n)} \times \Delta^1) \rightarrow \ldots
        \end{equation*}
        lies in $\sat(LC)$. It follows that
        \begin{equation*}
            (Y \times \{0\}) \underset{X \times \{0\}}{\cup} (X \times \Delta^1) \rightarrow \colim\limits_{n} ((Y \times \{0\}) \underset{Y^{(n)} \times \{0\}}{\cup} (Y^{(n)} \times \delta^1)) \cong Y \times \Delta^1
        \end{equation*}
        also lies in $\sat(LC)$.\\
        For \ref{chV-2-9-2}, it suffices to show that $(\partial \Delta^n \to \Delta^n) \boxtimes (\{0\} \to \Delta^1)$ can be obtained by filling left horns in the domain. Note that $\Delta^n \times \Delta^1 \cong N([n] \times [1])$, so simplices correspond to order-preserving maps $\alpha \colon [k] \to [n] \times [1]$. Such a simplex is non-degenerate if and only if $\alpha$ is injective.\\
        For $k < n$, $\alpha$ factors as $[k] \to [n-1] \times [1] \xrightarrow{d_i \times \id} [n] \times [1]$ for some $i$. So every $k$-simplex is contained in $\partial \Delta^n \times \Delta^1$ for $k < n$.\\
        Let $p \colon [n] \times [1] \to [n]$ and $q \colon [n] \times [1] \to [1]$ be the two projection maps. A $n$-simplex $\alpha$ is not contained in $(\Delta^n \times \{0\}) \cup_{\partial \Delta^n \times \{0\}} (\partial \Delta^n \times \Delta^1)$ if and only if $p \alpha = \id_{[n]}$ and $\im(q \alpha) \neq \{0\}$. Moreover, $(\Delta^n \times \{0\}) \cup_{\partial \Delta^n \times \{0\}} (\partial \Delta^n \times \Delta^1)$ does not contain any non-degenerate $n+1$-simplices.\\
        A $(n+1)$-simplex $\alpha$ is uniquely determined by the number $j(\alpha) \in [n]$ satisfying $q \alpha(j(\alpha)) = 0$ and $q \alpha(j(\alpha) + 1) = 1$. Hence, there are precisely $n + 1$ non-degenerate $(n + 1)$-simplices. These come with a canonical ordering $\alpha_0, \ldots, \alpha_n$ by requiring that $j(\alpha_i) = i$.\\
        Consider $\alpha_n$. Then there is exactly one face which does not lie in $(\Delta^n \times \{0\}) \cup_{\partial \Delta^n \times \{0\}} (\partial \Delta^n \times \Delta^1)$, namely $d_n \alpha_n$. The remaining faces of $\alpha_n$ define a map
        \begin{equation*}
            \horn{n+1}{n} \to (\Delta^n \times \{0\}) \cup_{\partial \Delta^n \times \{0\}} (\partial \Delta^n \times \Delta^1) \,,
        \end{equation*}
        and filling this inner horn adds precisely $\alpha_n$ and $d_n \alpha_n$ to give a simplicial subset $X_n \subseteq \Delta^n \times \Delta^1$.\\
        For $\alpha_{n-1}$, observe that $d_{n+1} \alpha_{n-1}$ lies in $\partial \Delta^n \times \Delta^1$, $d_n \alpha_{n-1} = d_n \alpha_n$ and $d_k \alpha_{n-1}$ also lies in $\partial \Delta^n \times \Delta^1$ for $k < n - 1$. So all faces of $\alpha_{n-1}$ except $d_{n-1} \alpha_{n-1}$ define a map $\horn{n+1}{n-1} \to X_n$. Filling this horn adds $\alpha_{n-1}$ and $d_{n-1} \alpha_{n-1}$ to $X_n$ to obtain $X_{n-1}$.\\
        Proceeding like this inductively, one ends up with $\alpha_0$. All faces of $\alpha_0$ except $d_0 \alpha_0$ have been added in previous steps. Filling the corresponding (left!) horn $\horn{n+1}{0} \to X_1$ adds the last two missing simplices $\alpha_0$ and $d_0 \alpha_0$ to $X_1$ to give $X_0 = \Delta^n \times \Delta^1$. There are closed formulas for all the face relations which are not difficult to figure out. If you want to look them up, see Goerss and Jardine's ``Simplicial homotopy theory'', proposition 4.2. This shows \ref{chV-2-9-2}.\\
        Finally, consider the inclusion of a left horn $\horn{n}{k} \to \Delta^n$, i.e. $k < n$. Then there is a retract diagram
        \begin{equation*}
            \begin{tikzcd}
                \horn{n}{k} \arrow[r, "{j|}_{\horn{n}{k}}"] \arrow[d] & (\Delta^n \times \{0\}) \cup_{\horn{n}{k} \times \{0\}} (\horn{n}{k} \times \Delta^1) \arrow[r, "{r_k|}_{\ldots}"] \arrow[d] & \horn{n}{k} \arrow[d] \\
                \Delta^n \arrow[r, "j"] & \Delta^n \times \Delta^1 \arrow[r, "r_k"] & \Delta^n
            \end{tikzcd}
        \end{equation*}
        in which $j$ is given by the inclusion $\Delta^n \cong \Delta^n \times \{1\} \to \Delta^n \times \Delta^1$ and $r_k$ is induced by the map of posets
        \begin{equation*}
            r_k \,\colon [n] \times [1] \to [n], \quad (l, i) \mapsto
            \begin{cases*}
                l &$i = 1$ or $l < k$, \\
                k & else
            \end{cases*} \,.
        \end{equation*}
        Note that $r_k j = \id$, and $r_k$ does not restrict as indicated in the diagram: we only need to check that the inclusion $[n] \setminus \{k\} \subseteq \im(r_k \alpha)$ is impossible. If $\alpha \colon [p] \to [n] \times \{0\}$ is a simplex, then $r_k \alpha$is a map $[p] \to [k]$. Since $k < n$, $\im(r_k \alpha)$ cannot contain $n$. Since $[n] \setminus \{k\}$ is not a subset of $\im(q \alpha)$, it cannot be a subset of $\im(r_k \alpha)$ either. This proves \ref{chV-2-9-3}.\hfill$\Box$

    \subsection{Theorem}\label{chV-2-20}
        A map of simplicial sets is a left cofibration if and only if it is injective.\footnote{As in the case of fibrations, this is usually the definition of a cofibration.}
        \smallskip

        \emph{Proof}. Exercise: Fix an injection $j$. Let $\Sigma$ be a saturated class of morphisms. Then the class of morphisms
        \begin{equation*}
            S \coloneq \left\{ i \mid i \boxtimes j \in \Sigma \right\}
        \end{equation*}
        is saturated.\\
        Consequently, the class of left cofibrations is saturated. Since $\partial \Delta^n \to \Delta^n$ is a left cofibration for all $n \geq 0$, \ref{chV-2-10} implies that every injection is a left cofibration.\\
        Let $i \colon A \to B$ be a cofibration. By \ref{chV-2-19}, there is an anodyne extension
        \begin{equation*}
            B \times \{0\} \cup_{A \times \{0\}} A \times \Delta^1 \to K
        \end{equation*}
        for some Kan complex $K$. Since $i$ is a left cofibration, this injection extends to a map $B \times \Delta^1 \to K$. Consequently, $B \times \{0\} \cup_{A \times \{0\}} A \times \Delta^1 \to B \times \Delta^1$ is injective. Since $i$ can be recovered from this map via the inclusion of $A \times \{1\}$, it follows that $i$ is injective.\hfill$\Box$

    \subsection{Corollary}\label{chV-2-23}

        \begin{equation*}
            F(B, X) \to F(A, X) \underset{F(A, Y)}{\times} F(B, Y)
        \end{equation*}
        is a left/right/Kan fibration. In particular, the following holds:
        \begin{enumerate}[label=\roman*)]
            \item 
                If $B$ is any simplicial set and $p \colon X \to Y$ is a left/right/Kan fibration, then
                \begin{equation*}
                    p^{*} \,\colon F(B, X) \to F(B, Y)
                \end{equation*}
                is a left/right/Kan fibration.\label{chV-2-23-i}
            \item 
                If $i \colon A \to B$ is a cofibration and $X$ is a Kan complex, then
                \begin{equation*}
                    i^{*} \,\colon F(B, X) \to F(A, X)
                \end{equation*}
                is a Kan fibration of Kan complexes.
        \end{enumerate}

        \emph{Proof}. Every lifting problem
        \begin{equation*}
            \begin{tikzcd}
                \horn{n}{k} \arrow[r] \arrow[d] & F(B, X) \arrow[d] \\
                \Delta^n \arrow[r] & F(A, X) \underset{F(A, Y)}{\times} F(B, Y)
            \end{tikzcd}
        \end{equation*}
        corresponds to a commutative diagram
        \begin{equation*}
            \begin{tikzcd}
                (B \times \horn{n}{k}) \underset{A \times \horn{n}{k}}{\cup} (A \times \Delta^n) \arrow[r] \arrow[d] & X \arrow[d, "p"] \\
                B \times \Delta^n \arrow[r] & Y
            \end{tikzcd}
        \end{equation*}
        The left vertical map is an anodyne extension by \ref{chV-2-22}, and hence has the LLP with respect to $p$. The resulting lift provides the required lift of the original lifting problem.\hfill$\Box$

    \subsection{Definition}\label{chV-2-28}
        \begin{enumerate}[label=\roman*)]
            \item 
                A category $\Ii$ is \emph{filtered} if for every diagram $F \colon \Jj \to \Ii$ where $\Jj$ has only finitely many objects and morphisms there exists an object $i \in \Ii$ and a natural transformation $F \Rightarrow \const_i$.
            \item 
                A \emph{filtered colimit} is a colimit of a diagram indexed by a filtered category.
            \item 
                Let $\Cc$ be a category with all filtered colimits. Then $c \in \Cc$ is \emph{compact} if $\Hom_{\Cc}(c, -)$ commutes with all filtered colimits.
            \item 
                Let $\Cc$ be a cocomplete category. Then $c \in \Cc$ is \emph{tiny} if $\Hom_{\Cc}(c, -)$ commutes with all colimits. 
        \end{enumerate}

    \subsection{Examples}\label{chV-2-29}
        \begin{enumerate}[label=\roman*)]
            \item 
                Every directed poset is a filtered category, e.g. $\mathbb{N}$.
            \item 
                A set $X$ is compact in $\Set$ if and only if $X$ is finite.
            \item 
                A simplicial set $X$ is compact in $\sSet$ if and only if $X$ is finite.
            \item 
                A module $M$ over some ring $R$ is compact in $\Mod{R}$ if and only if $M$ is finitely generated.
        \end{enumerate}

        The following statement is often referred to as the ``small object argument'':

    \subsection{Proposition}\label{chV-2-30}
        Let $\Cc$ be a cocomplete category. Let $S$ be a set of morphisms such that the domain of each morphism in $S$ is compact. Then every morphism $f \colon X \to Y$ of simplicial sets can be factored as $f = p \circ i$, where $i$ lies in $\sat(S)$ and $p$ has the RLP with respect to $S$.

        \emph{Proof}. For an arbitrary map $g \colon V \to W$, let
        \begin{equation*}
            \mathrm{LP}(S, g) \coloneq \left\{ (a \colon A \to V, \, b \colon B \to W, \, i \colon A \to B) \mid i \in S, \, g \circ a = b \circ i \right\} \,.
        \end{equation*}
        Define $V'$ as the pushout
        \begin{equation*}
            \begin{tikzcd}
                \coprod\limits_{(a,b,i) \in \mathrm{LP}(S, g)}A \arrow[r, "\Sigma a"] \arrow[d, "\coprod i"'] & V \arrow[d, "g"] \\
                \coprod\limits_{(a,b,i) \in \mathrm{LP}(S, g)}B \arrow[r, "\Sigma b"] & V'
            \end{tikzcd}
        \end{equation*}
        Then there is a canonically induced map $V' \to W$, and the map $V \to V'$ lies in the saturation of $S$.\\
        Now define $X^0 \coloneq X$ and $X^{n+1} \coloneq (X^n)'$. Set $E \coloneq \colim_n X^n$. By definition, the induced map $i \colon X \to E$ is in $\sat(S)$. Moreover, the induced map $p \colon E \to Y$ satisfies $p \circ i = f$.\\
        Consider a lifting problem
        \begin{equation*}
            \begin{tikzcd}
                A \arrow[r, "a"] \arrow[d, "j"] & V \arrow[d, "f"] \\
                B \arrow[ur, dashed] \arrow[r, "b"] & W
            \end{tikzcd}
        \end{equation*}
        with $j \in S$. Since $A$ is compact by assumption, $a$ factors as
        \begin{equation*}
            a \,\colon A \to X^n \to \colim\limits_{n}X^n = E
        \end{equation*}
        for some $n \in \mathbb{N}$. By construction, the induced lifting problem
        \begin{equation*}
            \begin{tikzcd}
                A \arrow[r, "a'"] \arrow[d, "j"] & X^{n+1} \arrow[d, "f"] \\
                B \arrow[ur, dashed] \arrow[r, "b"] & Y
            \end{tikzcd}
        \end{equation*}
        has a solution, and this also provides a solution to the original lifting problem.\hfill$\Box$

        Note that this contains \ref{chV-2-19} as a special case (take $S = H$ and $f$ to be the projection map $X \to \Delta^0$).

    \subsection{Corollary}\label{chV-2-31}
        Let $\Cc$ be a cocomplete category and $S$ a set of maps such that the domain of every morphism in $S$ is compact. Then
        \begin{equation*}
            \sat(S) = \left\{ f \colon X \to Y \mid \begin{array}{r} f \text{ has the LLP with respect to the class of morphisms} \\ \text{which have the RLP with respect to } S\end{array} \right\} \eqcolon \overline{S}
        \end{equation*}

        \smallskip

        \emph{Proof}. Since $\overline{S}$ is a saturated class by \ref{chV-2-2} and contains $S$, we have $\sat(S) \subseteq \overline{S}$.\\

        \begin{equation*}
            \begin{tikzcd}
                X \arrow[r, "i"] \arrow[d, "f"] & E \arrow[d, "p"] \\
                Y \arrow[ur, dashed, "s"] \arrow[r, "\id"] & Y
            \end{tikzcd}
        \end{equation*}
        has a solution since $f$ has the LLP with respect to $p$. These maps define a retract diagram

            \begin{tikzcd}
                X \arrow[r, "\id"] \arrow[d, "f"] & X \arrow[r, "\id"] \arrow[d, "i"] & X \arrow[d, "f"] \\
                Y \arrow[r, "s"] & E \arrow[r, "p"] & Y
            \end{tikzcd}
        \end{equation*}
        So $f$ is in $\sat(S)$.\hfill$\Box$

\section{Simplicial homotopy groups}\label{chV-3}
    \subsection{Proposition}\label{chV-3-1}
        Let $K$ be a Kan complex. Then $\pi X$ is a groupoid. In particular, homotopy is an equivalence relation on $\Hom_{\sSet}(\Delta^0, K)$.
        \smallskip

        \emph{Proof}. Let $x, y \in X$ and let $e$ be an edge such that $d_0(e) = y$ and $d_1(e) = x$. Consider $e$ and $s_0(x)$ as a map $\horn{2}{0} \to X$. For any filler $F$, the edge $d_0(F)$ provides a morphism $y \to x$ in $\pi X$ such that $d_0F \circ e = \id_x$. Similarly, $e$ and $s_0(x)$ also define a map $\horn{2}{2} \to X$, and $d_2$ of any filler provides a right inverse to $e$. Since isomorphisms satisfy two-out-of-six, $e$ is an isomorphism in $\pi X$.

    \subsection{Corollary}\label{chV-3-2}
        Let $K$ be a Kan complex, let $X$ be a simplicial set, and let $A$ be a simplicial subset of $X$. Then homotopy relative $A$ is an equivalence relation on $\Hom_{\sSet}(X, K)$.
        \smallskip

        \emph{Proof}. Let $f \colon A \to K$ be given and consider the pullback
        \begin{equation*}
            \begin{tikzcd}
                F(X, K)_f \arrow[d] \arrow[r] & F(X, K) \arrow[d] \\
                \Delta^0 \arrow[r, "f"] & F(A, K)
            \end{tikzcd}
        \end{equation*}
        Then $F(X, K)_f$ is a Kan complex by \ref{chV-2-23}. Homotopy of $0$-simplices in $F(X, K)_f$ corresponds precisely to homotopy relative $A$ of maps $X \to K$ which restrict to the given map $f$ on $A$. So the corollary follows from \ref{chV-3-1}.\hfill$\Box$

    \subsection{Corollary}\label{chV-3-3}
        Every weak equivalence between Kan complexes is a homotopy equivalence.
        \smallskip

        \emph{Proof}. Let $f \colon K \to L$ be a weak equivalence between Kan complexes. Since $f^{*} \colon \pi_0(L, K) \to \pi_0(K, K)$ is a bijection, there exists by \ref{chV-3-2} a map $g \colon L \to K$ such that $gf \simeq \id_K$. Since $fgf \simeq f$, the injectivity of $f^{*}$ implies that $fg \simeq \id_L$ as well.\hfill$\Box$

    \subsection{Definition}\label{chV-3-4}
        Let $A \subseteq X$ and $B \subseteq Y$ be inclusions of simplicial subsets. Define $F((X, A),(Y, B))$ as the pullback
        \begin{equation*}
            \begin{tikzcd}
                F((X, A), (Y, B)) \arrow[d] \arrow[r] & F(X, Y) \arrow[d] \\
                F(A, B) \arrow[r] & F(A, Y)
            \end{tikzcd}
        \end{equation*}
        Note that $F((X, A),(Y, B))$ is a Kan complex if $B$ and $Y$ are Kan complexes (by \ref{chV-2-23} and \ref{chV-2-24}).

    \subsection{Definition}\label{chV-3-5}
        For $n \geq 0$, let
        \begin{equation*}

        \end{equation*}
        be the \emph{n-cube}. The simplicial subset

        is called the \emph{boundary} of the $n$-cube. We agree that $\partial \Box^0 = \varnothing$.

    \subsection{Definition}\label{chV-3-6}
        Let $X$ be a Kan complex and $x \in X_0$. Define
        \begin{equation*}
            \pi_n(X, x) \coloneq \pi_0(F((\Box^n, \partial \Box^n),(X,x)))
        \end{equation*}
        Let $Y$ be a Kan complex and $y \in Y$. Define
        \begin{equation*}
            \pi_1(Y, y) \coloneq \End_{\pi Y}(y)
        \end{equation*}
        Unwinding definitions and using \ref{chV-2-30}, there is a natural isomorphism
        \begin{equation*}
            \pi_n(X, x) \cong \pi_1(F((\Box^{n-1}, \partial \Box^{n-1}),(X, x)), c_x)
        \end{equation*}
        for all $n \geq 1$, where $c_x$ denotes the constant map with value $x$. So the composition law in the homotopy category equips $\pi_x(X, x)$ with the structure of a monoid.\\
        Conjugating the identification
        \begin{equation*}
            \pi_n(X, x) \cong \pi_1(F((\Box^{n-1}, \partial \Box^{n-1}),(X, x)), c_x)
        \end{equation*}
        by the transposition $\tau_{1, i}$ acting on the factors of $(\Delta^1)^n$, we i fact obtain $n$ monoid structures $\{ \cdot_i \}_{i \leq i \leq n}$ on $\pi_n(X, x)$ which have the same neutral element (namely the class represented by the constant map).

    \subsection{Proposition}\label{chV-3-7}
        Let $X$ be a Kan complex and $x \in X$. Then $\pi_n(X, x)$ is a group for all $n \geq 1$. For $n \geq 2$, the compositions $\cdot_i$ all agree and are commutative.
        \smallskip

        \emph{Proof}. Since $F((\Box^{n-1}, \partial \Box^{n-1}),(X, x))$ is a Kan complex for all $n \geq 1$, the first assertion follows from the fact that $\pi Y$ is a groupoid for every Kan complex Y.\\
        Let $\alpha, \beta, \gamma, \delta \colon (\Box^n, \partial \Box^n) \to (X, x)$ represent elements in $\pi_n(X, x)$ for $n \geq 2$. We claim that
        \begin{equation*}
            ([\alpha] \cdot_1 [\beta]) \cdot_2 ([\gamma] \cdot_1 [\delta]) = ([\alpha] \cdot_2 [\gamma]) \cdot_1 ([\beta] \cdot_2 [\delta]) \,.
        \end{equation*}
        If this holds, the Eckmann-Hilton argument shows that $\cdot_1 = \cdot_2$ and that this composition is abelian.\\
        The four given elements induce a map $\horn{2}{1} \times \horn{2}{1} \times \Box^{n-2} \to X$. Decoding the definition of the composition laws, the left hand composition amounts to extending this map along the anodyne extensions
        \begin{equation*}
            \horn{2}{1} \times \horn{2}{1} \times \Box^{n-2} \to \Delta^2 \times \horn{2}{1} \times \Box^{n-2} \to \Delta^2 \times \Delta^2 \times \Box^{n-2} \,,
        \end{equation*}
        while the right hand side corresponds to extending along the anodyne extensions
        \begin{equation*}
            \horn{2}{1} \times \horn{2}{1} \times \Box^{n-2} \to \horn{2}{1} \times \Delta^2 \times \Box^{n-2} \to \Delta^2 \times \Delta^2 \times \Box^{n-2} \,.
        \end{equation*}
        Note that we are ignoring various boundary conditions for legibility. Both extensions provide an extension along the anodyne extension
        \begin{equation*}
            \horn{2}{1} \times \horn{2}{1} \times \Box^{n-2} \to \Delta^2 \times \Delta^2 \times \Box^{n-2} \,.
        \end{equation*}
        Since such an extension is unique up to homotopy (\ref{chV-2-26}), this proves the claim.\\
        For arbitrary $\cdot_i$ and $\cdot_j$, the same proof works after appropriate permutation of the factors.\hfill$\Box$
        \medskip

        Of course, one can construct the group structure on $\pi_n(X, x)$ directly and observe that there are $n$ different compositions on $\pi_n(X, x)$. As soon as there are at least two, the Eckmann-Hilton argument applies.\\
        Each $\pi_n$ defines a functor
        \begin{equation*}
            \pi_n \,\colon \Kan_{*} \to \Grp
        \end{equation*}
        which is invariant under pointed simplicial homotopies.

    \subsection{Theorem}\label{chV-3-8}
        Let $p \colon X \to Y$ be a Kan fibration of Kan complexes and $x \in X_0$. Let $F$ denote the fibre of $p$ over $p(x)$. Then there exists a natural long exact sequence
        \begin{equation*}
            \ldots \rightarrow \pi_{n+1}(X,x) \xrightarrow{p_{*}} \pi_{n+1}(Y, p(x)) \xrightarrow{\partial} \pi_n(F,x) \xrightarrow{i_{*}} \pi_n(X,x) \xrightarrow{p_{*}} \pi_n(Y, p(x)) \rightarrow \ldots 
        \end{equation*}
        In low degrees, exactness means the following:
        \begin{enumerate}[label=\roman*)]
            \item 
                There is an action $\pi_1(Y, p(x)) \times \pi_0(F, x) \to \pi_0(F, x)$ such that
                \begin{itemize}
                    \item 
                        the image of $\pi_{*} \colon \pi_1(X, x) \to \pi_1(Y,y)$ is precisely the stabilizer of $[x] \in \pi_0(F)$;
                    \item 
                        two elements in $\pi_0(F,x)$ map to the same element in $\pi_0(X, x)$ if and only if they lie in the same orbit of this action.
                \end{itemize}
            \item 
                An element in $\pi_0(X, x)$ maps to the basepoint in $\pi_0(Y, p(x))$ if and only if they lie in the image of $\pi_0(F, x) \to \pi_0(X, x)$.
        \end{enumerate}
        \smallskip

        \emph{Proof}. Since we are using cubes to model homotopy groups, the proof is more or less identical to the proof for topological spaces. Set $y \coloneq p(x)$.
        We begin by defining the boundary map. Let $\alpha \colon (\Box^{n+1}, \partial \Box^{n+1}) \to (Y, y)$ represent any element in $\pi_{n+1}(Y, y)$. Since $(\partial \Box^n \to \Box^n) \boxtimes (\{0\} \to \Delta^1)$ is anodyne, the lifting problem
        \begin{equation*}
            \begin{tikzcd}
                (\Delta^1 \times \partial \Box^n) \underset{\{0\} \times \partial \Box^n}{\cup} (\{0\} \times \Box^n) \arrow[d] \arrow[r, "\const_x"] & X \arrow[d, "p"] \\
                \Delta^1 \times \Box^n \arrow[ur, dashed, "\overline{\alpha}"] \arrow[r, "\alpha"] & Y 
            \end{tikzcd}
        \end{equation*}
        has a solution. Set
        \begin{equation*}
            \partial [\alpha] \coloneq [{\overline{\alpha}|}_{\{1\} \times \Box^n}] \,.
        \end{equation*}
        This is well-defined: let $h \colon \Box^{n+1} \times \Delta^1 \cong \Delta^1 \times \Box^n \times \Delta^1 \to Y$ be a homotopy relative $\partial \Box^{n+1}$ from $\alpha$ to $\alpha'$. Any two lifts $\overline{\alpha}$ and $\overline{\alpha}'$ as above define a map $(\Delta^1 \times \Box^n \times \partial \Delta^1) \to X$. This map glues with the constant map on
        \begin{equation*}
            ((\Delta^1 \times \partial \Box^n) \underset{\{0\} \times \partial \Box^n}{\cup} (\{0\} \times \Box^n)) \times \Delta^1
        \end{equation*}
        to define a partial lift of $h$. Since $(\{0\} \to \Delta^1) \boxtimes (\partial \Box^n \to \Box^n)$ is anodyne, so is
        \begin{equation*}
            ((\{0\} \to \Delta^1) \boxtimes (\partial \Box^n \to \Box^n)) \boxtimes (\partial \Delta^1 \to \Delta^1) \,.
        \end{equation*}
        Any lift $\overline{h} \colon \Delta^1 \times \Box^n \times \Delta^n \to X$ witnesses that $\alpha$ and $\alpha'$ are homotopic relative $\partial \Box^n$. So $\partial$ is well-defined.
        If $n \geq 1$, we claim that $\partial$ is a group homomorphism. Two elements $[\alpha], [\beta]$ define a map
        \begin{equation*}
            \Delta^1 \times \horn{2}{1} \times \Box^{n-1} \to Y \,.
        \end{equation*}
        Similarly, the lifts $\overline{\alpha}$ and $\overline{\beta}$ glue to a map on $\Delta^1 \times \horn{2}{1} \times \Box^{n-1}$ which has an extension to a map $\gamma \colon \Delta^1 \times \Delta^2 \times \Box^{n-1} \to X$. Then ${\gamma|}_{\{1\} \times d_1 \Delta^2 \times \Box^{n-1}}$ defines $\partial[\alpha] \cdot \partial[\beta]$. On the other hand, $p \circ \gamma$ is an extension of the original map, and ${(p \circ \gamma)|}_{\{1\} \times d_1 \Delta^2 \times \Box^{n-1}}$ defines $[\alpha] \cdot_2 [\beta]$, so
        \begin{equation*}
            \partial ([\alpha] \cdot_2 [\beta]) = \partial [\alpha] \cdot \partial [\beta] \,.
        \end{equation*}
        Since $\cdot_2 = \cdot$, this proves that $\partial$ is a homomorphism.
        Let $e \in F_0$ and $[\alpha] \in \pi_1(Y, y)$. Solve the lifting problem
        \begin{equation*}
            \begin{tikzcd}
                \Delta^0 \arrow[d] \arrow[r, "e"] & X \arrow[d, "p"] \\
                \Delta^1 \arrow[ur, dashed, "\overline{\alpha}"] \arrow[r, "\alpha"] & Y
            \end{tikzcd}
        \end{equation*}
        to define $[\alpha][e] \coloneq [\overline{\alpha}(1)]$. It is easy to check that this is well-defined and defines a group action. We recover the definition of the connecting map by letting $\pi_1(Y, y)$ act on $[x]$.
        We begin by showing exactness in low degrees.
        
        \emph{Exactness at $\pi_0(X)$:} The composition $\pi_0(F) \to \pi_0(X) \to \pi_0(Y)$ obviously hits the basepoint. Suppose that $x' \in X$ such that $[p(x')] = [y] \in \pi_0(Y)$. Then the lifting problem
        \begin{equation*}
            \begin{tikzcd}[column sep=large]
                \Delta^0 \arrow[d] \arrow[r, "x'"] & X \arrow[d, "p"] \\
                \Delta^1 \arrow[ur, dashed, "h"] \arrow[r, "p{(x')} \simeq y"] & Y
            \end{tikzcd}
        \end{equation*}
        has a solution, and $h(1)$ is a point in $F$ such that $[h(1)] = [x'] \in \pi_0(X)$.
        
        \emph{Exactness at $\pi_0(F)$:} Suppose that $[e], [e'] \in pi_0(F)$ satisfy $[\alpha][e] = [e']$ for some $[\alpha] \in \pi_1(Y, y)$. Then, by definition, there exists $\overline{\alpha} \colon \Delta^1 \to X$ such that $\overline{\alpha}(0) = e$ and $\overline{\alpha}(1)$ is homotopic to $[e']$ in $F$. This implies that $e$ and $e'$ are homotopic in $X$.
        Conversely, suppose that $[e] = [e'] \in \pi_0(X)$. Then there exists a homotopy $h \colon \Delta^1 \to X$ from $e$ to $e'$. The map $ph$ defines an element in $\pi_1(Y, y)$ since $p(e) = y = p(e')$. By definition, we have $[ph][e] = [e']$. So the fibres of $\pi_0(F) \to \pi_0(X)$ correspond precisely to the orbits of the $\pi_1(Y, y)$-action on $\pi_0(F)$.
        
        \emph{Exactness at $\pi_1(Y, y)$:} Let $[\alpha] \in \pi_1(X, x)$. Then $\alpha$ itself is a solution to the lifting problem
        \begin{equation*}
            \begin{tikzcd}
                \Delta^0 \arrow[d] \arrow[r, "x"] & X \arrow[d, "p"] \\
                \Delta^1 \arrow[ur, dashed] \arrow[r, "p \alpha"] & Y
            \end{tikzcd}
        \end{equation*}
        so $[p \alpha][x] = [x]$. Conversely, suppose that $[\alpha] \in \pi_1(Y, y)$ satisfies $[\alpha][x] = [x]$. Then $\alpha$ lifts by definition of the action to a loop in $X$. So the image of $p_{*} \colon \pi_1(X, x) \to \pi_1(Y, y)$ is precisely the stabilizer of $[x] \in \pi_0(F)$.
        
        \emph{Exactness at $\pi_n(X, x)$ for $n \geq 1$:} The composition $p_{*} i_{*}$ is obviously trivial. Suppose that $p_{*} [\alpha] = 1 \in \pi_n(Y, y)$. Then there exists a homotopy $h \colon \Box^n \times \Delta^1 \to Y$ relative $\partial \Box^n$ from $p \alpha$ to the constant map with value $y$. Since $p$ is a Kan fibration, the lifting problem
        \begin{equation*}
            \begin{tikzcd}[column sep=large]
                (\Box^n \times \{0\}) \underset{\partial \Box^n \times \{0\}}{\cup} (\partial \Box^n \times \Delta^1) \arrow[d] \arrow[r, "\alpha \, \cup \, \const_x"] & X \arrow[d, "p"] \\
                \Box^n \times \Delta^1 \arrow[r, "h"] & Y
            \end{tikzcd}
        \end{equation*}
        has a solution. The endpoint of this solution provides a preimage of $[\alpha]$ under $i_{*}$.
        
        \emph{Exactness at $\pi_n(F, x)$ for $n \geq 1$:} Let $[\alpha] \in \pi_{n+1}(Y, y)$. Then $\partial [\alpha]$ is by construction represented by a map which is homotopic to $\const_x$ relative $\partial \Box^n$ in $X$. Consequently, $i_{*} \partial$ is trivial.
        Conversely, suppose that $[i \alpha]$ is trivial in $\pi_n(X, x)$ for some $[\alpha] \in \pi_n(F, x)$. Then there exists a homotopy (note we are free to choose the direction of the homotopy) $h \colon \Delta^1 \times \Box^n \to X$ relative $\partial \Box^n$ from $\const_x$ to $i \alpha$. Then, by definition, $h$ represents an element in $\pi_{n+1}(Y, y)$ with $\partial [h] = [\alpha]$.
        
        \emph{Exactness at $\pi_{n+1}(Y, y)$ for $n \geq 1$:} Let $[\alpha] \in \pi_{n+1}(X, x)$. Then $\alpha$ solves the lifting problem used to define $\partial [p \alpha]$. Since $\alpha$ is constant on $\partial \Box^n$, this proves that $\partial p_{*}$ is trivial.
        Conversely, suppose that $\partial [\alpha] = [{\overline{\alpha}|}_{\{1\} \times \Box^n}] = 1$ for some $[\alpha] \in \pi_{n+1}(Y, y)$. Then there exists a homotopy $h \colon \Delta^1 \times \Box^n \to F$ relative $\partial \Box^n$ from ${\overline{\alpha}|}_{\{1\} \times \Box^n}$ to $\const_x$. Since $p$ is a Kan fibration, the lifting problem
        \begin{equation*}
            \begin{tikzcd}[column sep=huge]
                (\horn{2}{1} \times \Box^n) \underset{\horn{2}{1} \times \partial \Box^n}{\cup} (\Delta^2 \times \partial \Box^n) \arrow[d] \arrow[r, "{(\alpha \, \cup \, h)} \, \cup \, \const_x"] & X \arrow[d, "p"] \\
                \Delta^2 \times \Box^n \arrow[r, "\alpha {(s_1 \times \id)}"] & Y
            \end{tikzcd}
        \end{equation*}
        has a solution $\overline{h}$, where $s_1$ denotes the appropriate degeneration map. Then ${\overline{h}|}_{d_1 \Delta^2 \times \Box^n}$ defined an element in $\pi_{n+1}(X, x)$ such that $[{p \overline{h}|}_{d_1 \Delta^2 \times \Box^n}] = [\alpha]$.\hfill$\Box$

    \subsection{Lemma}\label{chV-3-9}
        Let $f \colon (X, x) \to (Y, y)$ be a pointed map of pointed Kan complexes. Then f is a simplicial homotopy equivalence if and only if it is a pointed simplicial homotopy equivalence.
        \smallskip

        \emph{Proof}. The basepoint inclusions induce Kan fibrations
        \begin{equation*}
            \ev_x \,\colon F(X, Z) \to Z, \quad \ev_y \,\colon F(Y, Z) \to Z
        \end{equation*}
        for every Kan complex $Z$. For $z \in Z_0$, the fibres of $\ev_x$ and $\ev_y$ over $z$ are precisely $F((X, x),(Z, z))$ and $F((Y, y),(Z ,z))$, respectively. Consider the induced map on long exact sequences in low degrees:
        \begin{equation*}
            \begin{tikzcd}
                \pi_1(Z, z) \arrow[d, "\id"] \arrow[r, "\partial_Y"] & \pi_0 F((Y, y),(Z, z)) \arrow[d, "f^{*}_p"] \arrow[r] & \pi_0 F(Y, Z) \arrow[d, "\sim"', "f^{*}"] \arrow[r] & \pi_0(Z) \arrow[d, "\id"] \\
                \pi_1(Z, z) \arrow[r, "\partial_X"] & \pi_0 F((X, x),(Z, z)) \arrow[r] & \pi_0 F(X, Z) \arrow[r] & \pi_0(Z) 
            \end{tikzcd}
        \end{equation*}
        By a diagram chase, $f^{*}_p$ is surjective. Inserting $(Z, z) = (X, x)$, there exists a pointed map $g \colon (Y, y) \to (Z, z)$ such that $gf$ is pointed homotopic to $\id_X$. In particular, $g$ is an unpointed homotopy equivalence, Repeating the argument with $g$ in place of $f$, we obtain a pointed map $f' \colon (X, x) \to (Y, y)$ such that $f'g$ is pointed homotopic to $\id_Y$. By two-out-of-six, it follows that $f$ is a pointed homotopy equivalence.\hfill$\Box$

    \subsection{Corollary}\label{chV-3-10}
        Let $f \colon (X, x) \to (Y, y)$ be a map of pointed Kan complexes such that $f \colon X \to Y$ is a homotopy equivalence. Then $f$ induces isomorphisms
        \begin{equation*}
            f_{*} \,\colon \pi_n(X, x) \xrightarrow{\cong} \pi_n(Y, y)
        \end{equation*}
        for all $n$.
        \medskip

        Our next goal is to prove a converse to \ref{chV-3-10}.

    \subsection{Theorem}\label{chV-3-11}
        A map $f \colon X \to Y$ between Kan complexes is a homotopy equivalence if and only if $f$ induces a bijection on $\pi_0$ and an isomorphism $\pi_n(X, x) \to \pi_n(Y, f(x))$ for all $x \in X_0$ and $n \geq 1$.
        \medskip

        We begin by characterizing contractible Kan complexes by their homotopy groups. The proof relies on a description of the homotopy groups in terms of simplices.

    \subsection{Definition}\label{chV-3-12}

            \pi^{\triangle}_n(X, x) \coloneq \pi_0(F((\Delta^n, \partial \Delta^n),(X, x))) \,.

    \subsection{Lemma}\label{chV-3-13}
        Let $A_1 \subseteq A_2 \subseteq X$ be simplicial sets and suppose that $A_1 \subseteq A_2$ is an anodyne extension. Then the restriction map
        \begin{equation*}
            F((X, A_2),(Y, B)) \to F((X, A_1),(Y, B))
        \end{equation*}
        induces a bijection on $\pi_0$.
        \smallskip

        \emph{Proof}. The map under consideration comes from a map of pullback squares
        \begin{equation*}
            \begin{tikzcd}
                & F((X, A_2),(Y, B)) \arrow[ld] \arrow[dd] \arrow[rr] & & F(X, Y) \arrow[ld, "\id"] \arrow[dd] \\
                F((X, A_1),(Y, B)) \arrow[dd] \arrow[rr, crossing over] & & F(X, Y) & \\
                & F(A_2, B) \arrow[ld] \arrow[rr] & & F(A_2, Y) \arrow[ld] \\
                F(A_1, B) \arrow[rr] & & F(A_1, Y) \arrow[from=uu, crossing over] &
            \end{tikzcd}
        \end{equation*}
        The restriction maps $F(A_2, B) \to F(A_1, B)$ and $F(A_2, Y) \to F(A_1, Y)$ have the RLP with respect to $I$ by \ref{chV-2-26}.

        Let $a \colon A_1 \to A_2$, $i_1 \colon A_1 \to X$, $i_2 \colon A_2 \to X$ and $j \colon B \to Y$ denote the various inclusion maps.

        We first prove surjectivity on $\pi_0$. Let $f \colon (X, A_1) \to (Y, B)$ be a vertex in $F((X, A_1),(Y, B))$. Let ${f|}_{A_1} \colon A_1 \to B$ be the induced map. Then ${f|}_{A_1}$ has a lift to map $f_2 \colon A_2 \to B$. By assumption, $j f_2 a = j {f|}_{A_1} = f i_1 \colon A_1 \to Y$. Since $F(A_2, Y) \to F(A_1, Y)$ has the RLP with respect to $I$, there exists a homotopy $f i_2 \simeq j f_2$ relative $A_1$.\\
        Since $f i_2$ extends to $X$ and $F(X, Y) \to F(A_2,Y)$ is a Kan fibration, we obtain a homotopy $X \times \Delta \to Y$ relative $A_1$ from $f$ to a map $f'$ satisfying $f' i_2 = j f_2$. By construction, this homotopy gives a $1$-simplex in $F((X, A_1),(Y, B))$ from $f$ to $f'$, and $f'$ is a map $(X, A_2) \to (Y, B)$.

        For injectivity, let $f_0, f_1 \colon (X, A_2) \to (Y, B)$ be two maps such that there exists a homotopy $h \colon X \times \Delta^1 \to Y$ from $f_0$ to $f_1$ which restricts to a homotopy ${h|}_{A_1} \colon A_1 \times \Delta^1 \to B$ from ${f_0|}_{A_1}$ to ${f_1|}_{A_1}$. Since $F(A_2, B) \to F(A_1, B)$ has the RLP with respect to $I$, the homotopy ${h|}_{A_1}$ extends to a homotopy $k \colon A_2 \times \Delta^1 \to B$ from ${f_0|}_{A_2} \simeq {f_1|}_{A_2}$.\\
        Consider the map
        \begin{equation*}
            A_2 \times (\Delta^1 \times \{0\} \underset{\partial \Delta^1 \times \{0\}}{\cup} \partial \Delta^1) \times \Delta^1 \to B
        \end{equation*}
        which is given by ${h|}_{A_2}$ on $A_2 \times \Delta^1 \times \{0\}$ and by the constant homotopies on $f_e$ on $A_2 \times \{e\} \times \Delta^1$. This map glues with $k$ to give a map
        \begin{equation*}
            A_2 \times (\Delta^1 \times \partial \Delta^1 \underset{\partial \Delta^1 \times \{0\}}{\cup} \partial \Delta^1 \times \Delta^1) \to B \,,
        \end{equation*}
        which extends to a map $A_2 \times \Delta^1 \times \Delta^1$. Since ${h|}_{A_2}$ extends to a homotopy on $X$ and $F(X, Y) \to F(A_2, Y)$ is a Kan fibration, there exists a map
        \begin{equation*}
            \overline{h} \,\colon X \times \Delta^1 \times \Delta^1 \to Y
        \end{equation*}
        such that ${\overline{h}|}_{X \times \Delta^1 \times \{0\}} = h$ and ${\overline{h}|}_{X \times \{e\} \times  \Delta^1}$ is the constant homotopy on $f_e$. The endpoint ${\overline{h}|}_{X \times \Delta^1 \times \{1\}}$ is a homotopy $f_0 \simeq f_1$ which restricts to a homotopy ${f_0|}_{A_2} \simeq {f_1|}_{A_2}$.

    \subsection{Proposition}\label{chV-3-14}
        There is a natural bijection
        \begin{equation*}
            \pi^{\triangle}_n(X, x) \sim \pi_n(X, x) \,.
        \end{equation*}
        \smallskip

        \emph{Proof}. By definition,
        \begin{equation*}
            \pi_n(X, x) = \pi_0(F((\Box^n, \partial \Box^n),(X, x))) \,.
        \end{equation*}
        Let $C^n \subseteq \Box^n$ denote the simplicial subset obtained by removing the simplex
        \begin{equation*}
            \sigma^n \,\colon [n] \to [1]^n, \quad i \mapsto \underbrace{1 \ldots 1}_{i \text{ times}} \; \underbrace{0 \ldots 0}_{n-i \text{ times}} \,.
        \end{equation*}
        Claim: \emph{The inclusion $\partial \Box^n \to C^n$ is anodyne}. For someone who is good at finite combinatorics, this can probably be proven directly by successively filling horns. We will derive it later from results that are independent of the present discussion, but would lead us on a tangent right now.
        Consequently, the restriction map induces by \ref{chV-3-13} a bijection
        \begin{equation*}
            \pi_0(F((\Box^n, C^n),(X, x))) \xrightarrow{\sim} \pi_0(F((\Box^n, \partial \Box^n),(X, x))) \,.
        \end{equation*}
        By inspection, we have for every pair of simplicial sets $A \subseteq B$ a natural isomorphism
        \begin{equation*}
            F((B, A), (X, x)) \cong F((B/A, *), (X, x)) \,.
        \end{equation*}
        Consequently, we have natural bijections
        \begin{align*}
            \pi_0(F((\Box^n, C^n),(X, x))) &\sim \pi_0(F((\Box^n/C^n, *),(X, x))) \\
                                 &\sim \pi_0(F((\Delta^n/\partial \Delta^n, *),(X, x))) \\
                                 &\sim \pi_0(F((\Delta^n, \partial \Delta^n),(X, x))) \,.
        \end{align*}
        \hfill$\Box$

    \subsection{Remark}\label{chV-3-15}
        The bijection $\pi^{\triangle}_n(X, x) \to \pi_n(X, x)$ from \ref{chV-3-14} sends a map $\alpha \colon (\Delta, \partial \Delta^n) \to (X, x)$ to the map $(\Box^n, \partial \Box^n) \to (X, x)$ which is given by $\alpha$ on the simplex $\sigma^n$ appearing in the proof and is constant on all other simplices.

    \subsection{Lemma}\label{chV-3-16}
        Let $X, x$ be a pointed Kan complex such that $\pi_n(X, x)$ is trivial for all $n \geq 0$. Let $f \colon B \to X$ be a map, let $A \subseteq B$ and let $h \colon A \times \Delta^1 \to X$ be a homotopy from ${f|}_A$ to the constant map with value $x$.\\
        Then there exists an extension of $h$ to a homotopy $B \times \Delta^1 \to X$ from $f$ to the constant map with value $x$.
        \smallskip

        \emph{Proof}. The extension may be constructed simplex by simplex, so it suffices to consider $B = \Delta^n$ and $A = \partial \Delta^n$. Since $(\partial \Delta^n \to \Delta^n) \boxtimes (\{0\} \to \Delta^1)$ is anodyne, there exists a homotopy $h' \colon \Delta^n \times \Delta^1 \to X$ such that ${h'|}_{\Delta^n \times \{0\}} = f$ and ${h'|}_{\partial \Delta^n \times \{1\}} = \const_x$. Therefore, $f' \coloneq {h'|}_{\Delta^n \times \{1\}}$ represents an element in $\pi^{\triangle}_n(X, x)$. Since $\pi_x(X, x)$ is trivial, \ref{chV-3-14} guarantees the existence of a homotopy $h''$ relative $\partial \Delta^n$ from $f'$ to the constant map with value $x$.\\
        The two homotopies $h'$ and $h''$ glue with the map
        \begin{equation*}
            \partial \Delta^n \times \Delta^2 \xrightarrow{\partial \Delta^n \times s_1} \partial \Delta^n \times \Delta^1 \xrightarrow{h} X
        \end{equation*}
        to give a map
        \begin{equation*}
            \widetilde{h} \,\colon (\Delta^n \times \horn{2}{1}) \underset{\partial \Delta^n \times \horn{2}{1}}{\cup} (\partial \Delta^n \times \Delta^2) \to X \,.
        \end{equation*}
        Since $(\partial \Delta^n \to \Delta^n) \boxtimes (\horn{2}{1} \to \Delta^2)$ is anodyne, there exists an extension of $\widetilde{h}$ to $\Delta^n \times \Delta^2$. restricting to $\Delta^n \times d_1 \Delta^2$ yields the desired extension of $h$.\hfill$\Box$

    \subsection{Corollary}\label{chV-3-17}
        Let $X$ be a Kan complex. The following are equivalent:
        \begin{enumerate}[label=\roman*)]
            \item 
                The map $X \to \Delta^0$ has the RLP with respect to $I$.\label{chV-3-17-1}
            \item 
                The map $X \to \Delta^0$ is a homotopy equivalence.\label{chV-3-17-2}
            \item 
                $X$ is non-empty and $\pi_n(X, x)$ is trivial for all $x \in X_0$ and for every $n \geq 0$.\label{chV-3-17-3}
            \item 
                $\pi_0(X)$ is a singleton and there exists some $x \in X_0$ such that $\pi_n(X, x)$ is trivial for all $n \geq 1$.\label{chV-3-17-4}
        \end{enumerate}
        \smallskip

        \emph{Proof}. \ref{chV-3-17-1} implies \ref{chV-3-17-2} by \ref{chV-2-8}. \ref{chV-3-17-2} implies \ref{chV-3-17-3} by \ref{chV-3-10}. \ref{chV-3-17-3} obviously implies \ref{chV-3-17-4}.\\
        Suppose \ref{chV-3-17-4} holds. Let
        \begin{equation*}
            \begin{tikzcd}
                \partial \Delta^n \arrow[d] \arrow[r, "f"] & X \arrow[d] \\
                \Delta^n \arrow[r] & \Delta^0
            \end{tikzcd}
        \end{equation*}
        be a lifting problem. By \ref{chV-3-16}, there exists a homotopy from $f$ to the constant map with value $x$. Since $(\partial \Delta^n \to \Delta^n) \boxtimes (\{1\} \to \Delta^1)$ is anodyne, we obtain a homotopy $\Delta^n \times \Delta^1 \to X$ whose restriction to $\Delta^n \times \{0\}$ is the desired lift.\hfill$\Box$

    \subsection{Corollary}\label{chV-3-18}
        Let $p \colon X \to Y$ be a Kan fibration. The following are equivalent:
        \begin{enumerate}[label=\roman*)]
            \item 
                $p$ has the RLP with respect to $I$.\label{chV-3-18-1}
            \item 
                For every $y \in Y$, the fibre $p^{-1}(y)$ is contractible.\label{chV-3-18-2}
            \item 
                For every $y \in Y$, the set $\pi_0(p^{-1}(y))$ is a singleton and there exists some $x \in p^{-1}(y)$ such that $\pi_n(p^{-1}(y), x)$ is trivial for all $n \geq 1$.\label{chV-3-18-3}
        \end{enumerate}
        \smallskip

        \emph{Proof}. Assume \ref{chV-3-18-1}. Then, being a pullback, $p^{-1}(y) \to \Delta^0$ has the RLP with respect to $I$. Then \ref{chV-3-18-2} \ref{chV-3-18-3} follow by \ref{chV-3-17}. Note that \ref{chV-3-18-2} and \ref{chV-3-18-3} are equivalent, also by \ref{chV-3-17}.\\
        Suppose \ref{chV-3-18-2} holds and that we are given a lifting problem
        \begin{equation*}
            \begin{tikzcd}
                \partial \Delta^n \arrow[d] \arrow[r, "\partial \overline{\sigma}"] & X \arrow[d, "p"] \\
                \Delta^n \arrow[r, "\sigma"] & Y
            \end{tikzcd}
        \end{equation*}
        Let $h \colon \Delta^n \times \Delta^1 \to \Delta^n$ be a simplicial homotopy from $\id$ to the constant map with value $n$ (this exists e.g. by Exercise 2 on Sheet 4). Then $\sigma \circ h$ is a homotopy  from $\sigma$ to the constant map with value $\sigma(n)$.\\
        Then ${(\sigma \circ h)|}_{\partial \Delta^n \times \Delta^1}$ admits a lift to a homotopy $k$ in $X$ whose endpoint is a map $\partial \tau \colon \partial \Delta^n \to p^{-1}(\sigma(n))$. Since $p^{-1}(\sigma(n))$ is assumed to be contractible, \ref{chV-3-17} implies that $\partial \tau$ extends to a map $\tau \colon \Delta^n \to p^{-1}(\sigma(n))$. Since $(\partial \Delta^n \to \Delta^n) \boxtimes (\{1\} \to \Delta^1)$ is anodyne, there exists a homotopy $\overline{k}$ extending $k$ such that ${\overline{k}|}_{\Delta^n \times \{1\}} = \tau$ and ${\overline{k}|}_{\partial \Delta^n \times \{0\}} = \partial \overline{\sigma}$. So $\overline{\sigma} \coloneq {\overline{k}|}_{\Delta^n \times \{0\}}$ is the desired lift.\hfill$\Box$
        \medskip

        \emph{Proof of Theorem \ref{chV-3-11}}. One direction is provided by \ref{chV-3-10}.\\
        For the converse, assume that $f$ induces a bijection on $\pi_0$ and an isomorphism $\pi_n(X, x) \to \pi_n(Y, f(x))$ for all $x \in X_0$ and $n \geq 1$. By \ref{chV-2-30}, factor $f$ into an anodyne extension $i \colon X \to \overline{X}$ followed by a Kan fibration $p \colon \overline{X} \to Y$. Since $i$ is a weak equivalence of Kan complexes by \ref{chV-2-27}, it is a homotopy equivalence. So we only need to show that $p$ is an equivalence.

        We show that all fibres of $p$ are contractible.\\
        \emph{Exercise}. Show that the fibres over any two vertices in the same path component are homotopy equivalent.\\
        Since $f$ is surjective on $\pi_0$, we may therefore only consider $y \in Y_0$ such that there exists $x \in X_0$ with $f(x) = y$. By \ref{chV-3-17} and the long exact sequence of a fibration (\ref{chV-3-8}), it suffices to show that $p_{*} \colon \pi_n(\overline{X}, i(x)) \to \pi_n(Y, y)$ is a bijection for all $n \geq 0$. Since
        \begin{equation*}
            \begin{tikzcd}
                \pi_n(X, x) \arrow[rr, "i_{*}"] \arrow[dr, "f_{*}"'] & & \pi_n(\overline{X}, i(x)) \arrow[dl, "p_{*}"] \\
                & \pi_n(Y, y) &
            \end{tikzcd}
        \end{equation*}
        commutes, $f_{*}$ is assumed to be a bijection and $i_{*}$ is a bijection by \ref{chV-3-10}, the theorem follows.
        
        \quad\hfill$\Box$

    \subsection{Remark}\label{chV-3-19}
        It should in principle be possible to prove Theorem \ref{chV-3-11} without factoring the morphism, working with relative homotopy groups instead. However, this would force us to consider homotopy classes of maps of triples, which would certainly not reduce the amount of bookkeeping we would have to do.

\section[Ex]{$\Ex^{\infty}$}\label{chV-4}
    The discussion in \ref{chV-3} provides us with a good understanding of Kan fibrations which are homotopy equivalences. However, it remains unclear how these relate to Kan fibrations which are weak equivalences. For example, if $p \colon X \to Y$ is any Kan fibration, we could apply the small object argument (\ref{chV-2-30}) twice to replace $Y$ by a Kan complex $j \colon Y \to \overline{Y}$ and then the map $jp$ into an anodyne map $i \colon X \to \overline{X}$ followed by a Kan fibration $\overline{p} \colon \overline{X} \to \overline{Y}$:
    \begin{equation*}
        \begin{tikzcd}
            X \arrow[d, "p"'] \arrow[r, "i"] & \overline{X} \arrow[d, "\overline{p}"] \\
            Y \arrow[r, "j"] & \overline{Y}
        \end{tikzcd}
    \end{equation*}
    However, we are unable to control the effect of this construction on the fibres. This is remedied by a more sophisticated construction called $\Ex^{\infty}$.

    \subsection{Theorem}\label{chV-4-1}
        There exists a functor $\Ex^{\infty} \colon \sSet \to \sSet$ and a natural transformation $\rho^{\infty} \colon \id_{\sSet} \to \Ex^{\infty}$ such that the following holds:
        \begin{enumerate}[label=\roman*)]
            \item 
                $\Ex^{\infty} \Delta^0 \cong \Delta^0$;\label{chV-4-1-1}
            \item 
                $\Ex^{\infty} X$ is  Kan complex for every simplicial set $X$;\label{chV-4-1-2}
            \item 
                $\rho^{\infty} \colon X \to \Ex^{\infty} X$ is a weak equivalence for every simplicial set $X$;\label{chV-4-1-3}
            \item 
                $\Ex^{\infty}$ preserves finite limits;\label{chV-4-1-4}
            \item 
                $\Ex^{\infty}$ preserves Kan fibrations.\label{chV-4-1-5}
        \end{enumerate}
        \medskip

        Before we start with the proof of this theorem, we establish some important consequences.

    \subsection{Corollary}\label{chV-4-2}
        $\Ex^{\infty}$ preserves and detects weak equivalences.
        \smallskip

        \emph{Proof}. If $f \colon X \to Y$ is a map, it follows from the commutative square
        \begin{equation*}
            \begin{tikzcd}
                X \arrow[d, "f"'] \arrow[r, "\rho^{\infty}"] & \Ex^{\infty} X \arrow[d, "\Ex^{\infty} f"] \\
                Y \arrow[r, "\rho^{\infty}"] & \Ex^{\infty} Y
            \end{tikzcd}
        \end{equation*}
        and \ref{chV-4-1} that $f$ is a weak equivalence if and only if $\Ex^{\infty} f$ is a weak equivalence.\hfill$\Box$

    \subsection{Corollary}\label{chV-4-3}
        Let $p \colon X \to Y$ be a Kan fibration and a weak equivalence. Then $p$ has the RLP with respect to $I$.
        \smallskip

        \emph{Proof}. By \ref{chV-4-1}, we have a commutative square
        \begin{equation*}
            \begin{tikzcd}
                X \arrow[d, "p"'] \arrow[r, "\rho^{\infty}"] & \Ex^{\infty} X \arrow[d, "\Ex^{\infty} p"] \\
                Y \arrow[r, "\rho^{\infty}"] & \Ex^{\infty} Y
            \end{tikzcd}
        \end{equation*}
        Since $p$ is a weak equivalence, $\Ex^{\infty} p$ is a homotopy equivalence. Consequently, all fibres of $\Ex^{\infty} p$ are contractible. For any $y \in Y_0$, we have a weak equivalence $\rho^{\infty} \colon p^{-1}(y) \to (\Ex^{\infty} p)^{-1} (\rho^{\infty}(y))$. Since $\Ex^{\infty}$ preserves finite limits, the target of this map is the fibre of $\Ex^{\infty} p$ over $\rho^{\infty}(y)$. Consequently, $p^{-1}(y)$ is contractible. It follows that $p$ has the RLP with respect to $I$.\hfill$\Box$

    \subsection{Corollary}\label{chV-4-4}
        If $f \colon X \to Y$ is a cofibration and a weak equivalence, then it is anodyne.
        \smallskip

        \emph{Proof}. Factor $f$ into an anodyne map $i \colon X \to \overline{X}$ followed by a Kan fibration $p \colon \overline{X} \to Y$. Any lifting problem against a Kan fibration $q \colon V \to W$ gives rise to a commutative diagram
        \begin{equation*}
            \begin{tikzcd}
                X \arrow[d, "i"'] \arrow[rr] & & V \arrow[d, "q"] \\
                \overline{X} \arrow[r, "p"] & Y \arrow[r] & W
            \end{tikzcd}
        \end{equation*}
        There exists a lift $l \colon \overline{X} \to V$. Since $f$ is a weak equivalence, so is $p$. By \ref{chV-4-3}, $p$ has the RLP with respect to $I$. In particular, there exists by \ref{chV-2-8} a section $s$. Then $l \circ s$ is the desired lift.

    \subsection{Corollary (HELP)}\label{chV-4-5}
        Let $i \colon A \to B$ be a cofibration and let $p \colon X \to Y$ be a Kan fibration. If $i$ or $p$ is a weak equivalence, then the canonical map
        \begin{equation*}
            F(B, X) \to F(A, X) \underset{F(A, Y)}{\times} F(B, Y)
        \end{equation*}
        has the RLP with respect to $I$.
        \smallskip

        \emph{Proof}. If $i$ is a weak equivalence, it is an anodyne extension by \ref{chV-4-4}. Then the corollary follows from \ref{chV-2-23}.\\
        As usual, any right lifting problem with respect to $\partial \Delta^n \to \Delta^n$ corresponds to a lifting problem
        \begin{equation*}
            \begin{tikzcd}
                \Delta^n \times A \underset{\partial \Delta^n \times A}{\cup} \partial \Delta^n \times B \arrow[d, "{(\partial \Delta^n \to \Delta^n)} \, \boxtimes \, i"'] \arrow[r] & X \arrow[d, "p"] \\
                \Delta^n \times B \arrow[r] & Y
            \end{tikzcd}
        \end{equation*}
        If $p$ is a weak equivalence, it has the RLP with respect to $I$ by \ref{chV-4-3}, so this lifting problem has a solution.\hfill$\Box$

    \subsection{Remark}\label{chV-4-6}
        Informally, \ref{chV-4-5} tells us that ``the space of solutions to any lifting problem
        \begin{equation*}
            \begin{tikzcd}
                A \arrow[d, "i"'] \arrow[r] & X \arrow[d, "p"] \\
                B \arrow[r] & Y
            \end{tikzcd}
        \end{equation*}
        is contractible''.
        \medskip

        The proof of \ref{chV-4-1} occupies the remainder of this section. We begin by defining all the relevant constructions.

    \subsection{Definition}\label{chV-4-7}
        Let $P$ be a poset. Define $\chains(P)$ as the poset whose elements are finite, linearly ordered subsets of $P$ such that a chain $c$ is smaller then a chain $c'$ if $c$ is a suborder of $c'$.
        \medskip

        This defines an endofunctor on the category of posets.

    \subsection{Definition}\label{chV-4-8}
        Define $\sd \Delta^{\bullet} \colon \Delta \to \sSet$ as the functor sending $[n]$ to $N(\chains[n])$. The \emph{(barycentric) subdivision} $\sd \colon \sSet \to \sSet$ is the unique colimit-preserving extension of $\sd \Delta$. By Exercise 2 on Sheet 3. $\sd$ has a right adjoint $\Ex \colon \sSet \to \sSet$.

    \subsection{Definition}\label{chV-4-9}
        The \emph{last vertex map} is the natural transformation
        \begin{equation*}
            \lambda \,\colon \sd \to \id_{\sSet}
        \end{equation*}
        induced by the natural map
        \begin{equation*}
            \chains[n] \to [n], \quad c \mapsto \max c \,.
        \end{equation*}
        Denote by $\rho$ the adjoint transformation
        \begin{equation*}
            \rho \,\colon \id_{\sSet} \to \Ex \,.
        \end{equation*}

    \subsection{Definition}\label{chV-4-10}
        Define $\Ex^{\infty} \colon \sSet \to \sSet$ as the colimit of
        \begin{equation*}
            \id_{\sSet} \xrightarrow{\rho} \Ex \xrightarrow{\rho \Ex} \Ex^2 \xrightarrow{\rho \Ex^2} \Ex^3 \rightarrow \ldots
        \end{equation*}

        Let $\rho^{\infty} \colon \id_{\sSet} \to \Ex^{\infty}$ be the induced natural transformation.

    \subsection{Corollary}\label{chV-4-11}
        $\Ex^{\infty}$ commutes with finite limits.
        \smallskip

        \emph{Proof}. Since $\Ex$ is a right adjoint, it commutes with arbitrary limits. Filtered colimits commute with finite limits in Set (by inspection), so the same statement holds in $\sSet$. This implies the corollary.\hfill$\Box$
        \medskip

        Since $\Ex$ arises as the right adjoint of $\sd$, everything we want to know about $\Ex$ will be deduced from appropriate statements about the subdivision functor.

    \subsection{Proposition}\label{chV-4-12}
        Let $X$ be a simplicial set. Then every horn in $\Ex X$ has a filler in $\Ex^2 X$.
        \smallskip

        \emph{Proof}. For every map $f \colon \horn{n}{k} \to \Ex X$, we have to show that $\horn{n}{k} \xrightarrow{f} \Ex X \xrightarrow{\rho \Ex X} \Ex^2 X$ admits a filler. Let $g \colon \sd \horn{n}{k} X$ be the map determined by $f$. The map adjoint to $\rho \Ex X \circ f$ is given by $\lambda_{\Ex X} \circ \sd f$, which is the same as $f \circ \lambda_{\horn{n}{k}}$ by naturality of $\lambda$. The adjoint of this map is
        \begin{equation*}
            \sd^2 \horn{n}{k} \xrightarrow{\sd \lambda_{\horn{n}{k}}} \sd \horn{n}{k} \xrightarrow{g} X \,.
        \end{equation*}
        Consequently, we have to show that the dotted arrow in the diagram
        \begin{equation*}
            \begin{tikzcd}
                \sd^2 \horn{n}{k} \arrow[d] \arrow[r, "\sd \lambda"] & \sd \horn{n}{k} \arrow[d, "g"] \\
                \sd^2 \Delta^n \arrow[r, dotted] & X
            \end{tikzcd}
        \end{equation*}
        exists and makes the resulting square commute. Note that it suffices to find the dotted arrow in the case $g = \id$.\\
        Observe that $\sd \horn{n}{k}$ is the nerve of the poset $P \coloneq \chains[n] \setminus \{ [n], [n] \setminus \{k\}\}$, so we have to specify a map of posets
        \begin{equation*}
            \chains \chains[n] \to P, \quad (c_0 \subseteq \ldots \subseteq c_r) \mapsto \{ \widetilde{\max}(c_0), \ldots, \widetilde{\max}(c_r) \} \,,
        \end{equation*}
        where $\widetilde{\max}$ is defined by
        \begin{equation*}
            \widetilde{\max}(c) \coloneq
            \begin{cases*}
                k & $c = [n]$ or $c = [n] \setminus \{k\}$, \\
                \max c & else.
            \end{cases*}
        \end{equation*}
        It is straightforward to check that this is well-defined, and the map evidently extends $\sd \lambda$.\hfill$\Box$

    \subsection{Corollary}\label{chV-4-13}
        $\Ex^{\infty} X$ is a Kan complex for every simplicial set $X$.
        \smallskip

        \emph{Proof}. Since $\horn{n}{k}$ is compact, every map $\horn{n}{k} \to \Ex^{\infty}$ factors through some $\Ex^n X$. Assume $n \geq 1$. Then $\horn{n}{k} \to \Ex^n X \xrightarrow{\rho \Ex^n} \Ex^{n+1} X$ extends to a map $\Delta^n \to \Ex^{n+1} X$ by \ref{chV-4-12}, and this provides also a filler in $\Ex^{\infty} X$.\hfill$\Box$
        \medskip

        We consider the question whether $\Ex$ preserves Kan fibrations next. By the adjunction between $\sd$ and $\Ex$, this reduces to the question whether $\sd$ preserves anodyne extensions.

    \subsection{Lemma}\label{chV-4-14}
        Let $J$ be a non-empty, finite set. The inclusion $L_J \coloneq \Pp(J) \setminus \{J\} \to \Pp(J)$ induces a left anodyne map on nerves.
        \smallskip

        \emph{Proof}. Pick some element $j \in J$. Then the inclusion map $N(L_J) \to N(\Pp(J))$ is isomorphic to the map
        \begin{equation*}
            (N(L_{J \setminus \{j\}}) \times \Delta^1) \underset{N(L_{J \setminus \{j\}}) \times \{0\}}{\cup} (N(\Pp(J \setminus \{j\})) \times \{0\}) \rightarrow N(\Pp(J \setminus \{j\})) \times \Delta^1 \,.
        \end{equation*}
        This map is left anodyne by \ref{chV-2-22}.\hfill$\Box$

    \subsection{Proposition}\label{chV-4-15}
        The subdivision functor preserves anodyne extensions.
        \smallskip

        \emph{Proof}. Since $\sd$ is a colimit-preserving functor, the class of morphisms
        \begin{equation*}
            \Sigma \coloneq \{ i \mid \sd i \text{ is anodyne} \}
        \end{equation*}
        is saturated. Hence it suffices to show that $H \subseteq \Sigma$. Recall that $\sd \Delta^n$ is the nerve of the poset $\chains[n]$, and that $\horn{n}{k}$ is the nerve of the subposet $P \coloneq \chains[n] \setminus \{ [n], [n] \setminus \{k\} \}$. In this proof, we identify $\chains[n]$ with the set $\Pp_{*}([n])$ of non-empty subsets of $[n]$.\\
        Set $[n]_k \coloneq [n] \setminus \{k\}$. The map
        \begin{equation*}
            \Pp([n]_k) \to \Pp_{*}([n]), \quad S \mapsto S \cup \{ k \}
        \end{equation*}
        is injective. Let $X \subseteq \sd \Delta^n$ denote the image of the induced map on nerves. Then $X \cap \sd \horn{n}{k} \cong N(L_{[n]_k})$, and there is a pushout square
        \begin{equation*}
            \begin{tikzcd}
                N(L_{[n]_k}) \cong X \cap \sd \horn{n}{k} \arrow[d] \arrow[r] & \sd \horn{n}{k} \arrow[d] \\
                N(\Pp([n]_k)) \cong X \arrow[r] & Y
            \end{tikzcd}
        \end{equation*}
        Consequently, $\sd \horn{n}{k} \to X$ is left anodyne by \ref{chV-4-14}.

        Consider now the injective map
        \begin{equation*}
            (\Pp([n]_k) \setminus \{\varnothing\}) \times [1] \to \Pp_{*}([n]), \quad (S, r) \mapsto
            \begin{cases*}
                S & $r = 0$,\\
                c(S) = S \cup \{k\} & $r = 1$,
            \end{cases*}
        \end{equation*}
        and denote its image by $Z$. Then
        \begin{equation*}
            Z \cap Y \cong N((\Pp([n]_k) \setminus \{\varnothing\}) \times \{1\}) \underset{N((\Pp([n]_k) \setminus \{\varnothing, [n]_k\}) \times \{1\})}{\cup} N((\Pp([n]_k) \setminus \{\varnothing, [n]_k\}) \times \{1\})
        \end{equation*}
        and there is a pushout square
        \begin{equation*}
            \begin{tikzcd}
                Z \cap Y \arrow[d] \arrow[r] & Y \arrow[d] \\
                Z \arrow[r] & \sd \Delta^n
            \end{tikzcd}
        \end{equation*}
        The left vertical map is right anodyne by \ref{chV-2-22}, so $X \to \sd \Delta^n$ is also right anodyne. Consequently, the composition $\sd \horn{n}{k} \to X \to \sd \Delta^n$ is anodyne.\hfill$\Box$

    \subsection{Corollary}\label{chV-4-16}
        $\Ex^n$ preserves Kan fibrations for every $n \in \mathbb{N} \cup \{\infty\}$.
        \smallskip

        \emph{Proof}. Suppose by induction that $\Ex^n$ preserves Kan fibrations. Any lifting problem for $\Ex^{n+1} p$, where $p \colon X \to Y$ is a Kan fibration, corresponds to a lifting problem
        \begin{equation*}
            \begin{tikzcd}
                \sd \horn{n}{k} \arrow[d] \arrow[r] & \Ex^n X \arrow[d, "\Ex^n p"] \\
                \sd \Delta^n \arrow[r] & \Ex^n Y
            \end{tikzcd}
        \end{equation*}
        Since $\sd \horn{n}{k} \to \sd \Delta^n$ is anodyne by \ref{chV-4-15}, a solution to the lifting problem exists.\\
        The statement for $n = \infty$ follows since $\horn{n}{k}$ and $\Delta^n$ are compact objects in $\sSet$.\hfill$\Box$
        \medskip

        Since $\Ex^{\infty} \Delta^0$ is obviously isomorphic to $\Delta^0$, \ref{chV-4-11}, \ref{chV-4-13} and \ref{chV-4-16} take care of \ref{chV-4-1} with the exception of \ref{chV-4-1-3}.

    \subsection{Proposition}\label{chV-4-17}
        Let $X, Y \colon \mathbb{N} \to \sSet$ be diagrams and let $f \colon X \to Y$ be a transformation. Suppose that $x(n) \colon X(n) \to X(n+1)$ and $y(n) \colon Y(n) \to Y(n+1)$ are cofibrations for all $n$, and that $f(n) \colon X(n) \to Y(n)$ is a weak equivalence for every $n$.\\
        Then the induced map $\colim\limits_{\mathbb{N}} X \to \colim\limits_{\mathbb{N}} Y$ is a weak equivalence.
        \smallskip

        \emph{Proof}. Let $g \colon V \to W$ be an arbitrary transformation of diagrams $\mathbb{N} \to \sSet$. We claim that $g$ can be factored as $g = p i$, where $i(n)$ is anodyne for every $n$ and $p(n)$ is a Kan fibration for every $n$.\\
        This factorisation can be built by induction. In degree $0$, apply \ref{chV-2-20} to factor $g(0)$ into an anodyne $i(0) \colon V(0) \to \overline{V}(0)$ and a Kan fibration $p(0) \colon \overline{V}(0) \to W(0)$. Then the induced map $j(1) \colon V(1) \to V(1) \underset{V(0)}{\cup} \overline{V}(0)$ is anodyne and the map $\overline{V}(0) \to V(1) \underset{V(0)}{\cup} \overline{V}(0)$ is a cofibration. Moreover, $g(1)$ induces a map $V(1) \underset{V(0)}{\cup} \overline{V}(0) \to W(0)$. Factor this map into an anodyne map $j'(1) \colon V(1) \underset{V(0)}{\cup} \overline{V}(0) \overline{V}(1)$ and a Kan fibration $p(1) \colon \overline{V}(1) \to W(1)$. Set $i(1) \coloneq j'(1) \circ j(1)$. Processing like this inductively produces the desired factorisaton.\\
        Apply the claim to the unique transformation $Y \to \Delta^0$ to obtain a transformation $j \colon Y \to \overline{Y}$ such that $j(n)$ is anodyne for all $n$ and $\overline{Y}(n)$ is a Kan complex for every $n$. Apply the claim a second time to $jf \colon X \to \overline{Y}$, producing a pointwise anodyne transformation $i \colon X \to \overline{X}$ and a pointwise Kan fibration $p \colon \overline{X} \to \overline{Y}$. Since $Y(n)$ is a Kan complex and $p(n)$ is a Kan fibration, $\overline{X}(n)$ is a Kan complex for every $n$. Moreover, $p(n)$ is a weak equivalence, so it has the RLP with respect to $I$ by \ref{chV-3-18}, \ref{chV-3-8} and \ref{chV-3-11}.\\
        Since $\partial \Delta^n$ and $\Delta^n$ are compact, it follows directly that $\colim\limits_{\mathbb{N}} p$ has the RLP with respect to $I$. In particular, $\colim\limits_{\mathbb{N}} p$ is a weak equivalence.\\
        Therefore, it suffices to show the proposition in the case when all components of the transformation are anodyne: then both $\colim\limits_{\mathbb{N}} j$ and $\colim\limits_{\mathbb{N}} i$ are weak equivalences, which readily implies that $\colim\limits_{\mathbb{N}} f$ is a weak equivalence.\\
        Hence, assume that $f(n)$ is anodyne for all $n$. Let $K$ be a Kan complex. We have to show that the induced map
        \begin{equation*}
            (\colim\limits_{\mathbb{N}} f)^{*} \,\colon \pi_0 F(\colim\limits_{\mathbb{N}} Y, K) \to \pi_0 F(\colim\limits_{\mathbb{N}} X, K)
        \end{equation*}
        is a bijection. This map fits into a commutative square
        \begin{equation*}
            \begin{tikzcd}[column sep=large]
                \pi_0 F(\colim\limits_{\mathbb{N}} Y, K) \arrow[d, "\cong"'] \arrow[r, "{(\colim\limits_{\mathbb{N}} f)^{*}}"] & \pi_0 F(\colim\limits_{\mathbb{N}} X, K) \arrow[d, "\cong"] \\
                \pi_0(\lim\limits_{\mathbb{N}^{\op}} F(Y, K)) \arrow[r, "{(\lim\limits_{\mathbb{N}^{\op}} f)^{*}}"] & \pi_0(\lim\limits_{\mathbb{N}^{\op}} F(X, K))
            \end{tikzcd}
        \end{equation*}
        Let us show directly that the lower horizontal map is a bijection.\\
        We claim in slightly greater generality that if we have a transformation between towers of Kan fibrations of Kan complexes
        \begin{equation*}
            \begin{tikzcd}
                \ldots \arrow[r] & U(2) \arrow[d, "{q(2)}"] \arrow[r, "{u(2)}"] & U(1) \arrow[d, "{q(1)}"] \arrow[r, "{u(1)}"] & U(0) \arrow[d, "{q(0)}"] \\
                \ldots \arrow[r] & V(2) \arrow[r, "{v(2)}"] & V(1) \arrow[r, "{v(1)}"] & V(0)
            \end{tikzcd}
        \end{equation*}
        such that $q(n)$ has the RLP with respect to $I$ for all $n$, then the induced map $\pi_0(\lim\limits_{\mathbb{N}^{\op}} U) \to \pi_0(\lim\limits_{\mathbb{N}^{\op}} V)$ is a bijection.\\
        Given $(b_n)_n \in \lim\limits_{\mathbb{N}^{\op}} V$, there exists a sequence $(a'_n)_n \prod_{\mathbb{N}} U(n)$ such that $q(n)(a'_n) = b_n$. Set $a_0 \coloneq a'_0$. Since $q(0)(u(1)(a'_1)) = b_0$, the constant homotopy on $b_0$ lifts to a homotopy $u(1)(a'_1) \simeq a_0$. Since $u(1)$ is a Kan fibration, this homotopy lifts to a homotopy $h_1 \colon \Delta^1 \to U(1)$ from $a'_1$ to some vertex $a_1$. Note that $u(1)(a_1) = a_0$. Moreover, $q(1) \circ h_1$ provides a homotopy $b_1 \simeq q(1)(a_1)$ such that $v(1) \circ q(1) \circ h_1$ is the constant homotopy on $b_0$. Continuing inductively, we find a vertex $(a_n)_n \in \lim\limits_{\mathbb{N}^{\op}} U$ and a homotopy $\Delta^1 \to \lim\limits_{\mathbb{N}^{\op}} V$ from $(b_n)_n$ to $(q(n)(a_n))_n$. So the comparison map is surjective.\\
        Suppose that $(a_n)_n$, $(a'_n)_n$ are two vertices such that there exists a homotopy $\Delta^1 \to \lim\limits_{\mathbb{N}^{\op}} V$ from $(q(n)(a_n))_n$ to $(q(n)(a'_n))_n$, i.e. a sequence of homotopies $(h_n \colon \Delta^1 \to V(n))_n$ such that $h_n$ is a homotopy $q(n)(a_n) \simeq q(n)(a'_n)$ and $v(n+1) \circ h_{h+1} = h_n$. Since each $q(n)$ has the RLP with respect to $I$, there exists a sequence of homotopies $(k'_n \colon \Delta^1 \to U(n))_n$ such that $k'_n$ is a homotopy $a_n \simeq a'_n$. Set $k_0 \coloneq k'_0$.\\
        Consider the map $H_0 \colon \partial \Box^2 \to U(0)$ which is given by $u(1)k'_1$ and $k_0$ on two opposite faces, respectively. Then $q(0) \circ H_0$ extends to the map $\Box^2 \xrightarrow{\pr} \Delta^1 \xrightarrow{h_0} V(0)$. As $q(0)$ has the RLP with respect to $I$, the map $H_0$ extends to a map $K_0 \colon \Box^2 \to U(0)$ lifting the given map. Since $u(1)$ is a Kan fibration, $K_0$ lifts further to a homotopy $k'_1 \simeq k_1$ (relative endpoints). Again, one continues this procedure inductively to obtain a homotopy $\Delta^1 \to \lim\limits_{\mathbb{N}^{\op}} U$ from $(a_n)_n$ to $(a'_n)_n$. This proves injectivity, and thus finishes the proof of the proposition.\hfill$\Box$

    \subsection{Proposition}\label{chV-4-19}
        The last vertex map $\lambda \colon \sd X \to X$ is a weak equivalence for every simplicial set $X$.
        \smallskip

        \emph{Proof}. Since $\sd$ preserves colimits and cofibrations, it follows from \ref{chV-4-17} that it suffices to show that $\lambda \colon \sd \sk_n(X) \to \sk_n(X)$ is a weak equivalence for every $n$. We do this by induction on $n$. For $n = 0$, one sees directly that $\lambda_{\sk_0(X)}$ is an isomorphism.\\
        Let $K$ be a Kan complex. The vertical maps in the commutative square
        \begin{equation*}
            \begin{tikzcd}[column sep=large]
                F(\sk_{n+1}(X), K) \arrow[d] \arrow[r, "{\lambda^{*}_{\sk_{n+1}(X)}}"] & F(\sd \sk_{n+1}(X), K) \arrow[d] \\
                F(\sk_n(X), K) \arrow[r, "{\lambda^{*}_{\sk_n(X)}}"] & f(\sd \sk_n(X), K)
            \end{tikzcd}
        \end{equation*}
        are Kan fibrations of Kan complexes, and $\lambda^{*}_{\sk_n (X)}$ is a weak equivalence by induction hypothesis.\\
        By \ref{chV-4-18}, it suffices to show that the induced maps on vertical fibres are weak equivalences.\\
        Consider the pullback square
        \begin{equation*}
            \begin{tikzcd}
                F(\sk_{n+1} X, K) \arrow[d] \arrow[r] & F(\coprod\limits_{I_{n+1}} \partial \Delta^{n+1}, K) \arrow[d] \\
                F(\sk_n X, K) \arrow[r] & F(\coprod\limits_{I_{n+1}} \Delta^{n+1}, K)
            \end{tikzcd}
        \end{equation*}
        arising from the pushout describing the attachment of $(n+1)$-simplices. Since $\sd$ preserves colimits, there is an analogous square in which all sources of function complexes have bin replaced by their subdivisions. Consequently, for $f \in F(\sk_n(X), K)$ the induced map on fibres can be identified with
        \begin{equation*}
            \{f'\} \underset{F(\coprod\limits_{I_{n+1}} \partial \Delta^{n+1}, K)}{\times} F(\coprod\limits_{I_{n+1}} \Delta^{n+1}, K) \to \{f'\} \underset{F(\coprod\limits_{I_{n+1}} \sd \partial \Delta^{n+1}, K)}{\times} F(\coprod\limits_{I_{n+1}} \sd \Delta^{n+1}, K) \,, 
        \end{equation*}
        where $f' \colon \coprod\limits_{I_{n+1}} \partial \Delta^{n+1} \to K$ is the map induced by $f$.\\
        Again by \ref{chV-4-18}, it suffices to show that the two horizontal maps in the diagram 
        \begin{equation*}
            \begin{tikzcd}
                F(\coprod\limits_{I_{n+1}} \Delta^{n+1}, K) \arrow[d] \arrow[r, "\lambda^{*}"] & F(\coprod\limits_{I_{n+1}} \sd \Delta^{n+1}, K) \arrow[d] \\
                F(\coprod\limits_{I_{n+1}} \partial \Delta^{n+1}, K) \arrow[r, "\lambda^{*}"] & F(\coprod\limits_{I_{n+1}} \sd \partial \Delta^{n+1}, K)
            \end{tikzcd}
        \end{equation*}
        are homotopy equivalences. The lower horizontal map is again a homotopy equivalence by induction hypothesis.\\
        A product of homotopy equivalences is a homotopy equivalence, so it suffices to see that $\lambda_{\Delta^{n+1}} \colon \sd \Delta^{n+1} \to \Delta^{n+1}$ is a homotopy equivalence. Since both $\sd \Delta^{n+1}$ and $\Delta^{n+1}$ are nerves of posets with a largest element, this is immediate from Exercise 2 on Sheet 4.\hfill$\Box$

    \subsection{Lemma}\label{chV-4-20}
        Let $X$ be a simplicial set and let $K$ be a Kan complex. Then the bijection $\Hom_{\sSet}(\sd X, K) \cong \Hom_{\sSet}(X, \Ex K)$ induces a bijection
        \begin{equation*}
            \pi_0 F(\sd X, K) \cong \pi_0 F(X, \Ex K) \,.
        \end{equation*}
        \smallskip

        \emph{Proof}. Suppose that $h \colon \sd X \times \Delta^1 \to K$ is a homotopy. Then the adjoint morphism to
        \begin{equation*}
            \sd(X \times \Delta^1) \to \sd X \times \sd \Delta^1 \xrightarrow{\sd X \times \lambda_{\Delta^1}} \sd X \times \Delta^1 \xrightarrow{h} K
        \end{equation*}
        provides a homotopy between the morphisms adjoint to $h_0$ and $h_1$.\\
        Suppose that $X \times \Delta^1 \to \Ex K$ is a homotopy. Let $h \colon \sd(X \times \Delta^1) \to K$ be the adjoint morphism. Let $k$ denote the composition $\sd(X \times \Delta^1) \to \sd X \xrightarrow{h_0} K$. Since $X \times \{0\} \to X \times \Delta^1$ is anodyne, the restriction map
        \begin{equation*}
            F(\sd(X \times \Delta^1), K) \to F(\sd(X \times \{0\}), K)
        \end{equation*}
        has the RLP with respect to $I$ by \ref{chV-4-15} and \ref{chV-2-26}. Since $h$ and $k$ restrict to the same map $\sd(X \times \{0\}) \to K$, there exists a homotopy $\sd(X \times \Delta^1) \times \Delta^1 \to K$ between $h$ and $k$.The restriction of this homotopy to $\sd(X \times \{1\}) \times \Delta^1$ provides the required homotopy between $h_0$ and $h_1$.\hfill$\Box$

    \subsection{Corollary}\label{chV-4-21}
        The transformation $\rho \colon K \to \Ex K$ induces a bijection
        \begin{equation*}
            \pi_0 F(X, K) \xrightarrow{\sim} \pi_0 F(X, \Ex K)

        for every Kan complex $K$ and every simplicial set $X$.
        \smallskip

        \emph{Proof}. By \ref{chV-4-20}, there is a commutative triangle

                & \pi_0 F(\sd X, K)

        The lower diagonal arrow is a bijection by \ref{chV-4-19}.\hfill$\Box$

    \subsection{Lemma}\label{chV-4-22}
        Let $X$ be a simplicial set. Then there is a simplicial homotopy $\rho_{\Ex X} \simeq \Ex \rho_X$.
        \smallskip

        \emph{Proof}. Define the map of posets
        \begin{align*}
            h \,\colon \chains(\chains[n]) \times [q] &\to \chains[n] \\
            (c_0 \subseteq \ldots \subseteq c_k, i) &\mapsto
                \begin{cases*}
                    \max c_0 \leq \ldots \leq \max c_k & $i = 0$,\\
                    c_k & $i = 1$
                \end{cases*}
        \end{align*}
        Note that $\max c_0 \leq \ldots \leq \max c_k$ is a suborder of $c_k$, so this is well-defined. Taking nerves, we obtain a homotopy from $\sd \lambda_{\Delta^n}$ to $\lambda_{\sd \Delta^n}$. This is natural in $n$, so we obtain a natural transformation $\sd^2(-) \times \Delta^1 \Rightarrow \sd$ (using that $ - \times \Delta^1$ commutes with arbitrary colimits). Since there is also a natural transformation $\sd^2(- \times \Delta^1) \Rightarrow \sd^2(-) \times \Delta^1$, we obtain a natural transformation $\sd^2(- \times \Delta^1) \Rightarrow \sd$. Using the solution to Exercise 1 on Sheet 4, we obtain a natural transformation $\Ex \Rightarrow F(\Delta^1, \Ex^2(-))$ which provides (by ``currying'') the required homotopy $\rho \Ex \simeq \Ex \rho$.\hfill$\Box$

    \subsection{Proposition}\label{chV-4-23}
        The transformation $\rho \colon X \to \Ex X$ is a weak equivalence for every simplicial set $X$.
        \smallskip

        \emph{Proof}. Let $K$ be a Kan complex. Consider the diagram
        \begin{equation*}
            \begin{tikzcd}
                \pi_0 F(\Ex X, K) \arrow[d, "{(\rho_K)_{*}}"'] \arrow[r, "\rho^{*}_X"] & \pi_0 F(X, K) \arrow[dl, dashed] \arrow[d, "{(\rho_K)_{*}}"] \\
                \pi_0 F(\Ex X, \Ex K) \arrow[r, "\rho^{*}_X"] & \pi_0 F(X, \Ex K)
            \end{tikzcd}
        \end{equation*}
        The dashed diagonal map is induced by the functoriality of $\Ex$: note that $\Ex$ preserves simplicial homotopy since a homotopy $X \times \Delta^1 \to K$ gives rise to a homotopy $\Ex X \times \Delta^1 \to \Ex X \times \Ex \Delta^1 \cong \Ex(X \times \Delta^1) \to K$. Both vertical maps are bijections by \ref{chV-4-21}. Then the lower triangle commutes by the naturality of $\rho$.\\
        For $f \colon \Ex X \to K$, we have by \ref{chV-4-22}
        \begin{equation*}
            \Ex(f \circ \rho_X) = \Ex f \circ \Ex \rho_X \simeq \Ex f \circ \rho_{\Ex X} = \rho_k \circ f \,.
        \end{equation*}
        So the upper triangle commutes as well.\\
        It follows that all maps in the diagram are bijections.\hfill$\Box$

\section{The comparison with the homotopy theory of CW-complexes}\label{chV-5}
    To finish our discussion of simplicial homotopy theory, we outline the relation to the homotopy theory of CW-complexes. We will assume some facts about the geometric realisation functor. This is summarized in the following theorem:

    \subsection{Theorem}\label{chV-5-1}
        Let $\Kan$ denote the simplicially enriched category of Kan complexes and let $\CW$ denote the simplicially enriched category of CW-complexes (by taking $\Sing$ of the spaces of all continuous maps). Then both
        \begin{equation*}
            \gr{-} \,\colon \Kan \to \CW
        \end{equation*}
        and
        \begin{equation*}
            \Sing \,\colon \CW \to \Kan
        \end{equation*}
        are Dwyer-Kan equivalences, i.e.
        \begin{itemize}
            \item 
                they induced essentially surjective functors on homotopy categories and
            \item 
                they induce weak equivalences on all mapping spaces.
        \end{itemize}
        \medskip

        The remainder of this section os occupied by the proof of \ref{chV-5-1}.

    \subsection{Theorem}\label{chV-5-6}
        The following are equivalent:
        \begin{enumerate}[label=\roman*)]
            \item 
                Simplicial approximation holds for maps from finite simplicial sets: Let $A \subseteq X$ be finite simplicial sets and let $Y$ be a simplicial set. Suppose that $f_0 \colon A \to Y$ and $g \colon \gr{X} \to \gr{Y}$ are maps such that ${g|}_{\gr{A}} = \gr{f_0}$.\\
                Then there exists for some $n$ a map $f \colon \sd^n X \to Y$ such that
                \begin{equation*}
                    \begin{tikzcd}
                        \sd^n A \arrow[d] \arrow[r, "\lambda^n"] & A \arrow[r, "f_0"] & Y \\
                        \sd^n X \arrow[urr, "f"] & &
                    \end{tikzcd}
                \end{equation*}
                commutes and a homotopy $\gr{F} \simeq g\gr{\lambda^n}$ relative $\gr{\sd^n A}$.\label{chV-5-6-1}
            \item 
                Simplicial approximation holds for maps into Kan complexes: Let $A \subseteq X$ be simplicial sets and let $K$ be a Kan complex. Suppose that $f_0 \colon A \to K$ and $g \colon \gr{X} \to \gr{K}$ are maps such that ${g|}_{\gr{A}} = f_0$.\\
                Then there exists a map $f \colon X \to K$ and a homotopy $\gr{f} \simeq g$ relative $\gr{A}$.\label{chV-5-6-2}
            \item 
                The unit morphism $X \to \Sing \gr{X}$ is a homotopy equivalence for every Kan complex. \label{chV-5-6-3}
            \item 
                The unit morphism $X \to \Sing \gr{X}$ is a weak equivalence for every simplicial set.\label{chV-5-6-4}
            \item 
                Geometric realisation induces an equivalence $F(X, K) \xrightarrow{\simeq} F(\gr{X}, \gr{K})$ for every simplicial set $X$ and every Kan complex $K$.\label{chV-5-6-5}
        \end{enumerate}
        \smallskip

        \emph{Proof}. We only sketch why \ref{chV-5-6-1} is equivalent \ref{chV-5-6-2}. If \ref{chV-5-6-1} holds, one constructs an approximation simplex by simplex. This reduces the problem to the case $A = \partial \Delta^n$ and $X = \Delta^n$. In this case, the simplicial approximation whose existence is guaranteed by \ref{chV-5-6-1} yields a commutative square
        \begin{equation*}
            \begin{tikzcd}
                \partial \Delta^n \arrow[d] \arrow[r, "f_0"] & K \arrow[d, "\rho^n"] \\
                \Delta^n \arrow[r, "f"] & \Ex^n K
            \end{tikzcd}
        \end{equation*}
        Since $\rho^n$ is an anodyne map between Kan complexes, it has a retraction. This yields the desired extension.\\
        Conversely, we may replace $Y$ by $\Ex^{\infty} Y$ and apply \ref{chV-5-6-2}. Since $X$ is assumed to be finite, the resulting map $X \to \Ex^{\infty} Y$ factors through some $Ex^n Y$ and gives thus rise to a map $\sd^n X \to Y$ with the desired properties.\\
        We now show \ref{chV-5-6-2} $\Rightarrow$ \ref{chV-5-6-3}. By one of the triangle identities, the composition
        \begin{equation*}
            \gr{X} \xrightarrow{\gr{u_X}} \gr{\Sing \gr{X}} \xrightarrow{c_{\gr{X}}} \gr{X}
        \end{equation*}
        is the identity. By \ref{chV-5-6-2}, there exists a map $k \colon \Sing \gr{X} \to X$ extending $\id_X$ such that $\gr{k}$ is homotopic to $c_{\gr{X}}$ relative $\gr{X}$. Let $h \colon \gr{\Sing \gr{X}} \times [0, 1] \to \gr{X}$ be such a homotopy. Since $\gr{\Delta^1} \cong [0, 1]$ and $\gr{-}$ commutes with finite products, this corresponds to a map
        \begin{equation*}
            \gr{\Sing \gr{X} \times \Delta^1} \to \gr{X}
        \end{equation*}
        whose adjoint provides a deformation retraction of $\Sing \gr{X}$ onto $\gr{X}$.\\
        For \ref{chV-5-6-3} $\Rightarrow$ \ref{chV-5-6-4}, consider the commutative square
        \begin{equation*}
            \begin{tikzcd}[column sep=large]
                X \arrow[d, "\rho^{\infty}"'] \arrow[r, "u_X"] & \Sing \gr{X} \arrow[d, "\Sing \gr{\rho^{\infty}}"] \\
                \Ex^{\infty} \arrow[r, "u_{\Ex^{\infty} X}"] & \Sing \gr{\Ex^{\infty} X}
            \end{tikzcd}
        \end{equation*}
        The vertical maps are weak equivalences by \ref{chV-4-1} and \ref{chV-5-5}, and the lower horizontal map is a weak equivalence by assumption. So $u_X$ is also a weak equivalence.\\
        For \ref{chV-5-6-4} $\Rightarrow$ \ref{chV-5-6-5}, consider the commutative diagram
        \begin{equation*}
            \begin{tikzcd}
                & F_{\CW}(\gr{X}, \gr{K}) \arrow[dd, "\cong"] \\
                F_{\sSet}(X, K) \arrow[ur, "\gr{-}"] \arrow[dr, "{(u_K)_{*}}"] & \\
                & F_{\sSet}(X, \Sing \gr{K})
            \end{tikzcd}
        \end{equation*}
        Since $u_K$ is a homotopy equivalence by assumption, \ref{chV-5-6-5} follows.\\

            \end{tikzcd}

        and by assumption both horizontal maps are weak equivalences. It follows from \ref{chV-4-18} that the induced map on fibres over $f_0$ and $\gr{f_0}$ is a homotopy equivalence. Picking a preimage of $g$ yields the desired simplicial approximation.\hfill$\Box$

    \subsection{Corollary}\label{chV-5-7}
        The counit $\gr{\Sing T} \to T$ is a weak equivalence for every $T \in \Top$.
        \smallskip

        \emph{Proof}. By \ref{chV-5-4}, it suffices to check that $\Sing c_T \colon \Sing \gr{\Sing T} \to \Sing T$ induces bijections on all $\pi_n$. By one of the triangle identities, the identity on $\Sing T$ factors as
        \begin{equation*}
            \Sing T \xrightarrow{u_{\Sing T}} \Sing \gr{\Sing T} \xrightarrow{\Sing c_T} \Sing T \,.
        \end{equation*}
        Since $u_{\Sing T}$ is a homotopy equivalence, $\Sing c_T$ is also a homotopy equivalence. So the corollary follows from \ref{chV-3-10}.\hfill$\Box$

    \subsection{Corollary}\label{chV-5-8}
        The geometric realisation detects weak equivalences, i.e. if $f \colon X \to Y$ is a map of simplicial sets such that $\gr{f}$ is a weak equivalence, then $f$ is a weak equivalence.
        \smallskip

        \emph{Proof}. By \ref{chV-5-5}, $\Sing \gr{f}$ is a weak equivalence. Then it follows from the commutative diagram
        \begin{equation*}
            \begin{tikzcd}
                X \arrow[d, "u_X"'] \arrow[r, "f"] & Y \arrow[d, "u_Y"] \\
                \Sing \gr{X} \arrow[r, "\Sing \gr{f}"] & \Sing \gr{Y}
            \end{tikzcd}
        \end{equation*}
        and \ref{chV-5-6} that $f$ is a weak equivalence.\hfill$\Box$
        \medskip

        \emph{Proof of \ref{chV-5-1}}. \ref{chV-5-6} deals almost with the entire statement since it already tells us that $\Sing$ is essentially surjective on homotopy categories and that $\gr{-}$ induces weak equivalences on mapping spaces. By \ref{chV-5-7}, the functor $\gr{-}$ is also essentially surjective on homotopy categories.\\
        Using \ref{chV-5-7} a second time, the commutative diagram
        \begin{equation*}
            \begin{tikzcd}
                & F_{\sSet}(\Sing T, \Sing U) \arrow[dd, "\cong"] \\
                F_{\CW}(T, U) \arrow[ur, "\Sing"] \arrow[dr, "c^{*}_T"] & \\
                & F_{\CW}(\gr{\Sing T}, U)
            \end{tikzcd}
        \end{equation*}
        shows that $\Sing$ also induces weak equivalences on mapping spaces.\hfill$\Box$

        In the remainder of this section, we outline why simplicial approximation holds for simplicial sets. Our discussion follows Jardine, but many of the properties in which we are interested (plus many other, more subtle properties) are also studies in WJR13, Section 2. We assume the classical simplicial approximation theorem for simplicial complexes. To formulate it, we need to recall the barycentric subdivision of a simplicial complex: Given an (abstract) simplicial complex $(V, S)$, we define $\sd(V, S)$ as the abstract simplicial complex whose set of vertices is given by $S$, and whose $n$-simplices are given by strictly ascending chains $s_0 \subseteq s_1 \subseteq \ldots \subseteq s_n$ of length $n$.

    \subsection{Remark}\label{chV-5-9}
        Recall that any locally ordered simplicial complex gives rise to a simplicial set by formally adjoining degeneracies (see Exercise 3 on Sheet 3). Denote this functor by
        \begin{equation*}
            (-)^{+} \,\colon \SC^{\lo} \to \sSet \,.
        \end{equation*}
        Canonical examples of locally ordered simplicial complexes are those arising from posets: For a poset $P$, let $V(P)$ be the simplicial complex with vertex set $P$ and $n$-simplices strictly ascending chains $p_0 < \ldots p_n$. The simplicial complex $V(P)$ is often called the nerve of the poset $P$, but we will avoid that terminology because the nerve of a poset is a simplicial set for us. However, note that there is a natural isomorphism $NP \cong V(P)^{+}$, so the two notions are not unrelated.\\
        Moreover, it is straightforward to check that $(\sd V(P))^{+} \cong \sd V(P)^{+}$ for any poset $P$.

                sd^n V \arrow[urr, "f"'] & &

        \end{equation*}
        commutes and a homotopy $\gr{f} \simeq g \gr{\lambda^n}$ relative $\gr{\sd^n U}$.

        Our line of approach is to deduce a simplicial approximation theorem for simplicial sets by turning the simplicial sets in question into simplicial complexes. Here is a particularly brutal way of building a simplicial complex from simplicial set:

    \subsection{Definition}\label{chV-5-11}
        Let $X$ be a simplicial set. Let $\nd X$ denote the set of non-degenerate simplices of $X$. Denoting by $\langle x \rangle$ the smallest simplicial subset generated by a non-generate simplex $x$, equip $\nd X$ with the partial order given by
        \begin{equation*}
            x \leq y \quad :\Leftrightarrow \quad \langle x \rangle \subseteq \langle y \rangle \,.
        \end{equation*}
        The \emph{Barratt nerve} $\B X$ of $X$ is defined as
        \begin{equation*}
            \B X \coloneq N(\nd X) \,.
        \end{equation*}
        Any map $f \colon X \to Y$ of simplicial sets induces a map of posets $\nd f \colon \nd X \to \nd Y$ (since every simplex is the unique, possibly trivial, degeneration of an unique non-degenerate simplex), and thus induces a map $\B f \colon \B X \to \B Y$. Therefore, we obtain a functor
        \begin{equation*}
            \B \,\colon \sSet \to \sSet \,.
        \end{equation*}

    \subsection{Remark}\label{chV-5-12}
        In general, the Barratt nerve of a simplicial set has little to do with the original simplicial set. For example, $\B (\Delta^1 / \partial \Delta^1)$ is the nerve of the poset $[1]$ and therefore simplicially contractible.\\
        On the other hand, $\B (V, S)^{+} \cong \sd(V, S)^{+}$ for any locally ordered simplicial complex $(V, S)$.

    \subsection{Lemma}\label{chV-5-13}
        For every simplicial set $X$, the canonical maps
        \begin{equation*}
            \colim\limits_{x \in \nd X} \sd \langle x \rangle \to \sd X \quad \text{and} \quad \colim\limits_{x \in \nd X} \B \langle x \rangle \to \B X
        \end{equation*}
        are isomorphisms.
        \smallskip

        \emph{Proof}. Since $\colim\limits_{x \in \nd X} \sd \langle x \rangle \cong X$, the first part follows because $\sd$ is colimit-preserving.\\
        Let $\xi = x_0 \leq \ldots \leq x_0$ be an $n$-simplex in $\B X$. Then $\xi$ is also an $n$-simplex in $\langle x_n \rangle$. So $\colim\limits_{x \in \nd X} \B \langle x \rangle \to \B X$ is surjective.\\
        Suppose that $\xi = x_0 \leq \ldots \leq x_k \in \B \langle x \rangle$ and $\eta = y_0 \leq \ldots \leq y_l \in \B \langle y \rangle$ map to the same simplex in $\B X$. Then $\xi$ and $\eta$ define the same sequence in $\nd X$. In particular, $y_l = x_k$. Call this simplex $z$. Then $\xi$ and $\eta$ both lie in $\B \langle z \rangle$ and define the same element there. So $\xi = \eta$ in $\colim\limits_{x \in \nd X} \B \langle x \rangle$.\hfill$\Box$
        \medskip

        By \ref{chV-5-9}, the restrictions of $\sd$ and $\B$ along Yoneda embedding $\Delta \ \to \sSet$ agree. By the universal property of the left Kan extension, there is an induced natural transformation
        \begin{equation*}
            \pi \,\colon \sd \to \B \,.
        \end{equation*}

    \subsection{Lemma}\label{chV-5-14}
        Let $X$ be a simplicial set. the map $\pi \colon \sd X \to \B X$ is
        \begin{enumerate}[label=\roman*)]
            \item 
                surjective in all degrees;
            \item 
                bijective on vertices.
        \end{enumerate}
        In particular, two $n$-simplices in $\sd X$ map to the same simplex in $\B X$ if and only if they have the same set of vertices.
        \smallskip

        \emph{Proof}. By definition, an $n$-simplex in $\B X$ is a sequence $x_0 \leq \ldots \leq x_n$ in $\nd X$. Choose injective maps $\delta_i$ such that $x_i = \delta^{*}_i(x_n)$. Then each $\delta_i$ can be considered a simplex in $\Delta^{\gr{x_n}}$, and thus $\delta_0 \leq \ldots \leq \delta^n$ defines an $n$-simplex in $\sd \Delta^{\gr{x_n}}$. the pair $(\delta_0 \leq \ldots \leq \delta^n, x_n)$ defines an $n$-simplex in $\sd X$ which maps to the original $n$-simplex in $\B X$ under $\pi$.\\
        For any two vertices in $\sd X$, choose $k$, $l$ minimal such that they can be represented as
        \begin{equation*}
            \Delta^0 \xrightarrow{v} \sd \Delta^k \xrightarrow{\sd x} \sd X \quad \text{and} \quad \Delta^0 \xrightarrow{w} \sd \Delta^l \xrightarrow{\sd y} \sd X 
        \end{equation*}
        for non-degenerate simplices $x$ and $y$ of $X$. By the minimality of $k$ and $l$, $v = [k]$ and $w = [l]$. If $\pi_X \circ \sd x \circ v = \pi_X = \sd y \circ w$, then use $(\B X)_0 = \nd X$ and $\B \Delta^k \cong \sd \Delta^k$ to see that
        \begin{equation*}
            x = \B x(v) = \pi_X(\sd x(v)) = \pi_X(\sd y(w)) = \B y(w) = y \,.
        \end{equation*}
        \quad\hfill$\Box$

    \subsection{Definition}\label{chV-5-15}
        A non-degenerate simplex $x \in X_n$ is \emph{regular} if
        \begin{equation*}
            \begin{tikzcd}
                \Delta^{n-1} \arrow[d, "d_n"'] \arrow[r, "{d_n(x)}"] & \langle d_n(x) \rangle \arrow[d] \\
                \Delta^n \arrow[r, "x"] & \langle x \rangle
            \end{tikzcd}
        \end{equation*}
        is a pushout square.\\
        A simplicial set is \emph{regular} if all its non-degenerate simplices are regular.

    \subsection{Example}\label{chV-5-16}
        It the characteristic map of every non-degenerate simplex in $X$ is injective (i.e. $X$ is \emph{non-singular}), then $X$ is regular. In particular, the nerve of every poset is regular.

    \subsection{Proposition}\label{chV-5-17}
        The subdivision $\sd X$ is regular for every simplicial set $X$.
        \smallskip

        \emph{Proof}. Let $c \colon \Delta^n \to \sd X$ be a non-degenerate simplex. Choose $k$ minimal such that $c$ factors as
        \begin{equation*}
            c \,\colon \Delta^n \xrightarrow{\kappa} \sd \Delta^k \xrightarrow{\sd x} \sd \sk_k(X)
        \end{equation*}
        for some non-degenerate $k$-simplex $x$ of $X$ and some non-degenerate $n$-simplex $\kappa$ of $\sd \delta^k$ (if $\kappa$ was degenerate, $c$ would also be degenerate; if $x$ was degenerate, $k$ would not have been minimal). We may assume $k > 0$. Since $k$ is minimal, $\kappa(n) = [k]$. In particular, $d_n(\kappa)$ is a simplex in $\sd \partial \Delta^k$. Therefore, there is an induced commutative diagram
        \begin{equation*}
            \begin{tikzcd}
                \Delta^{n-1} \arrow[d, hook, "d_n"'] \arrow[r, "{d_n(\kappa)}"] & \sd \partial \Delta^k \arrow[d, hook] \arrow[r] & \sd \partial \langle x \rangle \arrow[d, hook] \\
                \Delta^n \arrow[r, "\kappa"] & \sd \Delta^k \arrow[r] & \sd \langle x \rangle
            \end{tikzcd}
        \end{equation*}
        where $\partial \langle x \rangle$ denotes the simplicial set $\langle x \rangle$ with the unique top-dimensional simplex $x$ removed. The right hand square is a pushout. Consider the outer square
        \begin{equation*}
            \begin{tikzcd}
                \Delta^{n-1} \arrow[d, hook, "d_n"'] \arrow[r, "{d_n(c)}"] & \sd \partial \langle x \rangle \arrow[d, hook] \\
                \Delta^n \arrow[r, "c"] & \sd \langle x \rangle
            \end{tikzcd}
        \end{equation*}
        If $y$ is a simplex in $\Delta^n$ which does not lie in $d_n \Delta^n$, its image contains $[n]$. Hence $c(y) \notin \sd \partial \langle x \rangle$. Consequently, this square is a pullback. Since this square fits into the commutative diagram
        \begin{equation*}
            \begin{tikzcd}
                \Delta^{n-1} \arrow[d, hook, "d_n"'] \arrow[r, "{d_n(\kappa)}"] & \langle d_n(c) \rangle \arrow[d, hook] \arrow[r, hook] & \sd \partial \langle x \rangle \arrow[d, hook] \\
                \Delta^n \arrow[r, "\kappa"] & \langle x \rangle \arrow[r, hook] & \sd \langle x \rangle
            \end{tikzcd}
        \end{equation*}
        and all arrows in the right hand square are monomorphisms as indicated, the square
        \begin{equation*}
            \begin{tikzcd}
                \Delta^{n-1} \arrow[d, hook, "d_n"'] \arrow[r, "{d_n(c)}"] & \langle d_n(c) \rangle \arrow[d, hook] \\
                \Delta^n \arrow[r, "c"] & \langle c \rangle
            \end{tikzcd}
        \end{equation*}
        is also a pullback. Therefore, it suffices to show that the induced map
        \begin{equation*}
            \Delta^n \setminus d_n \Delta^n \to \langle c \rangle \setminus \langle d_n(c) \rangle 
        \end{equation*}
        is a degree-wise bijection. This map is surjective since $c \colon \Delta^n \to \langle c \rangle$ is surjective.\\
        Suppose that $\sigma$, $\tau$ are in $\Delta^n \setminus d_n \Delta^n$ such that $c(\sigma) = c(\tau) \in \langle c \rangle \subseteq \sd \langle x \rangle$. Then $c(\sigma)$ and $c(\tau)$ factor through $\sd \Delta^k$, so $\kappa(\sigma) = \kappa(\tau)$. Since $\sd \Delta^k$ is non-singular and $\kappa$ is non-degenerate, $\sigma = \tau$. This proves the proposition.\hfill$\Box$

    \subsection{Definition}\label{chV-5-18}
        A simplicial set $X$ is \emph{op-regular} if $X^{\op}$ is regular, i.e. for every $n$-simplex $x$ the diagram
        \begin{equation*}
            \begin{tikzcd}
                \Delta^{n-1} \arrow[d, "d_0"'] \arrow[r, "{d_0(c)}"] & \langle d_0(x) \rangle \arrow[d] \\
                \Delta^n \arrow[r, "x"] & \langle x \rangle
            \end{tikzcd}
        \end{equation*}
        is a pushout. Let $\sSet^{\opreg}$ denote the full subcategory of $\sSet$ spanned by the op-regular simplicial sets.

    \subsection{Proposition}\label{chV-5-19}
        There exists a natural transformation
        \begin{equation*}
            \gamma \,\colon {\B|}_{\sSet^{\opreg}} \Rightarrow \id
        \end{equation*}
        such that ${\lambda|}_{\sSet^{\opreg}} = \gamma \circ {\pi|}_{\sSet^{\opreg}}$.
        \smallskip

        \emph{Proof}. If such a factorisation $\gamma_X$ exists for every op-regular simplicial set $X$ in some full subcategory $\Cc$ of $\sSet^{\opreg}$, then the collection $(\gamma_X)_{X \in \Cc}$ is automatically a natural transformation on $\Cc$: for every $f \colon X \to Y$ we have
        \begin{equation*}
            \pi_X \circ \B f \circ \gamma_Y = \gamma_y \circ \pi_Y \circ \sd f = \lambda_Y \circ \sd f = f \circ \lambda_X = f \circ \lambda_X \circ \pi_X \,.
        \end{equation*}
        Since $\pi$ is an epimorphism by \ref{chV-5-14}, naturality of $\gamma$ follows. So we only need to show that $\gamma_X$ exists for every $X \in \sSet^{\opreg}$.\\
        By \ref{chV-5-13} and the preliminary observation, the existence of $\gamma_X$ follows if we can show that for every non-degenerate simplex $x \in X$ there exists a commutative triangle
        \begin{equation*}
            \begin{tikzcd}
                \sd \langle x \rangle \arrow[dr, "\lambda_{\langle x \rangle}"] \arrow[rr, "\pi_{\langle x \rangle}"] & & \B \langle x \rangle \arrow[dl, dotted, "\gamma_x"] \\
                & \langle x \rangle &
            \end{tikzcd}
        \end{equation*}
        We proceed by induction on the dimension of $x$. If $x$ is a $0$-simplex, $\sd_{\op} \langle x \rangle$, $\B_{\op} \langle x \rangle$ and $\langle x \rangle$ are all isomorphic to $\Delta^0$, nd the existence of $\gamma_x$ follows trivially.\\
        Suppose that the dimension of $x$ is $n > 0$. By induction hypothesis, there exists a commutative triangle
        \begin{equation*}
            \begin{tikzcd}
                \sd \langle d_n(x) \rangle \arrow[dr, "{\lambda_{\langle d_0(x) \rangle}}"] \arrow[rr, "{\pi_{\langle d_0(x) \rangle}}"] & & \B \langle d_n(x) \rangle \arrow[dl, "{\gamma_{d_0(x)}}"] \\
                & \langle d_n(x) \rangle &
            \end{tikzcd}
        \end{equation*}
        Note the slight abuse of notation: $d_0(x)$ is not necessarily non-degenerate. However, it is the unique degeneration $d_0(x) = s(y)$ of a non-degenerate simplex $y$, and evidently $\langle y \rangle = \langle d_0(x) \rangle$.\\
        Since $X$ is op-regular, $\lambda_{\langle x \rangle}$ fits into a map of pushout squares (the front and back face of the cube):
        \begin{equation*}
            \begin{tikzcd}
                & \sd \Delta^{n-1} \arrow[ld, "\lambda"] \arrow[dd] \arrow[rr] & & \sd \langle d_0(x) \rangle \arrow[ld, "\lambda"] \arrow[dd] \\
                \Delta^{n-1} \arrow[dd, "d_0"] \arrow[rr, crossing over] & & \langle d_n(x) \rangle & \\
                & \sd \Delta^n \arrow[ld, "\lambda"] \arrow[rr] & & \sd \langle x \rangle \arrow[ld, "\lambda"] \\
                \Delta^n \arrow[rr, "x"] & & \langle x \rangle \arrow[from=uu, crossing over] &
            \end{tikzcd}
        \end{equation*}
        Consequently, we only need to factor the composite map
        \begin{equation*}
            \sd \Delta^n \xrightarrow{\sd x} \sd \langle x \rangle \xrightarrow{\lambda_{\langle x \rangle}} \langle x \rangle
        \end{equation*}
        in a way which is compatible with the given factorisation on $\sd \Delta^{n-1}$.\\
        Therefore, suppose that $\sigma, \tau \colon \Delta^k \to \sd \Delta^n$ are $k$-simplices such that $\pi_{\langle x \rangle} \circ \sd(x) \circ \sigma = \pi_{\langle x \rangle} \circ \sd(x) \circ \tau$. Write $\sigma = \sigma_0 \leq \ldots \leq \sigma_k$ and $\tau = \tau_0 \leq \ldots \leq \tau_k$. By \ref{chV-5-14} this means that
        \begin{equation*}
            \{ x \circ \sigma_0, \ldots, x \circ \sigma_k \} = \{ x \circ \tau_0, \ldots, x \circ \tau_k\} \,.
        \end{equation*}
        We have to show that
        \begin{equation*}
            \lambda_{\langle x \rangle} \circ \sd(x) \circ \sigma = \lambda_{\langle x \rangle} \circ \sd(x) \circ \tau \,.
        \end{equation*}
        Let $i$ be the smallest index such that $0$ lies in the image of $\sigma_i$ and $\tau_i$. Then $\sigma_j = \tau_j$ for all $j \geq i$ since $x$ is injective on that part of $\Delta^n$. Set $i \coloneq k +1$ if no such index exists.\\
        If $i = k + 1$, then $\sigma$ and $\tau$ both map to $\sd d_0 \Delta^n$. This case is covered by the induction hypothesis. So suppose $i < k + 1$.\\
        If $i = 0$, then $\sigma = \tau$, so obviously $\lambda \circ \sd(x) \circ \sigma = \lambda_{\langle x \rangle} \circ \sd(x) \circ \tau$.\\
        Suppose that $0 < i < k + 1$. Then $\lambda \circ \sigma_j = \lambda \circ d_0(\sigma_j)$ and $\lambda \circ \tau_j = \lambda d_0(\tau_j)$ for all $j \geq i$. Consequently, the simplices
        \begin{equation*}
            \sigma' \coloneq \sigma_0 \leq \ldots \leq \sigma_{i-1} \leq d_0(\sigma_i) \leq \ldots \leq d_0(\sigma_k)
        \end{equation*}
        and
        \begin{equation*}
            \tau' \coloneq \tau_0 \leq \ldots \leq \tau_{i-1} \leq d_0(\tau_i) \leq \ldots \leq d_0(\tau_k)
        \end{equation*}
        both lie in $\sd d_0 \Delta^n$ and map to the same simplex as $\sigma$ and $\tau$ under $\lambda \circ \sd(X)$, respectively. Since $x \circ d_0(\sigma_j) = x \circ \sigma_j \circ d_0 = x \circ \tau_j \circ d_0 = x \circ d_0(\tau_j)$, the simplices $\sigma'$ and $\tau'$ map to the same simplex under $\pi$, and therefore satisfy $\lambda \circ \sd(x) \circ \sigma' = \lambda \circ \sd(x) \circ \tau'$ by induction. This finishes the proof.\hfill$\Box$
        \medskip

        For the following discussion, abbreviate the composite functor
        \begin{equation*}
            (-)^{\op} \circ \sd \,\colon \sSet \to \sSet
        \end{equation*}
        by $\sd_{\op}$. By \ref{chV-5-17}, $\sd_{\op}$ takes values in the full subcategory of op-regular simplicial sets. It comes equipped with a natural transformation
        \begin{equation*}
            \varphi \,\colon \sd_{\op} \to \id_{\sSet} \,,
        \end{equation*}
        the \emph{first vertex map}, induced by the transformation $(\chains[n])^{\op} \to [n], \; c \mapsto \min c$.

    \subsection{Lemma}\label{chV-5-20}
        There is a natural isomorphism $\Phi \colon \sd \cong \sd \circ (-)^{\op}$.\\
        Let $\Psi \colon \sd^2 \cong \sd \sd_{\op}$ denote the induced isomorphism. Then there exists a simplicial homotopy
        \begin{equation*}
            \sd^2 X \times \Delta^1 \to X
        \end{equation*}
        from $\varphi \lambda \Psi$ to $\lambda^2$.
        \smallskip

        \emph{Proof}. It suffices to show that the lemma holds on the full subcategory of standard simplices.\\
        For any poset $P$, $\Phi$ is given by
        \begin{equation*}
            \Phi_P \,\colon \chains P \to \chains(P^{\op}), \quad c \mapsto c \text{ with reversed order}.
        \end{equation*}
        The transformation $\lambda^2$ is induced by
        \begin{equation*}
            \chains(\chains[n]) \to [n], \quad (c_0 \subseteq \ldots \subseteq c_r) \mapsto \max c_r \,,
        \end{equation*}
        and $\varphi \lambda \Psi$ is induced by
        \begin{equation*}
            \chains(\chains[n]) \to [n], \quad (c_0 \subseteq \ldots \subseteq c_r) \mapsto \min c_0 \,.
        \end{equation*}
        Since $\min c_0 \leq \max c_r$, these two maps combined define a natural map
        \begin{equation*}
            \chains(\chains[n]) \times [1] \to [n]
        \end{equation*}
        which induces the required homotopy.\hfill$\Box$

    \subsection{Theorem (Simplicial approximation for simplicial sets)}\label{chV-5-21}
        Let $A \subseteq X$ be finite simplicial sets and let $Y$ be a simplicial set. Suppose that $f_0 \colon A \to Y$ and $g \colon \gr{X} to \gr{Y}$ are maps such that ${g|}_{\gr{A}} = \gr{f_0}$.\\
        Then there exists for some $k$ a map $f \colon \sd^k \sd_{\op} \sd \sd_{\op} X \to Y$ such that
        \begin{equation*}
            \begin{tikzcd}
                \sd^k \sd_{\op} \sd \sd_{\op} A \arrow[d] \arrow[r, "\varphi \lambda \varphi \lambda^k"] & A \arrow[r, "f_0"] & Y \\
                \sd^k \sd_{\op} \sd \sd_{\op} X \arrow[urr, "f"']
            \end{tikzcd}
        \end{equation*}
        commutes and a homotopy $\gr{f} \simeq g \gr{\varphi \lambda \varphi \lambda^k}$ relative $\gr{\sd^k \sd_{\op} \sd \sd_{\op} A}$.
        \smallskip

        \emph{Proof}. Consider the commutative diagram
        \begin{equation*}
            \begin{tikzcd}[column sep=large]
                \gr{\sd \sd_{\op} A} \arrow[d, "\gr{\sd \sd_{\op} i}"'] \arrow[r, "\gr{\sd \sd_{\op} f_0}"] & \gr{\sd \sd_{\op} Y} \arrow[d, "\gr{\varphi \lambda}"] \\
                \gr{\sd \sd_{\op} X} \arrow[r, "g \gr{\varphi \lambda}"] & \gr{Y}
            \end{tikzcd}
        \end{equation*}
        Since $\sd \sd_{\op} i$ is a cofibration (even the inclusion of a sub-CW complex) and $\gr{\varphi \lambda}$ is a homotopy equivalence by \ref{chV-5-20} and \ref{chV-4-19}, there exists a map
        \begin{equation*}
            \overline{g} \,\colon \gr{\sd \sd_{\op} X} \to \gr{\sd \sd_{\op} Y}
        \end{equation*}
        such that $\overline{g} \circ \gr{\sd \sd_{\op} i} = \gr{\sd \sd_{\op} f_0}$ and $\gr{\varphi \lambda} \circ \overline{g} \simeq g \gr{\varphi \lambda}$ relative $\gr{\sd \sd_{\op} A}$.\\
        Using \ref{chV-5-17} and \ref{chV-5-19}, we can consider the maps
        \begin{equation*}
            \begin{tikzcd}[column sep=huge]
                \B \sd_{\op} \sd \sd_{\op} A \arrow[d, "\B \sd_{\op} \sd \sd_{\op} i"'] \arrow[r, "\B \sd_{\op} \sd \sd_{\op} f_0"] & \B \sd_{\op Y} \\
                \B \sd_{\op} \sd \sd_{\op} X &
            \end{tikzcd}
        \end{equation*}
        and
        \begin{equation*}
            \widetilde{g} \,\colon \gr{\B \sd_{\op} \sd \sd_{\op} X} \xrightarrow{\gr{\varphi \gamma}} \gr{\sd \sd_{\op} X} \xrightarrow{\overline{g}} \gr{\sd \sd_{\op} Y} \xrightarrow{\pi} \gr{\B \sd_{\op} Y} \,.
        \end{equation*}
        \ref{chV-5-10} applies to these to yield a commutative diagram
        \begin{equation*}
            \begin{tikzcd}[column sep=huge]
                \sd^n \B \sd_{\op} \sd \sd_{\op} A \arrow[d] \arrow[r, "\lambda^n"] & \B \sd_{\op} \sd \sd_{\op} A \arrow[r, "\B \sd_{\op} \sd \sd_{\op} f_0"] & \B \sd_{\op} Y \\
                \sd^n \B \sd_{\op} \sd \sd_{\op} X \arrow[urr, "f"']
            \end{tikzcd}
        \end{equation*}
        such that $\gr{f}$ is homotopic to $\widetilde{g} \gr{\lambda^n}$ relative $\gr{\sd^n \B \sd_{\op} \sd \sd_{\op} A}$. These maps can be composed with the appropriate instances of $\pi$ and $\varphi$ to obtain a commutative diagram
        \begin{equation*}
            \begin{tikzcd}[column sep=large]
                \sd^{n+1} \sd_{\op} \sd \sd_{\op} A \arrow[d] \arrow[r, "\varphi \lambda \varphi \lambda^{n+1}"] & A \arrow[r, "f_0"] & Y \\
                \sd^{n+1} \sd_{\op} \sd \sd_{\op} X \arrow[urr, "f \circ \sd^n \pi"']
            \end{tikzcd}
        \end{equation*}
        such that
        \begin{equation*}
            \gr{f \circ \sd^n \pi} \simeq \gr{\varphi \gamma \gr{\widetilde{g}} \lambda^n \sd^n \pi} \simeq g \gr{\varphi \lambda \varphi \lambda^{n+1}}
        \end{equation*}
        relative $\gr{\sd^{n+1} \sd_{\op} \sd \sd_{\op} A}$.\hfill$\Box$

    \subsection{Corollary}\label{chV-5-22}
        All statements listed in \ref{chV-5-6} hold.
        \smallskip

        \emph{Proof sketch}. We prove \ref{chV-5-6-1}. In this situation, \ref{chV-5-21} gives a simplicial approximation
        \begin{equation*}
            \begin{tikzcd}
                \sd^k \sd_{\op} \sd \sd_{\op} A \arrow[d] \arrow[r, "\varphi \lambda \varphi \lambda^k"] & A \arrow[r, "f_0"] & Y \\
                \sd^k \sd_{\op} \sd \sd_{\op} X \arrow[urr, "f'"']
            \end{tikzcd}
        \end{equation*}
        Compose this approximation with the canonical map $\rho^{\infty} \colon K \to \Ex^{\infty} Y$. Since $A \to X$ is a cofibration and $\Ex^{\infty} Y$ is Kan, there exists by \ref{chV-5-20} a map
        \begin{equation*}
            f \,\colon \sd^{k+3} X \to \Ex^{\infty} Y
        \end{equation*}
        such that $f$ is simplicially homotopic to $\sd^{k+3} X \cong \sd^k \sd_{\op} \sd \sd_{\op} X \xrightarrow{\rho^{\infty} \circ f} \Ex^{\infty} Y$ and extends $\rho^{\infty} f_0 \lambda^{k+3}$.\\
        As $X$ is assumed finite, $\sd^{k+3} X$ is also finite, so the approximation factors through some $\Ex^n Y$. Adjoining to a map $\sd^{k+n+3} X \to Y$ yields the desired approximation.\hfill$\Box$
